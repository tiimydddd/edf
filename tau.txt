local function nameIn(obj, list)
    local n = (obj.Name or ""):lower()
    for _, v in ipairs(list) do
        if n:find(v:lower(), 1, true) then return true end
    end
    return false
end

local function isShelterDoor(obj) return nameIn(obj, ShelterDoorNames) end
local function isChest(obj) return nameIn(obj, ChestNames) end
local function isCard(obj)
    if not nameIn(obj, CardNames) then return false end
    if nameIn(obj, StoreNames) then return false end
    return true
end
local function isStore(obj) return nameIn(obj, StoreNames) end

local function bestPart(obj)
    if obj:IsA("BasePart") then return obj end
    return obj:FindFirstChildWhichIsA("BasePart", true)
end

local function isDangerDisaster(name)
    if not name then return false end
    local n = tostring(name):lower()
    for _, d in ipairs(DangerDisasterList) do
        if n:find(d:lower(), 1, true) then return true end
    end
    return false
end

local function isDustBowl(name)
    if not name then return false end
    local n = tostring(name):lower()
    for _, v in ipairs(DustBowlNames) do
        if n:find(v:lower(), 1, true) then return true end
    end
    return false
end

local function isBlueMoonMall()
    for _, obj in ipairs(workspace:GetDescendants()) do
        local n = (obj.Name or ""):lower()
        if n:find("blue moon mall") or n:find("bluemoonmall") then return true end

        if isStore(obj) then return true end

local function isDangerAlertOn()
    local values = Workspace:FindFirstChild("Values")
    if not values then return false end
    local da = values:FindFirstChild("DangerAlert")
    if not da then return false end
    local val = da.Value
    if typeof(val) == "boolean" then return val end
    if typeof(val) == "number" then return val ~= 0 end
    if typeof(val) == "string" then
        local s = val:lower()
        return s == "true" or s == "on" or s == "1" or s == "yes"
    end
    return val and true or false
end

local function updateDisasterFromValues()
    local values = Workspace:FindFirstChild("Values")
    if not values then return end
    local cd = values:FindFirstChild("CurrentDisaster")
    if cd then
        local v = cd.Value
        if typeof(v) == "string" and v ~= "" then currentDisaster = v
        elseif v ~= nil then currentDisaster = tostring(v) end
    end
    dangerAlerts = isDangerAlertOn()
    theEndActive = tostring(currentDisaster):lower():find("the end") ~= nil
end

local function tryInteract(obj, hrp)
    local part = bestPart(obj)
    if not part then return false end
    local offsets = {
        Vector3.new(0, 3, 0), Vector3.new(0, 2.5, 2), Vector3.new(0, 2.5, -2),
        Vector3.new(2, 2.5, 0), Vector3.new(-2, 2.5, 0),
    }
    local opened = false
    for _, offset in ipairs(offsets) do
        pcall(function() hrp.CFrame = CFrame.new(part.Position + offset) end)
        task.wait(0.1)
        for _, d in ipairs(obj:GetDescendants()) do
            if d:IsA("ProximityPrompt") then
                for i = 1, 12 do
                    pcall(function() fireproximityprompt(d) end)
                    if i % 3 == 0 then task.wait(0.03) end
                end
                opened = true
            elseif d:IsA("ClickDetector") then
                for i = 1, 8 do pcall(function() fireclickdetector(d) end) end
                opened = true
            end
        end
        if opened then break end
    end
    return opened
end

local function tryGetCard()
    local hrp = getHRP()
    if not hrp then return false end
    local storePart = nil
    for _, obj in ipairs(workspace:GetDescendants()) do
        if isStore(obj) then
            storePart = bestPart(obj)
            if storePart then break end
        end
    end
    if storePart then
        hrp.CFrame = storePart.CFrame + Vector3.new(0, 4, 0)
        task.wait(0.25)
    end
    local got = false
    for _, obj in ipairs(workspace:GetDescendants()) do
        if isCard(obj) then
            if tryInteract(obj, hrp) then got = true gotCard = true end
        end
    end
    return got
end

local function rejoinUntilDanger()
    if not Settings.RejoinUntilDanger then
        rejoinTeleporting = false
        return
    end

    if isDangerAlertOn() then
        Settings.RejoinUntilDanger = false
        rejoinTeleporting = false
        RejoinDangerBtn.Text = "重进直到危险警报 ON"
        RejoinDangerBtn.BackgroundColor3 = T.Off
        StatusLabel.Text = "  Status: Danger Alert 已开，停止重进"
        Notify("Danger Alert 已开，停止重进")
        return
    end

    if rejoinTeleporting then
        StatusLabel.Text = "  Status: 正在重进，请等服务器..."
        return
    end

    local now = tick()
    if lastKickTime <= 0 then
        lastKickTime = now
    end

    local waited = now - lastKickTime
    if waited < REJOIN_AFTER_KICK_WAIT then
        local left = math.ceil(REJOIN_AFTER_KICK_WAIT - waited)
        RejoinDangerBtn.Text = "等待服务器 " .. left .. " 秒..."
        StatusLabel.Text = "  Status: 服务器未完全重开，还要等 " .. left .. " 秒"
        return
    end

    rejoinTeleporting = true
    StatusLabel.Text = "  Status: 服务器应已重开，正在重进..."
    Notify("尝试重进服务器...")

    task.delay(1.5, function()
        if not Settings.RejoinUntilDanger then
            rejoinTeleporting = false
            return
        end
        if isDangerAlertOn() then
            Settings.RejoinUntilDanger = false
            rejoinTeleporting = false
            RejoinDangerBtn.Text = "重进直到危险警报 ON"
            RejoinDangerBtn.BackgroundColor3 = T.Off
            Notify("Danger Alert 已开，取消重进")
            return
        end

        pcall(function()
            TeleportService:Teleport(PLACE_ID, LocalPlayer)
        end)

        task.delay(5, function()
            rejoinTeleporting = false
            if lastKickTime < now then
                lastKickTime = tick()
            end
        end)
    end)
end
