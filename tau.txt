local function inL(name, list)
    local n = (name or ""):lower()
    for _, v in ipairs(list) do
        if n == v:lower() or n:find(v:lower(), 1, true) then return true end
    end
    return false
end

local function isF(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, FragN) then return false end
    if o.Parent and inL(o.Parent.Name, FragN) then return false end
    return true
end

local function isG(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, GlassN) then return false end
    if o.Parent and inL(o.Parent.Name, GlassN) then return false end
    return true
end

local function isS(o) return inL(o.Name, ShelN) end
local function HRP() local c = LP.Character return c and c:FindFirstChild("HumanoidRootPart") end
local function Hum() local c = LP.Character return c and c:FindFirstChildOfClass("Humanoid") end

local function best(o)
    if o:IsA("BasePart") then return o end
    local p = o:FindFirstChildWhichIsA("ProximityPrompt", true)
    if p and p.Parent and p.Parent:IsA("BasePart") then return p.Parent end
    return o:FindFirstChildWhichIsA("BasePart", true)
end

local function clean(t)
    for o in pairs(t) do
        if not o or not o.Parent then t[o] = nil end
    end
end

local function cntF()
    clean(kF)
    local n = 0
    for _ in pairs(kF) do n = n + 1 end
    return n
end

local function scan()
    local c = {Fragment = 0, Glasses = 0, Key = 0, Ext = 0, Bat = 0}
    local seen = {}
    for _, o in ipairs(workspace:GetDescendants()) do
        if o:IsA("Model") or o:IsA("Tool") or o:IsA("BasePart") or o:IsA("Folder") then
            local id = tostring(o)
            if not seen[id] then
                if isF(o) then
                    c.Fragment = c.Fragment + 1
                    kF[o] = true
                    seen[id] = true
                elseif isG(o) then
                    c.Glasses = c.Glasses + 1
                    kG[o] = true
                    seen[id] = true
                elseif inL(o.Name, KeyN) and not (o.Parent and inL(o.Parent.Name, KeyN)) then
                    c.Key = c.Key + 1
                    kE[o] = "Key"
                    seen[id] = true
                elseif inL(o.Name, ExtN) and not (o.Parent and inL(o.Parent.Name, ExtN)) then
                    c.Ext = c.Ext + 1
                    kE[o] = "Ext"
                    seen[id] = true
                elseif inL(o.Name, BatN) and not (o.Parent and inL(o.Parent.Name, BatN)) then
                    c.Bat = c.Bat + 1
                    kE[o] = "Bat"
                    seen[id] = true
                end
            end
        end
    end
    mapC = c
    return c
end

local function fireP(pr)
    if not pr then return end
    pcall(function()
        if fireproximityprompt then
            fireproximityprompt(pr)
            task.wait(0.03)
            fireproximityprompt(pr, 1)
        end
    end)
end

local function pickup(o, typ)
    if not o or not o.Parent then return end
    local key = tostring(o)
    if tried[key] and tick() - tried[key] < 1.8 then return end
    tried[key] = tick()
    local h = HRP()
    if not h then return end
    local part = best(o)
    if not part then return end
    local offs = {
        Vector3.new(0, 2, 0),
        Vector3.new(0, 1.6, 1.5),
        Vector3.new(0, 1.6, -1.5),
        Vector3.new(1.5, 1.6, 0),
        Vector3.new(-1.5, 1.6, 0)
    }
    for _, off in ipairs(offs) do
        if not o.Parent then break end
        h.CFrame = part.CFrame + off
        task.wait(0.03)
        fireP(o:FindFirstChildWhichIsA("ProximityPrompt", true))
        local ti = part:FindFirstChildOfClass("TouchInterest")
        if ti then
            pcall(function()
                firetouchinterest(h, part, 0)
                task.wait(0.02)
                firetouchinterest(h, part, 1)
            end)
        end
        local cd = o:FindFirstChildWhichIsA("ClickDetector", true)
        if cd then pcall(fireclickdetector, cd) end
        task.wait(0.03)
    end
    task.delay(0.6, function()
        if not o or not o.Parent then
            if typ == "Fragment" then
                myF = math.min(5, myF + 1)
                sF = sF + 1
                if myF >= 5 then
                    alrt()
                    showBig("碎片个人已满 5/5！\n（仍需全服共5个）", Color3.fromRGB(100, 255, 120), 7)
                    N("个人碎片 5/5！全服集齐才出 Tartarus", true)
                else
                    N("碎片 " .. myF .. "/5")
                end
            elseif typ == "Glasses" then
                gotG = true
                sG = sG + 1
                N("眼镜已找到")
            end
        end
    end)
end

end

-- 音乐相关
local function changeGameMusic(id)
    local success = false
    pcall(function()
        id = tostring(id or "")
        if id == "" or id == "0" then
            N("请输入有效的 rbxassetid")
            return
        end
        if not string.find(id, "rbxassetid://") then
            id = "rbxassetid://" .. id
        end

        local targets = {}
        local musicFolder = SS:FindFirstChild("Music")
        if musicFolder then
            local gamemode = musicFolder:FindFirstChild("Gamemode")
            local bonus = musicFolder:FindFirstChild("BonusDisaster")
            if gamemode and gamemode:IsA("Sound") then table.insert(targets, gamemode) end
            if bonus and bonus:IsA("Sound") then table.insert(targets, bonus) end
        end

        if #targets == 0 then
            N("没找到 Gamemode 或 BonusDisaster")
            return
        end

        for _, s in ipairs(targets) do
            if not originalIds[s] then
                originalIds[s] = s.SoundId
            end
            s.SoundId = id
        end
        N("已更换音乐")
        success = true
    end)
    return success
end

local function restoreGameMusic()
    pcall(function()
        for s, oldId in pairs(originalIds) do
            if s and s.Parent then
                s.SoundId = oldId
            end
        end
        N("已恢复原来的音乐")
    end)
end

local function addToHistory(name, id)
    name = tostring(name or ""):gsub("^%s*(.-)%s*$", "%1")
    id = tostring(id or "")
    if name == "" then name = "未命名" end
    if id == "" then return end

    for i, item in ipairs(musicHistory) do
        if item.id == id or item.id == "rbxassetid://" .. id then
            table.remove(musicHistory, i)
            break
        end
    end

    table.insert(musicHistory, 1, {name = name, id = id})
    while #musicHistory > MAX_HISTORY do
        table.remove(musicHistory)
    end
    refreshMusicList()
end

local function enableNoInjury()
    if injuryConn then injuryConn:Disconnect() end
    injuryConn = RS.Heartbeat:Connect(function()
        if not C.NoInjury then return end
        local hum = Hum()
        local hrp = HRP()
        if not hum or not hrp then return end
        pcall(function()
            local state = hum:GetState()
            if state == Enum.HumanoidStateType.FallingDown
            or state == Enum.HumanoidStateType.Ragdoll
            or state == Enum.HumanoidStateType.Physics then
                hum:ChangeState(Enum.HumanoidStateType.GettingUp)
                hum:ChangeState(Enum.HumanoidStateType.Running)
            end
            if hum.WalkSpeed < 16 and not C.SpdOn then
                hum.WalkSpeed = 16
            end
            hum.PlatformStand = false
            hum.Sit = false
        end)
    end)
end

local function disableNoInjury()
    if injuryConn then
        injuryConn:Disconnect()
        injuryConn = nil
    end
end

local function checkFirst()

        rejSt = "已关闭"

    if not entered then
        rejSt = "等待进入地图..."
        return
    end
    local left = math.ceil(8 - (tick() - enterT))
    if left > 0 then
        rejSt = "进图检测: " .. left .. "秒"

    firstChk = true
    scan()

        task.spawn(alertNo)
    else
        rejSt = "有碎片(" .. fr .. ")"
        N("碎片: " .. fr)
    end
end

local gC, hC, fC
local function forceHP()
    local h = Hum()
    if not h then return end
    pcall(function()
        if h.Health < h.MaxHealth then h.Health = h.MaxHealth end
        if h:GetState() == Enum.HumanoidStateType.Dead then
            h:ChangeState(Enum.HumanoidStateType.Running)
            h.Health = h.MaxHealth
        end
    end)
end

local function offGod()
    if gC then gC:Disconnect() gC = nil end
    if hC then hC:Disconnect() hC = nil end
    if fC then fC:Disconnect() fC = nil end
end

local function onGod()
    offGod()
    gC = RS.Heartbeat:Connect(function()
        if C.God then forceHP() end
    end)
    local ch = LP.Character
    if ch then
        local h = ch:FindFirstChildOfClass("Humanoid")
        if h then
            hC = h.HealthChanged:Connect(function()
                if C.God then forceHP() end
            end)
        end
    end
    fC = RS.Heartbeat:Connect(function()
        if not C.God then return end
        local hr = HRP()
        if hr and hr.AssemblyLinearVelocity.Magnitude > 80 then
            hr.AssemblyLinearVelocity = Vector3.zero
            hr.AssemblyAngularVelocity = Vector3.zero
        end
    end)
end

local function doTorn()
    local h = HRP()
    if not h then return end
    local pos = h.Position
    local ch = LP.Character
    local parts
    local ok = pcall(function()
        parts = workspace:GetPartBoundsInRadius(pos, TRANGE)
    end)
    if not ok or not parts then return end
    for _, o in ipairs(parts) do
        if o:IsA("BasePart") and not o.Anchored and o.Parent then
            if not (ch and o:IsDescendantOf(ch)) then
                local d = (o.Position - pos).Magnitude
                if d >= 9 then
                    local dir = (pos - o.Position).Unit
                    local sf = 1 - (d / TRANGE)
                    if d < 20 then sf = sf * 0.3 end
                    local f = dir * (110 * sf) + Vector3.new(0, 22 * sf * 0.5, 0)
                    if isF(o) or (o.Parent and isF(o.Parent)) then
                        f = f * 1.45
                    end
                    pcall(function() o.AssemblyLinearVelocity = f end)
                end
            end
        end
    end
end

                bv.Parent = p
                task.delay(1.2, function()
                    if bv and bv.Parent then bv:Destroy() end
                end)
            end
        end
    end
end

local function startSpec()
    click()
    local ok, err = pcall(function()
        if typeof(firesignal) ~= "function" then error("无 firesignal") end
        local e = RPS:FindFirstChild("Assets") and RPS.Assets:FindFirstChild("Signals") and RPS.Assets.Signals:FindFirstChild("Spectate")
        if not e then error("无 Spectate") end
        if isSpec then
            pcall(function() firesignal(e.OnClientEvent, {A = false, reason = "ExitSpectate"}) end)
            pcall(function() firesignal(e.OnClientEvent, {A = false}) end)
            isSpec = false
            SpecB.Text = "Open Spectate"
            N("退出观战")
        else
            firesignal(e.OnClientEvent, {A = true, reason = "EnterSpectate"})
            isSpec = true
            SpecB.Text = "Exit Spectate"
            N("进入观战")
        end
    end)
    if not ok then
        isSpec = false
        SpecB.Text = "Open Spectate"
        N("观战: " .. tostring(err))
    end
end

task.spawn(function()
    while Gui and Gui.Parent do
        pcall(function()
            local now = tick()
            local c = mapC
            local fr = c.Fragment or cntF()

            if C.FA then
                StatusL.Text = "状态: Full Auto"
                StatusL.TextColor3 = Color3.fromRGB(100, 220, 140)
            elseif C.En then
                StatusL.Text = "状态: 仅自动捡取"
                StatusL.TextColor3 = Color3.fromRGB(100, 200, 255)
            else
                StatusL.Text = "状态: 关闭"
                StatusL.TextColor3 = Color3.fromRGB(255, 100, 100)
            end

            FragL.Text = string.format("血月碎片: 地图%d | 我%d/5", fr, myF)
            if myF >= 5 then
                FragFullL.Text = "个人碎片: 已满 5/5 ✓"
                FragFullL.TextColor3 = Color3.fromRGB(100, 255, 120)
            else
                FragFullL.Text = "个人碎片: " .. myF .. "/5"
                FragFullL.TextColor3 = Color3.fromRGB(180, 180, 195)
            end

            if gotG or (c.Glasses or 0) > 0 then
                GlassL.Text = "神秘眼镜: 已找到"
                GlassL.TextColor3 = Color3.fromRGB(100, 255, 120)
                ChadL.Text = "Chad条件: 已满足"
            else
                GlassL.Text = "神秘眼镜: 未找到"
                GlassL.TextColor3 = Color3.fromRGB(255, 160, 100)
                ChadL.Text = "Chad条件: 未满足"
            end

            TartL.Text = tart and "Tartarus: 已出现" or "Tartarus: 未解锁"
            RejL.Text = "重进提示: " .. rejSt
            SessL.Text = string.format("会话: 碎片%d 眼镜%d 换服%d", sF, sG, sH)
            MapL.Text = string.format("地图: 钥匙%d 灭火器%d 球棒%d", c.Key or 0, c.Ext or 0, c.Bat or 0)

            if now - tHeavy >= 2.5 then
                tHeavy = now
                scan()
                updDist()
            end
            if now - tEsp >= 0.5 then
                tEsp = now
                refESP()
            end

            if (C.En or C.FA) and now - tScan >= 0.45 then
                tScan = now
                clean(kF)
                clean(kG)
                clean(kE)
                local h = HRP()
                local pos = h and h.Position or Vector3.zero
                local fl = {}
                for o in pairs(kF) do
                    if o and o.Parent then
                        local p = best(o)
                        if p then table.insert(fl, {o = o, d = (p.Position - pos).Magnitude}) end
                    end
                end
                table.sort(fl, function(a, b) return a.d < b.d end)
                for _, d in ipairs(fl) do
                    pickup(d.o, "Fragment")
                    task.wait(0.1)
                end
                if not C.OnlyF then
                    for o in pairs(kG) do
                        if o and o.Parent then
                            pickup(o, "Glasses")
                            task.wait(0.1)
                        end
                    end
                end
                for o, k in pairs(kE) do
                    if o and o.Parent then
                        local w = (k == "Key" and C.Key) or (k == "Ext" and C.Ext) or (k == "Bat" and C.Bat)
                        if w then
                            pickup(o, k)
                            task.wait(0.1)
                        end
                    end
                end
            end

            if C.FA and C.FAShel and now - tShel >= 28 then
                tShel = now
                toShel(true)
            end
            if C.Torn and now - tTorn >= 0.35 then
                tTorn = now
                doTorn()
            end
            if C.Fly and now - tFly >= 1 then
                tFly = now
                doFly()
            end
            checkFirst()
        end)
        task.wait(0.12)
    end
end)

        drag = true

        p0 = Main.Position
    end
end)

    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        drag = false
    end

UIS.InputChanged:Connect(function(i)
    if drag and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - d0
        Main.Position = UDim2.new(p0.X.Scale, p0.X.Offset + d.X, p0.Y.Scale, p0.Y.Offset + d.Y)
    end
end)
