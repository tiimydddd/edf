-- ==================== Functions ====================
local function Notify(msg)
    pcall(function()
        StarterGui:SetCore("SendNotification", {Title = "TAU", Text = tostring(msg), Duration = 3})
    end)
    print("[TAU]", msg)
end

local function getHRP()
    local c = LocalPlayer.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function isFragment(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(FragmentNames) do
        if n:find(v:lower()) then return true end
    end
    return false
end

local function isGlasses(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(GlassesNames) do
        if n:find(v:lower()) then return true end
    end
    return false
end

local function isShelterKey(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(ShelterKeyNames) do
        if n:find(v:lower()) then return true end
    end
    return false
end

local function cleanCache(cache)
    for obj in pairs(cache) do
        if not obj or not obj.Parent then cache[obj] = nil end
    end
end

local function countFrags()
    cleanCache(knownFragments)
    local c = 0
    for _ in pairs(knownFragments) do c = c + 1 end
    return c
end

local function playSound()
    if not Settings.SpawnSound then return end
    local s = Instance.new("Sound")
    s.SoundId = "rbxassetid://6026984224"
    s.Volume = 1.2
    s.Parent = SoundService
    s:Play()
    task.delay(1.5, function() if s then s:Destroy() end end)
end

local function pickup(obj, typ)
    local hrp = getHRP()
    if not hrp then return end
    local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
    if not part then return end
    hrp.CFrame = part.CFrame + Vector3.new(0, 3, 0)
    local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
    if prompt then pcall(fireproximityprompt, prompt) end
    local touch = part:FindFirstChildOfClass("TouchInterest")
    if touch then
        pcall(firetouchinterest, hrp, part, 0)
        task.wait(0.03)
        pcall(firetouchinterest, hrp, part, 1)
    end
    local click = obj:FindFirstChildOfClass("ClickDetector") or part:FindFirstChildOfClass("ClickDetector")
    if click then pcall(fireclickdetector, click) end
    if typ == "Fragment" then
        collectedFragments = collectedFragments + 1
        Notify("Fragment " .. collectedFragments .. "/5")
        playSound()
    elseif typ == "Glasses" then
        glassesFound = true
        Notify("Glasses found")
        playSound()
    end
end

local function clearESP()
    if espFolder then espFolder:Destroy() espFolder = nil end
end

local function updateESP()
    clearESP()
    if not (Settings.ESPFragments or Settings.ESPGlasses) then return end
    espFolder = Instance.new("Folder")
    espFolder.Name = "TAU_ESP"
    espFolder.Parent = CoreGui
    if Settings.ESPFragments then
        for obj in pairs(knownFragments) do
            if obj and obj.Parent then
                local h = Instance.new("Highlight")
                h.Adornee = obj
                h.FillColor = Color3.fromRGB(255, 40, 40)
                h.OutlineColor = Color3.fromRGB(255, 255, 255)
                h.FillTransparency = 0.5
                h.Parent = espFolder
            end
        end
    end
    if Settings.ESPGlasses then
        for obj in pairs(knownGlasses) do
            if obj and obj.Parent then
                local h = Instance.new("Highlight")
                h.Adornee = obj
                h.FillColor = Color3.fromRGB(60, 160, 255)
                h.OutlineColor = Color3.fromRGB(255, 255, 255)
                h.FillTransparency = 0.5
                h.Parent = espFolder
            end
        end
    end
end

local function flyParts()
    local hrp = getHRP()
    if not hrp then return end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and not obj.Anchored and not obj:IsDescendantOf(LocalPlayer.Character) then
            if (obj.Position - hrp.Position).Magnitude <= 80 then
                local old = obj:FindFirstChild("FlyForce")
                if old then old:Destroy() end
                local bv = Instance.new("BodyVelocity")
                bv.Name = "FlyForce"
                bv.MaxForce = Vector3.new(0, 1e9, 0)
                bv.Velocity = Vector3.new(0, 42, 0)
                bv.Parent = obj
                task.delay(2, function() if bv then bv:Destroy() end end)
            end
        end
    end
end

-- Strict + Complete Mode Detection
local function checkText()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end
    local foundMode, foundIsNext = nil, false

    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextLabel") or g:IsA("TextButton") then
            local t = (g.Text or ""):lower()

            if Settings.DetectOverload and (t:find("system overload") or t:find("系统过载")) then
                if not systemOverload then systemOverload = true Notify("System Overload!") end
            end
            if t:find("this round's gamemode is") or t:find("gamemode is:") then foundIsNext = true end

            if t:find("toastnado") then foundMode = "ToastNado"
            elseif t:find("a dust devil is occurring") or t:find("dust devil is occurring") then foundMode = "Dust Devil"
            elseif t:find("infernado is occurring") or t:find("an infernado") then foundMode = "Infernado"
            elseif t:find("double trouble") then foundMode = "Double Trouble"
            elseif t:find("triple threat") then foundMode = "Triple Threat"
            elseif t:find("four tornadoes") or t:find("4-nado is occurring") then foundMode = "4-Nado"
            elseif t:find("chompnado") then foundMode = "Chompnado"
            elseif t:find("deadnado") then foundMode = "Deadnado"
            elseif t:find("trollnado") then foundMode = "Trollnado"
            elseif t:find("lurker") then foundMode = "Lurker"
            elseif t:find("unstoppable") then foundMode = "Unstoppable"
            elseif t:find("rainwrapped") then foundMode = "Rainwrapped"
            elseif t:find("freeze-nado") or t:find("freezenado") then foundMode = "Freeze-Nado"
            elseif t:find("ghostnado") then foundMode = "GhostNado"
            elseif t:find("megatornado") then foundMode = "MegaTornado"
            elseif t:find("riftnado") then foundMode = "Riftnado"
            elseif t:find("timebomb") then foundMode = "Timebomb Terror"
            elseif t:find("bombs away") then foundMode = "Bombs Away"
            elseif t:find("low-gravity") or t:find("low gravity") then foundMode = "Low-Gravity"
            elseif t:find("skirmish") then foundMode = "Skirmish"
            elseif t:find("taijitu") then foundMode = "Taijitu"
            elseif t:find("krampus") then foundMode = "Krampus"
            elseif t:find("hoshinado") then foundMode = "Hoshinado"
            elseif t:find("noobnado") then foundMode = "NoobNado"
            elseif t:find("retronado") or t:find("retro-nado") then foundMode = "Retro-Nado"
            elseif t:find("tartarus") then foundMode = "Tartarus"
            elseif t:find("doomsday") then foundMode = "Doomsday"
            elseif t:find("chad!") or t:find("chad is") then foundMode = "Chad!"
            elseif t:find("twisty the clown") or t:find("twisty!!!!") then foundMode = "Twisty The Clown"
            elseif t:find("normal survival") then foundMode = "Normal Survival"
            elseif t:find("zeus") then foundMode = "Zeus"
            end
        end
    end

    if foundMode then
        currentMode = foundMode
        isNext = foundIsNext
        lastModeTime = tick()
    elseif tick() - lastModeTime > 18 then
        currentMode = "Unknown"
        isNext = false
    end
end

workspace.DescendantAdded:Connect(function(obj)
    task.defer(function()
        if isFragment(obj) then
            knownFragments[obj] = true
            if Settings.Enabled then Notify("New Fragment!") playSound() pickup(obj, "Fragment") end
        elseif isGlasses(obj) then
            knownGlasses[obj] = true
            if Settings.Enabled and not Settings.OnlyFragments then Notify("New Glasses!") playSound() pickup(obj, "Glasses") end
        elseif Settings.AutoShelterKey and isShelterKey(obj) then
            if Settings.Enabled then pickup(obj, "Key") end
        end
    end)
end)

workspace.DescendantRemoving:Connect(function(obj)
    knownFragments[obj] = nil
    knownGlasses[obj] = nil
end)

task.spawn(function()
    task.wait(1)
    for _, obj in ipairs(workspace:GetDescendants()) do
        if isFragment(obj) then knownFragments[obj] = true end
        if isGlasses(obj) then knownGlasses[obj] = true end
    end
end)

task.spawn(function()
    while ScreenGui and ScreenGui.Parent do
        local now = tick()
        local frags = countFrags()
        checkText()

        StatusLabel.Text = "Status: " .. (Settings.Enabled and "Running" or "Off")
        StatusLabel.TextColor3 = Settings.Enabled and Color3.fromRGB(80, 255, 100) or Color3.fromRGB(255, 90, 90)
        FragLabel.Text = "Map Fragments: " .. frags
        CollectLabel.Text = "My Collected: " .. collectedFragments .. " / 5"
        TartarusLabel.Text = collectedFragments >= 5 and "Tartarus: UNLOCKED" or "Tartarus: Locked"
        ModeLabel.Text = (isNext and "Next: " or "Mode: ") .. currentMode
        OverloadLabel.Text = "System Overload: " .. (systemOverload and "YES" or "No")
        GlassesLabel.Text = "Glasses: " .. (glassesFound and "FOUND" or "Not Found")

        if Settings.Enabled and (now - lastScan) >= SCAN_INTERVAL then
            lastScan = now
            cleanCache(knownFragments)
            cleanCache(knownGlasses)
            for obj in pairs(knownFragments) do
                if obj and obj.Parent then pickup(obj, "Fragment") task.wait(0.1) end
            end
            if not Settings.OnlyFragments then
                for obj in pairs(knownGlasses) do
                    if obj and obj.Parent then pickup(obj, "Glasses") task.wait(0.1) end
                end
            end
        end

        if (Settings.ESPFragments or Settings.ESPGlasses) and (now - lastESPUpdate) >= ESP_INTERVAL then
            lastESPUpdate = now
            updateESP()
        elseif not Settings.ESPFragments and not Settings.ESPGlasses then
            clearESP()
        end

        if Settings.AntiAFK and (now - lastAFK) > 50 then
            lastAFK = now
            local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if hum then hum.Jump = true end
        end
        task.wait(0.3)
    end
end)

    clearESP()
    ScreenGui:Destroy()
end)

-- Drag
local dragging, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)
TitleBar.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local d = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.F then toggleFly() end
end)
