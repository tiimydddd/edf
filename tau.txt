local function snd(id, v)
    pcall(function()
        local s = Instance.new("Sound")
        s.SoundId = id
        s.Volume = v or 0.5
        s.Parent = SS
        s:Play()
        s.Ended:Connect(function() s:Destroy() end)
        task.delay(3, function() if s and s.Parent then s:Destroy() end end)
    end)
end

local function inL(name, list)
    local n = (name or ""):lower()
    for _, v in ipairs(list) do
        if n == v:lower() or n:find(v:lower(), 1, true) then return true end
    end
    return false
end

local function isPlayerRelated(p)
    if not p then return true end
    for _, plr in ipairs(Players:GetPlayers()) do
        local ch = plr.Character
        if ch and (p == ch or p:IsDescendantOf(ch)) then return true end
    end
    local mod = p:FindFirstAncestorOfClass("Model")
    while mod do
        if Players:FindFirstChild(mod.Name) then return true end
        if mod:FindFirstChildOfClass("Humanoid") then return true end
        local n = mod.Name:lower()
        if n:find("character") or n:find("dummy") or n:find("npc") or n:find("rig") then return true end
        mod = mod.Parent and mod.Parent:FindFirstAncestorOfClass("Model")
    end
    if p:IsA("BasePart") then
        if p:FindFirstAncestorOfClass("Tool") then return true end
        if p:FindFirstAncestorOfClass("Accessory") then return true end
        if p.Name == "HumanoidRootPart" or p.Name == "Torso" or p.Name == "UpperTorso"
            or p.Name == "Head" or p.Name:find("Arm") or p.Name:find("Leg")
            or p.Name:find("Hand") or p.Name:find("Foot") then
            return true
        end
    end
    return false
end

local function nearAnyPlayer(pos, minDist)
    minDist = minDist or 14
    for _, plr in ipairs(Players:GetPlayers()) do
        local ch = plr.Character
        local h = ch and ch:FindFirstChild("HumanoidRootPart")
        if h and (h.Position - pos).Magnitude < minDist then
            return true
        end
    end
    return false
end

local function isDebrisPart(p)
    if not p or not p:IsA("BasePart") then return false end
    if p.Anchored then return false end
    if p.AssemblyMass < 0.5 then return false end
    if p.AssemblyMass > 800 then return false end -- 太重易把人撞飞
    if isPlayerRelated(p) then return false end
    if isF(p) or (p.Parent and isF(p.Parent)) then return false end
    if isG(p) or (p.Parent and isG(p.Parent)) then return false end
    -- 离任何人太近的散件不处理（避免碰撞把人弹飞）
    if nearAnyPlayer(p.Position, 12) then return false end
    return true
end

local function safeLaunchPart(p, velocity)
    if not isDebrisPart(p) then return end
    pcall(function()
        p.Anchored = false
        p.CanCollide = false -- 发射后不撞人
        p.AssemblyLinearVelocity = velocity
        task.delay(1.2, function()
            if p and p.Parent then
                pcall(function() p.CanCollide = true end)
            end
        end)
    end)
end

    if not o or not o:IsA("BasePart") then return false end
    if isPlayerRelated(o) then return false end
    pcall(function()

        o.CanCollide = false
        o.CanQuery = false
        o.AssemblyLinearVelocity = Vector3.new(
            (math.random() - 0.5) * 60,
            30 + math.random() * 40,
            (math.random() - 0.5) * 60
        )
        o.AssemblyAngularVelocity = Vector3.new(
            (math.random() - 0.5) * 25,
            (math.random() - 0.5) * 25,
            (math.random() - 0.5) * 25
        )
        -- 常见碎裂表现
        if o.Transparency < 0.5 then
            -- keep visible while flying
        end
        for _, d in ipairs(o:GetDescendants()) do
            if d:IsA("Weld") or d:IsA("WeldConstraint") or d:IsA("Motor6D") then
                pcall(function() d:Destroy() end)
            end
        end
    end)

end

local function breakWindows()
    if not C.BreakWin then return end
    if not hasBat() then return end
    local hrp = HRP()
    local origin = hrp and hrp.Position or Vector3.zero
    local n = 0
    local MAX = C.WinAll and 200 or 60
    local RANGE = C.WinAll and 1e9 or 90  -- 附近 90 / 全图
    for _, o in ipairs(workspace:GetDescendants()) do
        if n >= MAX then break end
        if o:IsA("BasePart") and isWindowObj(o) then
            local dist = (o.Position - origin).Magnitude
            if dist <= RANGE then
                if smashPart(o) then n = n + 1 end
            end
        elseif o:IsA("Model") and inL(o.Name, NAMES.Window) then
            for _, p in ipairs(o:GetDescendants()) do
                if n >= MAX then break end
                if p:IsA("BasePart") then
                    local dist = (p.Position - origin).Magnitude
                    if dist <= RANGE and smashPart(p) then n = n + 1 end
                end
            end
        end
    end
end

local function scan()
    local c = {Fragment = 0, Glasses = 0, Key = 0, Ext = 0, Bat = 0}
    local seen = {}
    for _, o in ipairs(workspace:GetDescendants()) do
        if o:IsA("Model") or o:IsA("Tool") or o:IsA("BasePart") or o:IsA("Folder") then
            local id = tostring(o)
            if not seen[id] then
                if isF(o) then
                    c.Fragment = c.Fragment + 1
                    S.kF[o] = true
                    seen[id] = true
                elseif isG(o) then
                    c.Glasses = c.Glasses + 1
                    S.kG[o] = true
                    seen[id] = true
                elseif inL(o.Name, NAMES.Key) and not (o.Parent and inL(o.Parent.Name, NAMES.Key)) then
                    c.Key = c.Key + 1
                    S.kE[o] = "Key"
                    seen[id] = true
                elseif inL(o.Name, NAMES.Ext) and not (o.Parent and inL(o.Parent.Name, NAMES.Ext)) then
                    c.Ext = c.Ext + 1
                    S.kE[o] = "Ext"
                    seen[id] = true
                elseif inL(o.Name, NAMES.Bat) and not (o.Parent and inL(o.Parent.Name, NAMES.Bat)) then
                    c.Bat = c.Bat + 1
                    S.kE[o] = "Bat"
                    seen[id] = true
                end
            end
        end
    end
    S.mapC = c
    return c
end

local function pickup(o, typ)
    if not o or not o.Parent then return end
    local key = tostring(o)
    if S.tried[key] and tick() - S.tried[key] < 1.8 then return end
    S.tried[key] = tick()
    local h = HRP()
    if not h then return end
    local part = best(o)
    if not part then return end
    for _, off in ipairs({Vector3.new(0, 2, 0), Vector3.new(0, 1.6, 1.5), Vector3.new(1.5, 1.6, 0)}) do
        if not o.Parent then break end
        h.CFrame = part.CFrame + off
        task.wait(0.02)
        fireP(o:FindFirstChildWhichIsA("ProximityPrompt", true))
        local ti = part:FindFirstChildOfClass("TouchInterest")
        if ti then
            pcall(function()
                firetouchinterest(h, part, 0)
                task.wait(0.015)
                firetouchinterest(h, part, 1)
            end)
        end
        local cd = o:FindFirstChildWhichIsA("ClickDetector", true)
        if cd then pcall(fireclickdetector, cd) end
        task.wait(0.02)
    end
    task.delay(0.5, function()
        if not o or not o.Parent then
            if typ == "Fragment" then
                S.myF = math.min(5, S.myF + 1)
                S.sF = S.sF + 1
                if S.myF >= 5 then
                    alrt()
                    showBig("Fragments 5/5", Color3.fromRGB(100, 255, 120), 6)
                    N("Fragments complete (5/5)", true)
                else
                    N("Fragment " .. S.myF .. "/5")
                end
            elseif typ == "Glasses" then
                S.gotG = true
                S.sG = S.sG + 1
                N("Item found")
            end
        end
    end)
end

local function doBlackHoleContinuous()
    local hrp = HRP()
    if not hrp then return end
    local center = hrp.Position + Vector3.new(0, 2, 0)
    local parts
    pcall(function() parts = workspace:GetPartBoundsInRadius(center, S.BH_RANGE) end)
    if not parts then return end
    local n = 0
    for _, p in ipairs(parts) do
        if n >= S.BH_MAX then break end
        if isDebrisPart(p) then
            pcall(function()
                if nearAnyPlayer(p.Position, 14) then return end
                p.Anchored = false
                p.CanCollide = false
                local offset = p.Position - center
                local dist = math.max(offset.Magnitude, 2)
                if dist < 16 then return end -- 太近不吸，避免把自己卷起来
                local inward = -offset.Unit
                local tangent = Vector3.new(-offset.Z, 0, offset.X)
                if tangent.Magnitude > 0.1 then tangent = tangent.Unit else tangent = Vector3.new(1, 0, 0) end
                local pull = math.clamp(70 / dist, 8, 40)
                local spin = math.clamp(30 / math.sqrt(dist), 5, 28)
                local up = 10
                p.AssemblyLinearVelocity = inward * pull + tangent * spin + Vector3.new(0, up, 0)
                p.AssemblyAngularVelocity = Vector3.new((math.random() - 0.5) * 8, (math.random() - 0.5) * 10, (math.random() - 0.5) * 8)
            end)
            n = n + 1
        end
    end
end

local function doFly()
    local hrp = HRP()
    local origin = hrp and hrp.Position or Vector3.zero
    local list = {}
    local parts
    pcall(function() parts = workspace:GetPartBoundsInRadius(origin, 120) end)
    if parts then
        for _, p in ipairs(parts) do
            if isDebrisPart(p) then table.insert(list, p) end
        end
    else
        for _, p in ipairs(workspace:GetDescendants()) do
            if p:IsA("BasePart") and isDebrisPart(p) then table.insert(list, p) end
        end
    end
    local total = #list
    if total == 0 then return end
    if S.debrisCursor > total then S.debrisCursor = 1 end
    local n, i = 0, S.debrisCursor
    while n < S.FLY_MAX and n < total do
        local p = list[i]
        if p and p.Parent then
            pcall(function()
                if nearAnyPlayer(p.Position, 12) then return end
                p.Anchored = false
                p.CanCollide = true
                local up = 160 + math.random() * 90
                local sx = (math.random() - 0.5) * 28
                local sz = (math.random() - 0.5) * 28
                p.AssemblyLinearVelocity = Vector3.new(sx, up, sz)
                p.AssemblyAngularVelocity = Vector3.new((math.random()-0.5)*20, (math.random()-0.5)*20, (math.random()-0.5)*20)
                local old = p:FindFirstChild("DebrisFly")
                if old then old:Destroy() end
                local bv = Instance.new("BodyVelocity")
                bv.Name = "DebrisFly"
                bv.MaxForce = Vector3.new(5e5, 5e5, 5e5)
                bv.Velocity = Vector3.new(sx * 0.4, up, sz * 0.4)
                bv.Parent = p
                task.delay(0.35 + math.random() * 0.25, function()
                    if not p or not p.Parent then return end
                    pcall(function()
                        local oldBv = p:FindFirstChild("DebrisFly")
                        if oldBv then oldBv:Destroy() end
                        local dir = Vector3.new(math.random()-0.5, math.random()-0.2, math.random()-0.5)
                        if dir.Magnitude < 0.1 then dir = Vector3.new(1, 0.5, 0) end
                        dir = dir.Unit
                        local power = 70 + math.random() * 90
                        p.AssemblyLinearVelocity = dir * power + Vector3.new(0, 20 + math.random() * 40, 0)
                        p.AssemblyAngularVelocity = Vector3.new((math.random()-0.5)*35, (math.random()-0.5)*35, (math.random()-0.5)*35)
                        local boom = Instance.new("BodyVelocity")
                        boom.Name = "DebrisBoom"
                        boom.MaxForce = Vector3.new(3e5, 3e5, 3e5)
                        boom.Velocity = dir * (power * 0.7)
                        boom.Parent = p
                        task.delay(0.4, function() if boom and boom.Parent then boom:Destroy() end end)
                    end)
                end)
                task.delay(0.5, function() if bv and bv.Parent then bv:Destroy() end end)
            end)
            n = n + 1
        end
        i = i + 1
        if i > total then i = 1 end
        if i == S.debrisCursor and n > 0 then break end
    end
    S.debrisCursor = i
end

    local n = 0
    for _, p in ipairs(parts) do
        if n >= S.CANNON_MAX then break end
        if isDebrisPart(p) then
            pcall(function()
                if nearAnyPlayer(p.Position, 12) then return end
                p.Anchored = false
                p.CanCollide = false
                local spread = Vector3.new((math.random()-0.5)*0.25, (math.random()-0.5)*0.15, (math.random()-0.5)*0.25)
                local dir = look + spread
                if dir.Magnitude < 0.1 then dir = look end
                dir = dir.Unit
                local spd = S.CANNON_POWER + math.random() * 40
                p.AssemblyLinearVelocity = dir * spd + Vector3.new(0, 25, 0)
                p.AssemblyAngularVelocity = Vector3.new((math.random()-0.5)*30, (math.random()-0.5)*30, (math.random()-0.5)*30)
                local old = p:FindFirstChild("DebrisCannon")
                if old then old:Destroy() end
                local bv = Instance.new("BodyVelocity")
                bv.Name = "DebrisCannon"
                bv.MaxForce = Vector3.new(4e5, 4e5, 4e5)
                bv.Velocity = dir * (spd * 0.85)
                bv.Parent = p
                task.delay(0.35, function() if bv and bv.Parent then bv:Destroy() end end)
            end)
            n = n + 1
        end
    end
end

local function refreshBoatCache()
    S.boatCache = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("VehicleSeat") or obj:IsA("Seat") or obj:IsA("Model") then
            local model = obj:IsA("Model") and obj or obj:FindFirstAncestorOfClass("Model")
            if model then
                local name = model.Name:lower()
                if name:find("boat") or name:find("ship") or name:find("raft")
                or name:find("canoe") or name:find("vessel") or name:find("yacht")
                or name:find("帆船") or name:find("游艇") then
                    S.boatCache[model] = true
                end
            end
        end
    end
end

local function doBoatFly()
    local hrp = HRP()
    if not hrp then return end
    if tick() - S.tBoatScan >= 5 then
        S.tBoatScan = tick()
        refreshBoatCache()
    end
    for model in pairs(S.boatCache) do
        if model and model.Parent then
            local primary = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart", true)
            if primary and (primary.Position - hrp.Position).Magnitude < 150 then
                for _, p in ipairs(model:GetDescendants()) do
                    if p:IsA("BasePart") then
                        pcall(function()
                            p.Anchored = false
                            local up = 90 + math.random() * 50
                            p.AssemblyLinearVelocity = Vector3.new(p.AssemblyLinearVelocity.X * 0.25, up, p.AssemblyLinearVelocity.Z * 0.25)
                            local old = p:FindFirstChild("BoatLaunch")
                            if old then old:Destroy() end
                            local bv = Instance.new("BodyVelocity")
                            bv.Name = "BoatLaunch"
                            bv.MaxForce = Vector3.new(2e5, 5e5, 2e5)
                            bv.Velocity = Vector3.new(0, up, 0)
                            bv.Parent = p
                            task.delay(0.5, function() if bv and bv.Parent then bv:Destroy() end end)
                        end)
                    end
                end
            end
        else
            S.boatCache[model] = nil
        end
    end
end


local function saveMusicHistory()
    pcall(function()
        if not writefile then return end
        local lines = {}
        table.insert(lines, "LOCK|" .. tostring(lockedMusicId or ""))
        for _, item in ipairs(S.musicHistory) do
            local name = tostring(item.name or "Track"):gsub("|", "/")
            local id = tostring(item.id or "")
            if id ~= "" then table.insert(lines, name .. "|" .. id) end
        end
        writefile(MUSIC_SAVE, table.concat(lines, "\n"))
    end)
end

local function loadMusicHistory()
    pcall(function()
        if not (isfile and readfile) then return end
        if not isfile(MUSIC_SAVE) then return end
        local raw = readfile(MUSIC_SAVE)
        if not raw or raw == "" then return end
        S.musicHistory = {}
        lockedMusicId = nil
        local first = true
        for line in string.gmatch(raw, "[^\r\n]+") do
            if first and line:sub(1, 5) == "LOCK|" then
                local lockId = line:sub(6)
                if lockId ~= "" and lockId ~= "nil" then lockedMusicId = lockId end
                first = false
            else
                first = false
                local name, id = line:match("^(.-)|(.+)$")
                if id and id ~= "" and name ~= "LOCK" then
                    table.insert(S.musicHistory, {name = (name and name ~= "" and name) or "Track", id = id})
                end
            end
        end
        while #S.musicHistory > S.MAX_HISTORY do table.remove(S.musicHistory) end
    end)
end

local function collectMusicSounds()
    -- 只处理游戏自带的这两个，不新建、不扫全库
    local targets = {}
    local musicFolder = SS:FindFirstChild("Music")
    if not musicFolder then return targets end
    local gamemode = musicFolder:FindFirstChild("Gamemode")
    local bonus = musicFolder:FindFirstChild("BonusDisaster")
    if gamemode and gamemode:IsA("Sound") then table.insert(targets, gamemode) end
    if bonus and bonus:IsA("Sound") then table.insert(targets, bonus) end
    return targets
end

local function applyLockedMusic()
    if not lockedMusicId then return end
    pcall(function()
        for _, s in ipairs(collectMusicSounds()) do
            if not S.originalIds[s] then S.originalIds[s] = s.SoundId end
            if s.SoundId ~= lockedMusicId then
                s.SoundId = lockedMusicId
            end
            -- 不 Play：只改 ID，让游戏自己播
        end
    end)
end

local function changeGameMusic(id)
    local success = false
    pcall(function()
        id = tostring(id or "")
        if id == "" or id == "0" then return end
        if not string.find(id, "rbxassetid://") then id = "rbxassetid://" .. id end
        lockedMusicId = id
        local targets = collectMusicSounds()
        if #targets == 0 then
            N("Gamemode/BonusDisaster not found")
            return
        end
        for _, s in ipairs(targets) do
            if not S.originalIds[s] then S.originalIds[s] = s.SoundId end
            s.SoundId = id
        end
        N("Music locked (Gamemode/Bonus)")
        success = true
        saveMusicHistory()
    end)
    return success
end

local function restoreGameMusic()
    pcall(function()
        lockedMusicId = nil
        for s, oldId in pairs(S.originalIds) do
            if s and s.Parent then s.SoundId = oldId end
        end
        N("Music restored (unlocked)")
    end)
end

local function addToHistory(name, id)
    name = tostring(name or ""):gsub("^%s*(.-)%s*$", "%1")
    id = tostring(id or "")
    if name == "" then name = "Track" end
    if id == "" then return end
    for i, item in ipairs(S.musicHistory) do
        if item.id == id or item.id == "rbxassetid://" .. id then
            table.remove(S.musicHistory, i)
            break
        end
    end
    table.insert(S.musicHistory, 1, {name = name, id = id})
    while #S.musicHistory > S.MAX_HISTORY do table.remove(S.musicHistory) end
    refreshMusicList()
    saveMusicHistory()
end

local function enableNoInjury()
    if S.injuryConn then S.injuryConn:Disconnect() end
    S.injuryConn = RS.Heartbeat:Connect(function()
        if not C.NoInjury then return end
        local char = LP.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hum or not hrp then return end
        pcall(function()
            local state = hum:GetState()
            if state == Enum.HumanoidStateType.FallingDown
            or state == Enum.HumanoidStateType.Ragdoll
            or state == Enum.HumanoidStateType.Physics
            or state == Enum.HumanoidStateType.Freefall then
                if hrp.AssemblyLinearVelocity.Y < -100 then
                    hrp.AssemblyLinearVelocity = Vector3.new(
                        hrp.AssemblyLinearVelocity.X * 0.3,
                        math.max(hrp.AssemblyLinearVelocity.Y, -80),
                        hrp.AssemblyLinearVelocity.Z * 0.3
                    )
                end
                hum:ChangeState(Enum.HumanoidStateType.GettingUp)
                hum:ChangeState(Enum.HumanoidStateType.Running)
            end
            if tick() - S.tInjuryGui >= 0.4 then
                S.tInjuryGui = tick()
                local bones = char:FindFirstChild("Bones")
                if bones then
                    for _, bone in ipairs(bones:GetChildren()) do
                        if bone:IsA("BoolValue") and bone.Value == true then bone.Value = false end
                    end
                end
                local gui = LP:FindFirstChild("PlayerGui")
                if gui then
                    local menus = gui:FindFirstChild("Menus")
                    if menus then
                        local broken = menus:FindFirstChild("BrokenBones")
                        if broken and broken:FindFirstChild("Body") then
                            for _, part in ipairs(broken.Body:GetChildren()) do
                                if part:FindFirstChild("Broken") then
                                    part.Broken.Value = false
                                    part.BackgroundColor3 = Color3.fromRGB(85, 255, 127)
                                end
                            end
                        end
                    end
                end
            end
            if hum.WalkSpeed < 16 and not C.SpdOn then hum.WalkSpeed = 16 end
            hum.PlatformStand = false
            hum.Sit = false
        end)
    end)
end

                    S.tartarusActive = true
                    S.tart = true
                    alrt()
                    showBig("Tartarus", Color3.fromRGB(255, 100, 80), 6)

                S.currentMode = "Intermission"
            elseif reason == "DoomsDayForecast" then
                S.currentMode = "Doomsday"
                N("Doomsday forecast", true)
            elseif reason == "ChadSummon" then
                N("Chad summoned", true)
            elseif reason == "TwistyTalk" and data.A then
                N(tostring(data.A))
            end
        end)
    end
end

local function isLobbyText(t)
    if not t or #t < 3 then return true end
    local lo = string.lower(t)
    return lo:find("intermission") ~= nil
        or lo:find("voting") ~= nil
        or lo:find("vote") ~= nil
        or lo:find("waiting for") ~= nil
        or lo:find("waiting players") ~= nil
end

local function markEnter()
    if S.entered then return end
    S.entered = true
    S.enterT = tick()
    S.firstChk = false
    S.rejSt = "Map loaded, waiting for loot spawn..."
end

local function checkEnter()
    local ot = getObj()
    if not ot then
        S.rejSt = "Waiting for vote / map..."
        return
    end
    local lo = string.lower(ot)
    -- Still in intermission or map vote: do NOT start no-frag logic
    if isLobbyText(ot) then
        if S.entered then
            -- Round ended / back to lobby
            S.entered = false
            S.firstChk = false
            S.alerting = false
        end
        S.rejSt = "Waiting for vote / map..."
        return
    end
    -- In-round objective text → map is up
    markEnter()
    if lo:find("tartarus") then
        if not S.tart then
            S.tart = true
            S.tartarusActive = true
            alrt()
            showBig("Tartarus", Color3.fromRGB(255, 120, 80), 5)
            N("Tartarus active", true)
        end
        S.currentMode = "Tartarus"
    elseif lo:find("doomsday") or lo:find("doom") then
        S.currentMode = "Doomsday"
    elseif lo:find("chad") then
        S.currentMode = "Chad"
    elseif lo:find("twisty") then
        S.currentMode = "Twisty"
    elseif lo:find("weeping") then
        S.currentMode = "Weeping Angel"
    end
end

local function checkFirst()
    if not C.Rej then
        S.rejSt = "Off"
        return
    end
    if S.firstChk or S.alerting then return end
    if not S.entered then
        S.rejSt = "Waiting for vote / map..."
        return
    end
    local wait = S.LOOT_SPAWN_WAIT or 8
    local left = math.ceil(wait - (tick() - S.enterT))
    if left > 0 then
        S.rejSt = "Map ready, loot check in " .. left .. "s"
        return
    end
    S.firstChk = true
    scan()
    local fr = S.mapC.Fragment or cntF()
    if fr <= 0 then
        S.rejSt = "No fragments → hint"
        task.spawn(alertNo)
    else
        S.rejSt = "Fragments: " .. fr
        N("Fragments on map: " .. fr)
    end
end

local function forceHP()

        if h:GetState() == Enum.HumanoidStateType.Dead then
            h:ChangeState(Enum.HumanoidStateType.Running)
            h.Health = h.MaxHealth
        end
    end)
end

            UI.ModeL.Text = "Mode: " .. S.currentMode
            UI.TartL.Text = (S.tart or S.tartarusActive) and "Tartarus: active" or "Tartarus: locked"
            UI.RejL.Text = "Hint: " .. S.rejSt
            UI.SessL.Text = string.format("Session: F%d G%d Hop%d", S.sF, S.sG, S.sH)
            UI.MapL.Text = string.format("Map: Key%d Ext%d Bat%d", c.Key or 0, c.Ext or 0, c.Bat or 0)

            if now - S.tHeavy >= 5 then
                S.tHeavy = now
                scan()
                updDist()
            end
            if now - S.tEsp >= 1.5 then
                S.tEsp = now
                refESP()
            end
            if (C.En or C.FA) and now - S.tScan >= 0.55 then
                S.tScan = now
                clean(S.kF)
                clean(S.kG)
                clean(S.kE)
                local h = HRP()
                local pos = h and h.Position or Vector3.zero
                local fl = {}
                for o in pairs(S.kF) do
                    if o and o.Parent then
                        local p = best(o)
                        if p then table.insert(fl, {o = o, d = (p.Position - pos).Magnitude}) end
                    end
                end
                table.sort(fl, function(a, b) return a.d < b.d end)
                local picked = 0
                for _, d in ipairs(fl) do
                    if picked >= S.MAX_PICK then break end
                    pickup(d.o, "Fragment")
                    picked = picked + 1
                    task.wait(0.08)
                end
                if not C.OnlyF then
                    for o in pairs(S.kG) do
                        if picked >= S.MAX_PICK + 1 then break end
                        if o and o.Parent then
                            pickup(o, "Glasses")
                            picked = picked + 1
                            task.wait(0.08)
                        end
                    end
                end
                for o, k in pairs(S.kE) do
                    if picked >= S.MAX_PICK + 2 then break end
                    if o and o.Parent then
                        local w = (k == "Key" and C.Key) or (k == "Ext" and C.Ext) or (k == "Bat" and C.Bat)
                        if w then
                            pickup(o, k)
                            picked = picked + 1
                            task.wait(0.08)
                        end
                    end
                end
            end
            if C.FA and C.FAShel and now - S.tShel >= 28 then
                S.tShel = now
                toShel(true)
            end
            if C.BlackHole and now - S.tBH >= S.BH_TICK then
                S.tBH = now
                doBlackHoleContinuous()
            end
            if C.Cannon and now - S.tCannon >= S.CANNON_TICK then
                S.tCannon = now
                doCannon()
            end
            if C.BreakWin and now - S.tWin >= 1.2 then
                S.tWin = now
                breakWindows()
            end
            if C.Fly and now - S.tFly >= S.FLY_INTERVAL then
                S.tFly = now
                doFly()
            end
            if C.BoatFly and now - S.tBoat >= 1.2 then
                S.tBoat = now
                doBoatFly()
            end
            checkFirst()
        end)
        task.wait(0.28)
    end
end)
