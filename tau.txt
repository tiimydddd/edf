local function inL(name, list)
    local n = string.lower(tostring(name or ""))
    for _, v in ipairs(list) do
        if n == string.lower(v) or string.find(n, string.lower(v), 1, true) then
            return true
        end
    end
    return false
end

local function isF(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, FragN) then return false end
    if o.Parent and inL(o.Parent.Name, FragN) then return false end
    return true
end

local function isG(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, GlassN) then return false end
    if o.Parent and inL(o.Parent.Name, GlassN) then return false end
    return true
end

local function HRP()
    local c = LP.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function best(o)
    if o:IsA("BasePart") then return o end
    local p = o:FindFirstChildWhichIsA("ProximityPrompt", true)
    if p and p.Parent and p.Parent:IsA("BasePart") then return p.Parent end
    return o:FindFirstChildWhichIsA("BasePart", true)
end

local espFolder = Instance.new("Folder")
espFolder.Name = "tau_esp"
espFolder.Parent = Gui

local function clrESP()
    for _, c in ipairs(espFolder:GetChildren()) do
        c:Destroy()
    end
end

local function addESP(o, col, lab)
    local p = best(o)
    if not p then return end
    pcall(function()
        local hl = Instance.new("Highlight")
        hl.Adornee = o:IsA("Model") and o or p
        hl.FillColor = col
        hl.OutlineColor = col
        hl.FillTransparency = 0.55
        hl.OutlineTransparency = 0.2
        hl.Parent = espFolder
    end)
end

local function scan()
    for o in pairs(kF) do kF[o] = nil end
    for o in pairs(kG) do kG[o] = nil end
    local fc, gc = 0, 0
    for _, o in ipairs(workspace:GetDescendants()) do
        if isF(o) then
            kF[o] = true
            fc = fc + 1
        elseif isG(o) then
            kG[o] = true
            gc = gc + 1
        end
    end
    return fc, gc
end

local function pickup(o, typ)
    if not o or not o.Parent then return end
    local key = tostring(o)
    if tried[key] and tick() - tried[key] < 0.6 then return end
    tried[key] = tick()
    local h = HRP()
    if not h then return end
    local part = best(o)
    if not part then return end

    local offs = {
        Vector3.new(0, 2, 0),
        Vector3.new(0, 1.5, 1.5),
        Vector3.new(0, 1.5, -1.5),
        Vector3.new(1.5, 1.5, 0),
        Vector3.new(-1.5, 1.5, 0),
    }
    for _, off in ipairs(offs) do
        if not o.Parent then break end
        pcall(function() h.CFrame = part.CFrame + off end)
        task.wait(0.03)
        local pr = o:FindFirstChildWhichIsA("ProximityPrompt", true)
        if pr and typeof(fireproximityprompt) == "function" then
            pcall(fireproximityprompt, pr)
            pcall(fireproximityprompt, pr, 1)
        end
        if typeof(firetouchinterest) == "function" then
            pcall(function()
                firetouchinterest(h, part, 0)
                task.wait(0.02)
                firetouchinterest(h, part, 1)
            end)
        end
        task.wait(0.02)
    end
    task.delay(0.5, function()
        if not o or not o.Parent then
            if typ == "Fragment" then
                myF = math.min(5, myF + 1)
                N("碎片 " .. myF .. "/5")
            elseif typ == "Glasses" then
                gotG = true
                N("眼镜已找到")
            end
        end
    end)
end

local function toShel()
    local h = HRP()
    if not h then
        N("无角色")
        return
    end
    for _, o in ipairs(workspace:GetDescendants()) do
        if inL(o.Name, ShelN) then
            local p = best(o)
            if p then
                h.CFrame = p.CFrame + Vector3.new(0, 5, 0)
                N("已到避难所")
                return
            end
        end
    end
    N("没找到避难所")
end

end

end

workspace.DescendantAdded:Connect(function(o)
    task.defer(function()
        if isF(o) then
            kF[o] = true
            if C.Auto then
                task.wait(0.1)
                pickup(o, "Fragment")
            end
        elseif isG(o) then
            kG[o] = true
            if C.Auto and not C.OnlyFrag then
                task.wait(0.1)
                pickup(o, "Glasses")
            end
        end
    end)
end)

workspace.DescendantRemoving:Connect(function(o)
    kF[o] = nil
    kG[o] = nil
end)

-- 主循环
local tScan, tShel, tEsp, tAfk = 0, 0, 0, 0
task.spawn(function()
    while Gui and Gui.Parent do
        local ok, err = pcall(function()
            local now = tick()
            local fc, gc = 0, 0
            if now - tScan >= 0.5 then
                tScan = now
                fc, gc = scan()
                FragL.Text = string.format("碎片: 地图%d | 我%d/5", fc, myF)
                if gotG or gc > 0 then
                    GlassL.Text = "眼镜: 有"
                    GlassL.TextColor3 = Color3.fromRGB(100, 255, 120)
                else
                    GlassL.Text = "眼镜: 未找到"
                    GlassL.TextColor3 = Color3.fromRGB(180, 180, 195)
                end

                if C.Auto then
                    local h = HRP()
                    local pos = h and h.Position or Vector3.zero
                    local list = {}
                    for o in pairs(kF) do
                        if o and o.Parent then
                            local p = best(o)
                            if p then
                                table.insert(list, {o = o, d = (p.Position - pos).Magnitude})
                            end
                        end
                    end
                    table.sort(list, function(a, b) return a.d < b.d end)
                    for _, d in ipairs(list) do
                        pickup(d.o, "Fragment")
                        task.wait(0.08)
                    end
                    -- 第二遍防漏
                    for o in pairs(kF) do
                        if o and o.Parent then
                            tried[tostring(o)] = 0
                            pickup(o, "Fragment")
                            task.wait(0.06)
                        end
                    end
                    if not C.OnlyFrag then
                        for o in pairs(kG) do
                            if o and o.Parent then
                                pickup(o, "Glasses")
                                task.wait(0.08)
                            end
                        end
                    end
                end
            end

            if C.ESP and now - tEsp >= 1 then
                tEsp = now
                clrESP()
                for o in pairs(kF) do
                    if o and o.Parent then
                        addESP(o, Color3.fromRGB(255, 70, 70), "F")
                    end
                end
                for o in pairs(kG) do
                    if o and o.Parent then
                        addESP(o, Color3.fromRGB(80, 220, 220), "G")
                    end
                end
            end

            if C.ShelAuto and now - tShel >= 30 then
                tShel = now
                toShel()
            end

            if C.AFK and now - tAfk >= 30 then
                tAfk = now
                afk()
            end
        end)
        if not ok then
            warn("[tau] loop err:", err)
        end
        task.wait(0.15)
    end
end)
