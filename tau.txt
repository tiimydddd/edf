local recentlyTried = {}
local function pickup(obj, typ)
    if not obj or not obj.Parent then return end
    local key = tostring(obj)
    if recentlyTried[key] and tick() - recentlyTried[key] < TRIED_COOLDOWN then return end
    recentlyTried[key] = tick()

    local hrp = getHRP()
    if not hrp then return end

    -- === 修复：增加最大捡取距离保护 ===
    local part = getBestPart(obj)
    if not part then return end

    local maxDist = 65  -- 可调，防止捡不到
    if (hrp.Position - part.Position).Magnitude > maxDist then
        return
    end

    local offsets = { Vector3.new(0, 2.0, 0), Vector3.new(0, 1.6, 1.5), Vector3.new(0, 1.6, -1.5), Vector3.new(1.5, 1.6, 0), Vector3.new(-1.5, 1.6, 0), Vector3.new(0, 2.8, 0) }
    for _, offset in ipairs(offsets) do
        if not obj.Parent then break end
        hrp.CFrame = part.CFrame + offset
        task.wait(0.028)
        tryFirePrompt(obj:FindFirstChildWhichIsA("ProximityPrompt", true))
        local touch = part:FindFirstChildOfClass("TouchInterest")
        if touch then pcall(function() firetouchinterest(hrp, part, 0) task.wait(0.02) firetouchinterest(hrp, part, 1) end) end
        local click = obj:FindFirstChildWhichIsA("ClickDetector", true)
        if click then pcall(fireclickdetector, click) end
        task.wait(0.03)
    end

    task.delay(0.6, function()
        if not obj or not obj.Parent then
            if typ == "Fragment" then
                collectedFragments = collectedFragments + 1
                Notify("碎片 " .. collectedFragments .. "/5")
            elseif typ == "Glasses" then
                glassesFound = true
                Notify("眼镜已找到！Chad条件满足")
            end
        end
    end)
end

local tartarusUnlocked = false

local function checkAlerts()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end

    local sawTartarus = false
    local sawDoomsday = false

    local iconKeywords = {
        ["toastnado"] = "ToastNado", ["toast"] = "ToastNado",
        ["tornado"] = "ToastNado", ["dust devil"] = "Dust Devil",
        ["4-nado"] = "4-Nado", ["double trouble"] = "Double Trouble",
        ["triple threat"] = "Triple Threat", ["infernado"] = "Infernado",
        ["deadnado"] = "Deadnado", ["lurker"] = "Lurker",
        ["doomsday"] = "Doomsday", ["tartarus"] = "Tartarus",
        ["blood moon"] = "Tartarus", ["normal survival"] = "Normal Survival",
        ["ef"] = "Normal Survival", ["survive the end"] = "Doomsday",
    }

    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextLabel") or g:IsA("TextButton") then
            local t = string.lower(g.Text or "")
            for keyword, mode in pairs(iconKeywords) do
                if t:find(keyword, 1, true) then
                    if mode == "Doomsday" then sawDoomsday = true end
                    if mode == "Tartarus" then sawTartarus = true end
                    break
                end
            end
        end
    end

    if sawTartarus and not tartarusUnlocked then
        tartarusUnlocked = true
        Notify("Tartarus 已解锁！", true)
    end

    if sawDoomsday and not doomsdayDetected then
        doomsdayDetected = true
        Notify("检测到 Doomsday！", true)
    end
end

task.spawn(function()
    while ScreenGui and ScreenGui.Parent do
        local now = tick()
        local c = mapLootCounts
        local frags = c.Fragment or countFrags()

        -- Status, Goal, FragProgress, GlassesProgress 等 UI 更新同上（保持不变）

        FragProgressLabel.Text = string.format("血月碎片: 地图%d | 我%d/5 → Tartarus", frags, collectedFragments)
        FragProgressLabel.TextColor3 = (collectedFragments >= 5 or tartarusUnlocked) and Color3.fromRGB(100, 255, 120) or Color3.fromRGB(180, 180, 195)

        if glassesFound or (c.Glasses or 0) > 0 then
            GlassesProgressLabel.Text = "神秘眼镜: 已找到 → Chad可出"
            GlassesProgressLabel.TextColor3 = Color3.fromRGB(100, 255, 120)
        else
            GlassesProgressLabel.Text = "神秘眼镜: 未找到 → 需捡眼镜"
            GlassesProgressLabel.TextColor3 = Color3.fromRGB(255, 160, 100)
        end

        if tartarusUnlocked or collectedFragments >= 5 then
            -- 无需额外 TartarusLabel
        end

        RejoinStatusLabel.Text = "重进提示: " .. rejoinStatus
        MapLootLabel.Text = string.format("地图: 钥匙%d 灭火器%d 球棒%d 其他%d", c.Key or 0, c.Ext or 0, c.Bat or 0, modeLoot)

        -- 其他更新同上

        if now - lastHeavyScan >= HEAVY_SCAN_INTERVAL then
            lastHeavyScan = now
            checkAlerts()
            scanMapLoot()
        end

        -- 自动捡取逻辑（已修复距离保护）
        if (Settings.Enabled or Settings.FullAuto) and (now - lastScan) >= SCAN_INTERVAL then
            lastScan = now
            cleanCache(knownFragments)
            cleanCache(knownGlasses)

            local fragList, hrp = {}, getHRP()
            local myPos = hrp and hrp.Position or Vector3.zero
            for obj in pairs(knownFragments) do
                if obj and obj.Parent then
                    local part = getBestPart(obj)
                    if part and (myPos - part.Position).Magnitude <= 65 then
                        table.insert(fragList, {obj = obj, dist = (part.Position - myPos).Magnitude})
                    end
                end
            end
            table.sort(fragList, function(a, b) return a.dist < b.dist end)
            for _, data in ipairs(fragList) do
                pickup(data.obj, "Fragment")
                task.wait(0.07)
            end

            if not Settings.OnlyFragments then
                for obj in pairs(knownGlasses) do
                    if obj and obj.Parent then
                        pickup(obj, "Glasses")
                        task.wait(0.07)
                    end
                end
            end
        end

        if Settings.FullAuto and Settings.FullAutoShelter and (now - lastShelterTP) >= SHELTER_TP_INTERVAL then
            lastShelterTP = now
            forceToShelter(true)
        end

        if Settings.TornadoPull then doTornadoPull() end
        if Settings.FragmentFly then doFragmentFly() end

        checkFirstRoundNoFrag()

        task.wait(UI_REFRESH_INTERVAL)
    end
end)
