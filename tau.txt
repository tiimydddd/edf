-- ==================== 工具函数 ====================
local function Notify(text)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = "TAU Auto Loot",
            Text = text,
            Duration = 3.5
        })
    end)
    print("[TAU] " .. text)
end

local function getHRP()
    local char = LocalPlayer.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function isTarget(obj)
    local name = obj.Name:lower()
    for _, n in ipairs(FragmentNames) do
        if name:find(n:lower()) then return "Fragment" end
    end
    if not Settings.OnlyFragments then
        for _, n in ipairs(GlassesNames) do
            if name:find(n:lower()) then return "Glasses" end
        end
    end
    return nil
end

local function isExtinguisher(obj)
    local name = obj.Name:lower()
    for _, n in ipairs(ExtinguisherNames) do
        if name:find(n:lower()) then return true end
    end
    return false
end

local function countServerFragments()
    local count = 0
    for _, obj in ipairs(workspace:GetDescendants()) do
        local name = obj.Name:lower()
        for _, n in ipairs(FragmentNames) do
            if name:find(n:lower()) then
                count = count + 1
                break
            end
        end
    end
    return count
end

local function tryPickup(obj, itemType)
    local hrp = getHRP()
    if not hrp then return end

    local part = obj:IsA("BasePart") and obj or (obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")))
    if not part then return end

    hrp.CFrame = part.CFrame + Vector3.new(0, 2.8, 0)

    local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
    if prompt then pcall(fireproximityprompt, prompt) end

    local touch = part:FindFirstChildOfClass("TouchInterest")
    if touch then
        pcall(firetouchinterest, hrp, part, 0)
        task.wait(0.04)
        pcall(firetouchinterest, hrp, part, 1)
    end

    local click = obj:FindFirstChildOfClass("ClickDetector") or part:FindFirstChildOfClass("ClickDetector")
    if click then pcall(fireclickdetector, click) end

    if itemType == "Fragment" then
        foundAnyFragment = true
        collectedFragments = collectedFragments + 1
        Settings.AutoRejoin = false
        RejoinBtn.Text = "Auto Rejoin: OFF (Found)"
        Notify("捡到碎片！当前已收集: " .. collectedFragments .. "/5")
    elseif itemType == "Glasses" then
        glassesFound = true
        Notify("捡到神秘眼镜！")
    end

    lastFoundTime = tick()
    Notify("Trying: " .. itemType)
end

local function tryPickExtinguisher()
    local found = false
    for _, obj in ipairs(workspace:GetDescendants()) do
        if isExtinguisher(obj) then
            found = true
            tryPickup(obj, "Fire Extinguisher")
            task.wait(0.15)
        end
    end
    if found then
        Notify("找到灭火器，正在尝试拾取")
    else
        Notify("地图上没有灭火器")
    end
end

local function rejoin()
    if foundAnyFragment then return end
    local fragCount = countServerFragments()
    if fragCount >= Settings.MinFragmentsToStay then
        Notify("服务器还有 " .. fragCount .. " 个碎片，不重进")
        return
    end
    Notify("服务器碎片 = 0，准备重进...")
    task.wait(1.2)
    TeleportService:Teleport(PLACE_ID, LocalPlayer)
end

local function makePartsFly()
    local hrp = getHRP()
    if not hrp then return end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and not obj.Anchored and not obj:IsDescendantOf(LocalPlayer.Character) then
            local distance = (obj.Position - hrp.Position).Magnitude
            if distance <= FlyRange then
                local old = obj:FindFirstChild("FlyForce")
                if old then old:Destroy() end
                local bv = Instance.new("BodyVelocity")
                bv.Name = "FlyForce"
                bv.MaxForce = Vector3.new(0, math.huge, 0)
                bv.Velocity = Vector3.new(0, UpForce, 0)
                bv.Parent = obj
                task.delay(2.5, function()
                    if bv and bv.Parent then bv:Destroy() end
                end)
            end
        end
    end
end

local function toggleFly()
    flyEnabled = not flyEnabled
    if flyEnabled then
        FlyBtn.Text = "Fly Parts: ON"
        FlyBtn.BackgroundColor3 = Color3.fromRGB(40, 140, 70)
        Notify("飞方块已开启")
        flyConnection = RunService.Heartbeat:Connect(function()
            makePartsFly()
        end)
    else
        FlyBtn.Text = "Fly Parts: OFF"
        FlyBtn.BackgroundColor3 = Color3.fromRGB(55, 55, 70)
        Notify("飞方块已关闭")
        if flyConnection then
            flyConnection:Disconnect()
            flyConnection = nil
        end
    end
end

local function detectMode()
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return "Detecting...", false, nil end

    local foundMode = "Detecting..."
    local isNext = false
    local doomsdayRisk = nil

    for _, gui in ipairs(playerGui:GetDescendants()) do
        if gui:IsA("TextLabel") or gui:IsA("TextButton") then
            local text = string.lower(gui.Text or "")
            if text:find("this round's gamemode is") or text:find("gamemode is:") or text:find("round's gamemode") then
                isNext = true
            end
            for keyword, modeName in pairs(knownModes) do
                if text:find(keyword) then
                    foundMode = modeName
                    break
                end
            end
            if text:find("low risk") then
                doomsdayRisk = "Low Risk"
            elseif text:find("medium risk") then
                doomsdayRisk = "Medium Risk"
            elseif text:find("high risk") then
                doomsdayRisk = "High Risk"
            elseif text:find("extreme risk") then
                doomsdayRisk = "Extreme Risk"
            end
        end
    end
    return foundMode, isNext, doomsdayRisk
end

local function onNewItemAdded(obj)
    if not Settings.Enabled then return end
    task.wait()
    local itemType = isTarget(obj)
    if itemType then
        Notify("检测到新" .. itemType .. "生成！正在尝试拾取...")
        tryPickup(obj, itemType)
    end
end

workspace.DescendantAdded:Connect(onNewItemAdded)

-- ==================== 主循环 ====================
task.spawn(function()
    while true do
        local fragCount = countServerFragments()

        -- 更新UI
        if FragmentCountLabel then
            FragmentCountLabel.Text = "Server Fragments: " .. fragCount
            FragmentCountLabel.TextColor3 = fragCount > 0 and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(255, 120, 120)
        end
        if CollectedLabel then
            CollectedLabel.Text = "Collected: " .. collectedFragments .. " / 5"
            CollectedLabel.TextColor3 = collectedFragments >= 5 and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(180, 180, 200)
        end
        if GlassesLabel then
            GlassesLabel.Text = glassesFound and "Glasses: FOUND ✓" or "Glasses: Not Found"
            GlassesLabel.TextColor3 = glassesFound and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(180, 180, 200)
        end
        if TartarusLabel then
            if collectedFragments >= 5 then
                TartarusLabel.Text = "Tartarus: UNLOCKED!"
                TartarusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
            else
                TartarusLabel.Text = "Tartarus: Locked (" .. collectedFragments .. "/5)"
                TartarusLabel.TextColor3 = Color3.fromRGB(255, 120, 120)
            end
        end
        if ModeLabel then
            local currentMode, isNext, risk = detectMode()
            if isNext then
                ModeLabel.Text = "Next Mode: " .. currentMode
                ModeLabel.TextColor3 = Color3.fromRGB(100, 200, 255)
            else
                ModeLabel.Text = "Current Mode: " .. currentMode
                if currentMode == "Tartarus" or currentMode == "Doomsday" then
                    ModeLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
                elseif currentMode ~= "Detecting..." then
                    ModeLabel.TextColor3 = Color3.fromRGB(255, 200, 80)
                else
                    ModeLabel.TextColor3 = Color3.fromRGB(180, 180, 200)
                end
            end
            if risk then
                ModeLabel.Text = ModeLabel.Text .. " | Doomsday: " .. risk
            end
        end

        if Settings.Enabled then
            local hrp = getHRP()
            if hrp then
                for _, obj in ipairs(workspace:GetDescendants()) do
                    if not Settings.Enabled then break end
                    local itemType = isTarget(obj)
                    if itemType then
                        tryPickup(obj, itemType)
                        task.wait(0.18)
                    end
                end
            end

            if Settings.AutoRejoin and not foundAnyFragment then
                if fragCount < Settings.MinFragmentsToStay and (tick() - lastFoundTime) > Settings.NoFragmentTime then
                    rejoin()
                end
            end
        end
        task.wait(Settings.ScanSpeed)
    end
end)

-- ==================== Functions ====================
local function Notify(text)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = "TAU Auto Loot",
            Text = text,
            Duration = 3.5
        })
    end)
    print("[TAU] " .. text)
end

local function getHRP()
    local char = LocalPlayer.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function isTarget(obj)
    local name = obj.Name:lower()
    for _, n in ipairs(FragmentNames) do
        if name:find(n:lower()) then return "Fragment" end
    end
    if not Settings.OnlyFragments then
        for _, n in ipairs(GlassesNames) do
            if name:find(n:lower()) then return "Glasses" end
        end
    end
    return nil
end

local function isExtinguisher(obj)
    local name = obj.Name:lower()
    for _, n in ipairs(ExtinguisherNames) do
        if name:find(n:lower()) then return true end
    end
    return false
end

local function tryPickup(obj, itemType)
    local hrp = getHRP()
    if not hrp then return end

    local part = obj:IsA("BasePart") and obj or (obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")))
    if not part then return end

    hrp.CFrame = part.CFrame + Vector3.new(0, 2.8, 0)

    local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
    if prompt then pcall(fireproximityprompt, prompt) end

    local touch = part:FindFirstChildOfClass("TouchInterest")
    if touch then
        pcall(firetouchinterest, hrp, part, 0)
        task.wait(0.04)
        pcall(firetouchinterest, hrp, part, 1)
    end

    local click = obj:FindFirstChildOfClass("ClickDetector") or part:FindFirstChildOfClass("ClickDetector")
    if click then pcall(fireclickdetector, click) end

    if itemType == "Fragment" then
        foundAnyFragment = true
        Settings.AutoRejoin = false
        RejoinBtn.Text = "Auto Rejoin: OFF (Found)"
        Notify("Fragment found! Auto Rejoin disabled.")
    end

    lastFoundTime = tick()
    Notify("Trying to pick up: " .. itemType)
end

local function tryPickExtinguisher()
    local found = false
    for _, obj in ipairs(workspace:GetDescendants()) do
        if isExtinguisher(obj) then
            found = true
            tryPickup(obj, "Fire Extinguisher")
            task.wait(0.15)
        end
    end
    if found then
        Notify("Fire Extinguisher found and trying to pick up")
    else
        Notify("No Fire Extinguisher found on the map")
    end
end

local function rejoin()
    if foundAnyFragment then return end
    Notify("No fragments found, rejoining...")
    task.wait(1.2)
    TeleportService:Teleport(PLACE_ID, LocalPlayer)
end

local function makePartsFly()
    local hrp = getHRP()
    if not hrp then return end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and not obj.Anchored and not obj:IsDescendantOf(LocalPlayer.Character) then
            local distance = (obj.Position - hrp.Position).Magnitude
            if distance <= FlyRange then
                local old = obj:FindFirstChild("FlyForce")
                if old then old:Destroy() end

                local bv = Instance.new("BodyVelocity")
                bv.Name = "FlyForce"
                bv.MaxForce = Vector3.new(0, math.huge, 0)
                bv.Velocity = Vector3.new(0, UpForce, 0)
                bv.Parent = obj

                task.delay(2.5, function()
                    if bv and bv.Parent then bv:Destroy() end
                end)
            end
        end
    end
end

-- Main Loop
task.spawn(function()
    while true do
        if Settings.Enabled then
            local hrp = getHRP()
            if hrp then
                for _, obj in ipairs(workspace:GetDescendants()) do
                    if not Settings.Enabled then break end
                    local itemType = isTarget(obj)
                    if itemType then
                        tryPickup(obj, itemType)
                        task.wait(0.2)
                    end
                end
            end

            if Settings.AutoRejoin and not foundAnyFragment and (tick() - lastFoundTime) > Settings.NoFragmentTime then
                rejoin()
            end
        end
        task.wait(Settings.ScanSpeed)
    end
end)

-- Fly Parts Toggle (F key)
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.F then
        flyEnabled = not flyEnabled
        if flyEnabled then
            FlyLabel.Text = "Press F: Fly Parts ON"
            FlyLabel.TextColor3 = Color3.fromRGB(80, 255, 120)
            Notify("Fly Parts Enabled")
            flyConnection = RunService.Heartbeat:Connect(function()
                makePartsFly()
            end)
        else
            FlyLabel.Text = "Press F: Fly Parts OFF"
            FlyLabel.TextColor3 = Color3.fromRGB(180, 180, 190)
            Notify("Fly Parts Disabled")
            if flyConnection then
                flyConnection:Disconnect()
                flyConnection = nil
            end
        end
    end
end)
