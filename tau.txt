    for obj in pairs(cache) do
        if not obj or not obj.Parent then
            cache[obj] = nil
        end

local function countFrags()
    cleanCache(knownFragments)
    local c = 0
    for _ in pairs(knownFragments) do c = c + 1 end
    return c

local function shouldPickup(typ)
    if Settings.Mode == 1 then
        return typ == "Fragment"
    elseif Settings.Mode == 2 then
        return typ == "Glasses"
    elseif Settings.Mode == 3 then
        return typ == "Fragment" or typ == "Glasses"
    elseif Settings.Mode == 4 then
        return typ == "Fragment" or typ == "Glasses" or typ == "Key"
    elseif Settings.Mode == 5 then
        return typ == "Fragment" or typ == "Glasses" or typ == "Bat"
    end
    return false
end

-- ==================== 强制传送 + 捡取（修复版） ====================
local function pickup(obj, typ)
    local hrp = getHRP()
    if not hrp then return end

    local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
    if not part then return end

    -- 强制传送过去（无论有没有物品都传）
    hrp.CFrame = part.CFrame + Vector3.new(0, 3, 0)

    -- 触发拾取
    local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
    if prompt then pcall(fireproximityprompt, prompt) end

    local touch = part:FindFirstChildOfClass("TouchInterest")
    if touch then
        pcall(firetouchinterest, hrp, part, 0)
        task.wait(0.03)
        pcall(firetouchinterest, hrp, part, 1)
    end

    local click = obj:FindFirstChildOfClass("ClickDetector") or part:FindFirstChildOfClass("ClickDetector")
    if click then pcall(fireclickdetector, click) end

    if typ == "Fragment" then
        collectedFragments = collectedFragments + 1
        Notify("Fragment " .. collectedFragments .. "/5")
    elseif typ == "Glasses" then
        glassesFound = true
        Notify("Glasses found!")
    elseif typ == "Key" then
        keyFound = true
        Notify("Key found!")
    elseif typ == "Bat" then
        batFound = true
        Notify("Bat found!")
    end
end

-- ==================== 龙卷风物理约束 ====================
local activeConstraints = {}

local function applyTornadoConstraint(part, root)
    if activeConstraints[part] then return end
    if #activeConstraints >= 40 then return end

    local att = Instance.new("Attachment")
    att.Parent = part

    local vf = Instance.new("VectorForce")
    vf.Attachment0 = att
    vf.RelativeTo = Enum.ActuatorRelativeTo.World
    vf.Parent = part

    local torque = Instance.new("Torque")
    torque.Attachment0 = att
    torque.RelativeTo = Enum.ActuatorRelativeTo.World
    torque.Parent = part

    activeConstraints[part] = {vf = vf, torque = torque, att = att}
end

local function updateTornadoConstraint(part, root, data)
    local offset = part.Position - root.Position
    local dist = offset.Magnitude
    if dist > 55 or dist < 1.5 then
        for _, v in ipairs(part:GetChildren()) do
            if v:IsA("Attachment") or v:IsA("VectorForce") or v:IsA("Torque") then v:Destroy() end
        end
        activeConstraints[part] = nil
        return
    end

    local horizontal = Vector3.new(offset.X, 0, offset.Z)
    local hDist = horizontal.Magnitude or 1
    local inwardDir = -horizontal.Unit
    local tangent = Vector3.new(-horizontal.Z, 0, horizontal.X).Unit

    local force = inwardDir * 40 + Vector3.new(0, 60 * (1 - dist / 55), 0) + tangent * 35

    data.vf.Force = force * part.AssemblyMass
    data.torque.Torque = Vector3.new(0, 70 * part.AssemblyMass, 0)
end

-- ==================== 模式检测 ====================
local function detectModes()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end
    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextLabel") or g:IsA("TextButton") or g:IsA("TextBox") then
            local t = (g.Text or ""):lower()
            if t == "" then continue end
            for _, pair in ipairs(GamemodeKeywords) do
                if t:find(pair[1]) then
                    if t:find("next") or t:find("upcoming") or t:find("下一") then
                        nextMode = pair[2]
                    else
                        currentMode = pair[2]
                    end
                end
            end
        end
    end
end

-- ==================== 自动投票地图 ====================
local function autoVoteMap()
    if not Settings.AutoVoteMap then return end
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end
    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextButton") or g:IsA("ImageButton") then
            local text = (g.Text or ""):lower()
            for _, mapName in ipairs({"Farm","Town","Neighborhood","Metropolis","Boardwalk","Isle Vista","Death Valley","Fort Blox","Junkyard","Rome"}) do
                if text:find(mapName:lower()) then
                    pcall(function() firesignal(g.MouseButton1Click) end)
                    Notify("已自动投地图: " .. mapName)
                    return
                end
            end
        end
    end
end

-- ==================== 额外灾难提醒 ====================
local function checkExtraDisaster()
    if not Settings.ExtraDisasterAlert then return end
    if tick() - lastExtraNotify < 8 then return end
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end
    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextLabel") or g:IsA("TextButton") then
            local t = (g.Text or ""):lower()
            for _, pair in ipairs(ExtraDisasterKeywords) do
                if t:find(pair[1]) then
                    lastExtraNotify = tick()
                    Notify("⚠ 额外灾难: " .. pair[2])
                    return
                end
            end
        end
    end
end

-- ==================== Chad 骰子提醒 ====================
local function checkChadDice()
    if not Settings.ChadDiceAlert then return end
    if tick() - lastChadDiceNotify < 6 then return end
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end
    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextLabel") or g:IsA("TextButton") or g:IsA("ImageLabel") then
            local t = ((g.Text or "") .. " " .. g.Name):lower()
            if t:find("dice") or t:find("roll") or t:find("chad") and (t:find("queue") or t:find("骰子")) then
                lastChadDiceNotify = tick()
                Notify("🎲 Chad 骰子出现！准备排队稀有模式")
                return
            end
        end
    end
end

                if obj and obj.Parent and countFrags() < 5 then
                    pickup(obj, "Fragment")
                    task.wait(0.07)
                end
            end
            for obj in pairs(knownGlasses) do
                if obj and obj.Parent then
                    pickup(obj, "Glasses")
                    task.wait(0.07)
                end
            end
            for obj in pairs(knownKeys) do
                if obj and obj.Parent then
                    pickup(obj, "Key")
                    task.wait(0.07)
                end
            end
            for obj in pairs(knownBats) do
                if obj and obj.Parent then
                    pickup(obj, "Bat")
                    task.wait(0.07)
                end
            end
        end

        -- 龙卷风物理约束
        if Settings.TornadoEnabled then
            local root = getHRP()
            if root then
                for _, obj in ipairs(workspace:GetDescendants()) do
                    if obj:IsA("BasePart") and not obj.Anchored and obj.Parent and not obj:IsDescendantOf(LocalPlayer.Character) then
                        local dist = (obj.Position - root.Position).Magnitude
                        if dist < 55 and not activeConstraints[obj] then
                            applyTornadoConstraint(obj, root)
                        end
                    end
                end

                for part, data in pairs(activeConstraints) do
                    if part and part.Parent then
                        updateTornadoConstraint(part, root, data)
                    else
                        activeConstraints[part] = nil
                    end
                end
            end
        else
            for part, _ in pairs(activeConstraints) do
                for _, v in ipairs(part:GetChildren()) do
                    if v:IsA("Attachment") or v:IsA("VectorForce") or v:IsA("Torque") then v:Destroy() end
                end
            end
            activeConstraints = {}
        end

        if (Settings.AutoGoShelter or Settings.FullAuto) and (now - lastShelterCheck) >= SHELTER_CHECK_INTERVAL then
            lastShelterCheck = now
        end

        if (now - lastModeCheck) >= MODE_CHECK_INTERVAL then
            lastModeCheck = now
            detectModes()
            checkExtraDisaster()
            checkChadDice()
        end

        if Settings.AutoVoteMap and (now - lastVoteCheck) >= VOTE_CHECK_INTERVAL then
            lastVoteCheck = now
            autoVoteMap()
        end

        checkFirstRoundNoFrag()

        task.wait(0.25)
    end
end)
