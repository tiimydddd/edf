local function inL(name, list)
    local n = string.lower(tostring(name or ""))
    for _, v in ipairs(list) do
        if n == string.lower(v) or string.find(n, string.lower(v), 1, true) then return true end
    end
    return false
end

local function nameLooksFrag(name)
    local n = string.lower(tostring(name or ""))
    if inL(name, FragN) then return true end
    if (string.find(n, "fragment", 1, true) or string.find(n, "碎片", 1, true))
        and (string.find(n, "blood", 1, true) or string.find(n, "moon", 1, true) or string.find(n, "血月", 1, true)) then
        return true
    end
    return false
end
local function isF(o)
    if not o or not o.Parent then return false end
    if not nameLooksFrag(o.Name) then return false end
    -- 父级已经是碎片 → 只算父级，子部件不重复计数
    if o.Parent and o.Parent ~= workspace and nameLooksFrag(o.Parent.Name) then
        return false
    end
    return true
end

local function isG(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, GlassN) then return false end
    if o.Parent and inL(o.Parent.Name, GlassN) then return false end
    return true
end

local lastVoteTry = 0
local function preferMapName(name)
    local n = string.lower(tostring(name or ""))
    for _, m in ipairs(VoteMaps) do
        if string.find(n, string.lower(m), 1, true) then return true end
    end
    return false
end

local function tryClick(gui)
    if not gui then return false end
    local ok = false
    pcall(function()
        if not (gui:IsA("TextButton") or gui:IsA("ImageButton") or gui:IsA("GuiButton")) then
            return
        end
        -- 不要用 :Fire()，部分执行器上是 nil 会报 attempt to call a nil value
        if typeof(firesignal) == "function" then
            pcall(firesignal, gui.MouseButton1Click)
            pcall(firesignal, gui.Activated)
            ok = true
        end
        if typeof(getconnections) == "function" then
            pcall(function()
                for _, c in ipairs(getconnections(gui.MouseButton1Click)) do
                    if c.Function then pcall(c.Function) end
                end
                for _, c in ipairs(getconnections(gui.Activated)) do
                    if c.Function then pcall(c.Function) end
                end
                ok = true
            end)
        end
        -- 退化：虚拟点击位置
        pcall(function()
            local pos = gui.AbsolutePosition + gui.AbsoluteSize / 2
            local vim = game:GetService("VirtualInputManager")
            if vim then
                vim:SendMouseButtonEvent(pos.X, pos.Y, 0, true, game, 0)
                vim:SendMouseButtonEvent(pos.X, pos.Y, 0, false, game, 0)
                ok = true
            end
        end)
    end)
    return ok
end

local function autoVoteMaps()
    if not C.AutoVote then return end
    if tick() - lastVoteTry < 1.2 then return end
    lastVoteTry = tick()
    local pg = LP:FindFirstChild("PlayerGui")
    if not pg then return end
    local candidates = {}
    for _, d in ipairs(pg:GetDescendants()) do
        if d:IsA("TextButton") or d:IsA("ImageButton") then
            local tx = ""
            pcall(function() tx = tostring(d.Text or "") end)
            local nm = tostring(d.Name or "")
            if preferMapName(tx) or preferMapName(nm) then
                if d.Visible ~= false and d.AbsoluteSize.X > 2 then
                    table.insert(candidates, d)
                end
            end
            -- 父级名字带地图
            local par = d.Parent
            if par and preferMapName(par.Name) then
                if d.Visible ~= false then table.insert(candidates, d) end
            end
        elseif d:IsA("TextLabel") and preferMapName(d.Text) then
            local btn = d.Parent and (d.Parent:IsA("GuiButton") and d.Parent or d.Parent:FindFirstChildWhichIsA("GuiButton"))
            if btn then table.insert(candidates, btn) end
        end
    end
    for _, b in ipairs(candidates) do
        if tryClick(b) then
            N("已投票: " .. tostring(b.Name or b.Text or "?"))
            return
        end
    end
end

local function detectMode()
    pcall(function()
        local g = LP:FindFirstChild("PlayerGui")
        if not g then return end
        local n = g:FindFirstChild("Notifications")
        local sg = n and n:FindFirstChild("SelectedGamemode")
        local o = sg and sg:FindFirstChild("Objective")
        local txt = o and (o:IsA("TextLabel") and o.Text or nil)
        if not txt then
            local l = o and o:FindFirstChildWhichIsA("TextLabel", true)
            txt = l and l.Text
        end
        if not txt or #txt < 2 then return end
        local lo = string.lower(txt)
        if string.find(lo, "tartarus") then currentMode = "Tartarus"
        elseif string.find(lo, "doomsday") or string.find(lo, "doom") then currentMode = "Doomsday"
        elseif string.find(lo, "chad") then currentMode = "Chad"
        elseif string.find(lo, "twisty") then currentMode = "Twisty"

-- 读大厅 Tartarus 预测进度
-- 路径: Workspace.Lobby.TartarusForecast.Screen.TartarusForecast
local function readLobbyFrag()
    pcall(function()
        local bestN = nil
        -- 官方路径（注意游戏里拼写是 Trartarus）
        -- 正确路径: Workspace.Lobby.TartarusForecast.Screen.TartarusForecast
        local lobby = workspace:FindFirstChild("Lobby")
        local forecast = lobby and (lobby:FindFirstChild("TartarusForecast") or lobby:FindFirstChild("TrartarusForecast"))
        local screen = forecast and (forecast:FindFirstChild("Screen") or forecast:FindFirstChild("屏幕"))
        local node = screen and (
            screen:FindFirstChild("TartarusForecast")
            or screen:FindFirstChild("TrartarusForecast")
            or screen:FindFirstChildWhichIsA("TextLabel")
            or screen:FindFirstChildWhichIsA("TextButton")
        )
        if not node and screen then node = screen end
        if node then
            local function takeText(obj)
                if not obj then return end
                if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
                    local tx = tostring(obj.Text or "")
                    local a, b = string.match(tx, "(%d+)%s*/%s*(%d+)")
                    if a and b and tonumber(b) == NEED_FRAG then
                        bestN = tonumber(a)
                    elseif bestN == nil then
                        local only = string.match(tx, "(%d+)")
                        if only then
                            local n = tonumber(only)
                            if n and n >= 0 and n <= NEED_FRAG then bestN = n end
                        end
                    end
                end
                -- StringValue / NumberValue
                if obj:IsA("StringValue") then
                    local a, b = string.match(tostring(obj.Value), "(%d+)%s*/%s*(%d+)")
                    if a and tonumber(b) == NEED_FRAG then bestN = tonumber(a) end
                elseif obj:IsA("IntValue") or obj:IsA("NumberValue") then
                    local n = tonumber(obj.Value)
                    if n and n >= 0 and n <= NEED_FRAG then bestN = n end
                end
            end
            takeText(node)
            for _, c in ipairs(node:GetDescendants()) do
                takeText(c)
            end
            -- Screen 层也扫一下
            if screen then
                for _, c in ipairs(screen:GetDescendants()) do
                    takeText(c)
                end
            end
        end
        if bestN ~= nil then
            serverFrag = bestN
            if serverFrag >= NEED_FRAG then tartUnlocked = true end
        end
    end)
    if myF > serverFrag then serverFrag = math.min(NEED_FRAG, myF) end
    if serverFrag >= NEED_FRAG then tartUnlocked = true end
end

local function creditFragment(src)
    local key = tostring(src)
    if countedFragKeys[key] then return end
    -- 同位置近似也去重
    local posKey = nil
    pcall(function()
        local part = best(src)
        if part then
            local p = part.Position
            posKey = string.format("f_%.0f_%.0f_%.0f", p.X/5, p.Y/5, p.Z/5)
        end
    end)
    if posKey and countedFragKeys[posKey] then return end
    countedFragKeys[key] = true
    if posKey then countedFragKeys[posKey] = true end
    foundThisRound = true
    sawLootThisRound = true
    myF = math.min(NEED_FRAG, myF + 1)
    serverFrag = math.min(NEED_FRAG, math.max(serverFrag, myF))

local function pickup(o, typ)
    if not o or not o.Parent then return end
    local key = tostring(o)
    if tried[key] and tick() - tried[key] < 0.8 then return end
    tried[key] = tick()
    -- 一开始捡就标记：避免捡完地图变0后误判为空图去重建
    if typ == "Fragment" or typ == "Glasses" then
        foundThisRound = true
    end
    local h = HRP()
    if not h then return end
    local part = best(o)
    if not part then return end
    -- 只试 2 个位置，减少卡顿
    local offs = { Vector3.new(0, 2, 0), Vector3.new(0, 1.5, 2) }
    for _, off in ipairs(offs) do
        if not o.Parent then break end
        pcall(function() h.CFrame = part.CFrame + off end)
        local pr = o:FindFirstChildWhichIsA("ProximityPrompt", true)
        if pr and typeof(fireproximityprompt) == "function" then
            pcall(fireproximityprompt, pr)
        end
        if typeof(firetouchinterest) == "function" then
            pcall(function()
                firetouchinterest(h, part, 0)
                firetouchinterest(h, part, 1)
            end)
        end
        task.wait(0.04)
    end
    task.delay(0.4, function()
        if not o or not o.Parent then
            if typ == "Fragment" then
                creditFragment(o)
                if serverFrag >= NEED_FRAG or myF >= NEED_FRAG then
                    tartUnlocked = true
                    N("Tartarus 应已解锁!")
                end
            elseif typ == "Glasses" then
                gotG = true
                foundThisRound = true
                N("眼镜已找到")
            end
        end
    end)
end

-- 先处理 kick，再 rejoin
local function rejoin(reason)
    if rejoining then return end
    if not C.Rejoin and not pendingHop then return end
    rejoining = true
    N("重进中: " .. tostring(reason or "?"))
    print("[tau] rejoin:", reason, "kickHandled=", kickHandled)
    queueScript()
    -- 若还没走过 kick 流程，先等一下让离开生效
    if not kickHandled then
        N("等待 kick/离开处理...")
        task.wait(2.0)
        kickHandled = true
    else
        -- 已 kick 过：再稍等，避免同 JobId
        task.wait(pendingHop and 1.5 or 0.5)
    end
    N("开始 rejoin 新实例...")
    doTeleport()
    task.delay(10, function()
        rejoining = false
        pendingHop = false
        kickHandled = false
    end)
end

-- 是否还能搜：有碎片，或（非仅碎片模式且有眼镜）

local function liveScanHasLoot()
    local hasF, hasG = false, false
    pcall(function()
        for _, o in ipairs(workspace:GetDescendants()) do
            if isF(o) then
                hasF = true
                kF[o] = true
            elseif isG(o) then
                hasG = true
                kG[o] = true
            end
            if hasF and hasG then break end
        end
    end)
    if hasF or hasG then
        sawLootThisRound = true
    end
    if wantFrag() and hasF then return true end
    if wantGlass() and hasG then return true end
    return false
end
local function stillHasLoot()
    if wantFrag() and mapFragCount() > 0 then return true end
    if wantGlass() and mapGlassCount() > 0 then return true end
    if liveScanHasLoot() then return true end
    return false
end
local function shouldAutoHop()
    if not C.AutoHop or not C.Auto then return false end
    if pendingHop or rejoining then return false end
    local inRest = (currentMode == "Intermission" or currentMode == "Unknown")
    if inRest then return false end
    -- 已达目标 / 本局捡过 → 留下，绝不因为目标达成而 rejoin
    if reachedFragTarget() or foundThisRound or myF > 0 or gotG then
        return false
    end
    if (tick() - serverJoinTime) < MIN_STAY then return false end
    -- 仅：空图且本局从未见过货 → 重建
    if sawLootThisRound then
        return false
    end
    pcall(fullScan)
    if stillHasLoot() then
        sawLootThisRound = true
        return false
    end
    if liveScanHasLoot() then
        return false
    end
    return true
end

local function canSmartHop(force)
    if force then return true, "强制换服" end
    if not C.SmartHop then return true, "智能换服关" end
    -- 1) 先扫干净
    pcall(fullScan)
    local fc, gc = mapFragCount(), mapGlassCount()
    if fc > 0 then
        zeroFragSince = nil
        return false, string.format("地图还有 %d 碎片，搜干净再换", fc)
    end
    if not C.OnlyFrag and gc > 0 then
        zeroFragSince = nil
        return false, string.format("地图还有 %d 眼镜，先捡再换", gc)
    end
    -- 2) 确认 0：至少稳定一段时间
    local need = tonumber(C.HopWait) or 0
    if need <= 0 then
        return true, "70秒后无碎片，直接重建"
    end
    if zeroFragSince == nil then
        zeroFragSince = tick()
        return false, string.format("地图已是 0，再确认 %d 秒", need)
    end
    local waited = tick() - zeroFragSince
    if waited < need then
        return false, string.format("确认中 %.0f/%d 秒 (地图0)", waited, need)
    end
    return true, "搜净且确认地图0，可以换服"
end

local function hop(force)
    if pendingHop or rejoining then return end
    -- 非强制：有碎片就绝对不换
    if not force then
        -- 达目标/本局有货：留下，不 rejoin
        if reachedFragTarget() or foundThisRound or myF > 0 or gotG then
            N("本局已达目标或已有碎片 → 留下不换服")
            return
        end
        pcall(fullScan)
        if liveScanHasLoot() or stillHasLoot() then
            sawLootThisRound = true
            N("地图上还有碎片/眼镜 → 取消换服")
            return
        end
        if sawLootThisRound then
            N("本局曾发现过物资 → 不换服")
            return
        end
    end
    pendingHop = true
    kickHandled = false
    rejoining = false
    N("换服: ①离开 → ②重建")
    queueScript()
    task.spawn(function()
        pcall(function()
            if typeof(LP.Kick) == "function" then LP:Kick("tau auto hop") end
        end)
        task.wait(2.5)
        kickHandled = true
        task.wait(1.0)
        if pendingHop then
            rejoining = false
            rejoin("AutoHop")
        end
    end)
end

local tScan, tShel, tEsp, tAfk, tMode, tLoot = 0, 0, 0, 0, 0, 0
local lastEspKey = ""
task.spawn(function()
    -- 启动时扫一次
    pcall(fullScan)
    pcall(readLobbyFrag)
    lastLobbyRead = tick()
    while Gui and Gui.Parent do
        pcall(function()
            local now = tick()
            -- UI 数字：1 秒更新
            -- 地图数量 2 秒刷新；大厅进度在 updFragUI 内最多 5 秒读一次
            if now - tScan >= 2 then
                tScan = now
                local fc, gc = scan()
                updFragUI(fc, false)
                GlassL.Text = (gotG or gc > 0) and "眼镜: 有" or "眼镜: 未找到"
                -- 地图出来后：没碎片就短确认再换（不等90秒）
                -- 休息/投票中不换
                local inRest = (currentMode == "Intermission" or currentMode == "Unknown")
                if (tick() - serverJoinTime) >= MIN_STAY and not didEarlyScan then
                    didEarlyScan = true
                    pcall(fullScan)
                    fc = select(1, countKnown())
                end
                local hasLoot = stillHasLoot()
                if hasLoot or inRest or not C.Auto then
                    zeroFragSince = nil
                    if HopHint and C.AutoHop then
                        if inRest then
                            HopHint.Text = "自动换服: 休息/投票中..."
                        elseif hasLoot then
                            HopHint.Text = string.format("自动换服: 还有物资(碎%d 眼%d)", mapFragCount(), mapGlassCount())
                        elseif not C.Auto then
                            HopHint.Text = "自动换服: 请开 Auto"
                        end
                    end
                elseif shouldAutoHop() then
                    if HopHint then
                        HopHint.Text = "自动换服: 空图 → 重建"
                        HopHint.TextColor3 = Color3.fromRGB(100, 255, 140)
                    end
                    N("70秒空图无货 → 重建")
                    hop(false)
                elseif HopHint and C.AutoHop then
                    local left = MIN_STAY - (tick() - serverJoinTime)
                    if left > 0 then
                        HopHint.Text = string.format("自动换服: 进图等待 %.0fs", left)
                    end
                end
            end
            -- Auto 捡：每 0.9 秒最多捡 3 个最近的（不双遍）
            if C.Auto and now - tLoot >= 0.9 then
                tLoot = now
                local h = HRP()
                local pos = h and h.Position or Vector3.zero
                -- 碎片
                if wantFrag() and not reachedFragTarget() then
                    local list = {}
                    for o in pairs(kF) do
                        if o and o.Parent then
                            local p = best(o)
                            if p then table.insert(list, {o = o, d = (p.Position - pos).Magnitude}) end
                        end
                    end
                    table.sort(list, function(a, b) return a.d < b.d end)
                    local n = 0
                    for _, d in ipairs(list) do
                        if n >= 3 then break end
                        if reachedFragTarget() then break end
                        pickup(d.o, "Fragment")
                        n = n + 1
                        task.wait(0.05)
                    end
                end
                -- 眼镜
                if wantGlass() and not gotG then
                    for o in pairs(kG) do
                        if o and o.Parent then
                            pickup(o, "Glasses")
                            break
                        end
                    end
                end
            end
            -- ESP：2 秒刷新，数量不变则跳过重建
            if C.ESP and now - tEsp >= 2 then
                tEsp = now
                local fc, gc = countKnown()
                local key = fc .. ":" .. gc
                if key ~= lastEspKey then
                    lastEspKey = key
                    clrESP()
                    local n = 0
                    for o in pairs(kF) do
                        if o and o.Parent and n < 15 then
                            addESP(o, Color3.fromRGB(255, 70, 70))
                            n = n + 1
                        end
                    end
                    for o in pairs(kG) do
                        if o and o.Parent then
                            addESP(o, Color3.fromRGB(80, 220, 220))
                        end
                    end
                end
            elseif not C.ESP then
                lastEspKey = ""
            end
            if C.ShelAuto and now - tShel >= 30 then
                tShel = now
                toShel()
            end
            if C.AFK and now - tAfk >= 45 then
                tAfk = now
                afk()
            end
            if now - tMode >= 2.5 then
                tMode = now
                detectMode()
            end
        end)
        task.wait(0.25)
    end
end)
