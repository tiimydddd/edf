-- ==================== Functions ====================
local function Notify(msg)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = "tau only apple",
            Text = tostring(msg),
            Duration = 2.5
        })
    end)
end

local function getHRP()
    local c = LocalPlayer.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function isFragment(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(FragmentNames) do
        if n:find(v:lower()) then return true end
    end
    return false
end

local function isShelterKey(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(ShelterKeyNames) do
        if n:find(v:lower()) then return true end
    end
    return false
end

local function isShelter(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(ShelterNames) do
        if n:find(v:lower()) then return true end
    end
    return false
end

local function matchKeywords(obj, keywords)
    local n = obj.Name:lower()
    for _, k in ipairs(keywords) do
        if n:find(k:lower()) then return true end
    end
    return false
end

local function cleanCache(cache)
    for obj in pairs(cache) do
        if not obj or not obj.Parent then
            cache[obj] = nil
        end
    end
end

local function pickup(obj, typ)
    local hrp = getHRP()
    if not hrp then return end
    local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
    if not part then return end

    hrp.CFrame = part.CFrame + Vector3.new(0, 3, 0)

    local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
    if prompt then pcall(fireproximityprompt, prompt) end

    local touch = part:FindFirstChildOfClass("TouchInterest")
    if touch then
        pcall(firetouchinterest, hrp, part, 0)
        task.wait(0.03)
        pcall(firetouchinterest, hrp, part, 1)
    end

    local click = obj:FindFirstChildOfClass("ClickDetector") or part:FindFirstChildOfClass("ClickDetector")
    if click then pcall(fireclickdetector, click) end

    if typ == "Fragment" then
        collectedFragments = collectedFragments + 1
        Notify("Fragment " .. collectedFragments .. "/5")
    elseif typ == "Key" then
        hasShelterKey = true
        Notify("Shelter Key got!")
    end
end

local function goToShelter()
    local hrp = getHRP()
    if not hrp then return end
    cleanCache(knownShelters)

    for obj in pairs(knownShelters) do
        if obj and obj.Parent then
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                hrp.CFrame = part.CFrame + Vector3.new(0, 5, 0)
                local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
                if prompt then pcall(fireproximityprompt, prompt) end
                Notify("Trying Shelter...")
                return
            end
        end
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if isShelter(obj) then
            knownShelters[obj] = true
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                hrp.CFrame = part.CFrame + Vector3.new(0, 5, 0)
                local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
                if prompt then pcall(fireproximityprompt, prompt) end
                Notify("Found Shelter")
                return
            end
        end
    end
end

local function teleportToKeyword(keywords, label)
    local hrp = getHRP()
    if not hrp then
        Notify("没有角色")
        return
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if matchKeywords(obj, keywords) then
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                hrp.CFrame = part.CFrame + Vector3.new(0, 5, 0)
                Notify("已传送到 " .. label)
                return
            end
        end
    end
    Notify("没找到 " .. label)
end

local function teleportToPlayer(player)
    local hrp = getHRP()
    if not hrp then return end
    local targetChar = player.Character
    if not targetChar then
        Notify(player.Name .. " 没有角色")
        return
    end
    local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
    if targetHRP then
        hrp.CFrame = targetHRP.CFrame + Vector3.new(0, 3, 0)
        Notify("已传送到 " .. player.Name)
    end
end

local function refreshPlayerList()
    for _, child in ipairs(PlayerListFrame:GetChildren()) do
        if child:IsA("TextButton") then
            child:Destroy()
        end
    end

    local y = 0
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local btn = Instance.new("TextButton")
            btn.Size = UDim2.new(1, -8, 0, 26)
            btn.BackgroundColor3 = Color3.fromRGB(50, 50, 65)
            btn.Text = plr.Name
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            btn.Font = Enum.Font.Gotham
            btn.TextSize = 12
            btn.Parent = PlayerListFrame
            Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)

            btn.MouseButton1Click:Connect(function()
                teleportToPlayer(plr)
            end)
            y = y + 30
        end
    end
    PlayerListFrame.CanvasSize = UDim2.new(0, 0, 0, y + 10)
end

-- 监听
workspace.DescendantAdded:Connect(function(obj)
    task.defer(function()
        if isFragment(obj) then
            knownFragments[obj] = true
            if Settings.Enabled or Settings.FullAuto then
                pickup(obj, "Fragment")
            end
        elseif isShelterKey(obj) then
            knownShelterKeys[obj] = true
            if Settings.AutoShelterKey or Settings.FullAuto then
                pickup(obj, "Key")
            end
        elseif isShelter(obj) then
            knownShelters[obj] = true
        end
    end)
end)

workspace.DescendantRemoving:Connect(function(obj)
    knownFragments[obj] = nil
    knownShelterKeys[obj] = nil
    knownShelters[obj] = nil
end)

task.spawn(function()
    task.wait(1.2)
    for _, obj in ipairs(workspace:GetDescendants()) do
        if isFragment(obj) then knownFragments[obj] = true end
        if isShelterKey(obj) then knownShelterKeys[obj] = true end
        if isShelter(obj) then knownShelters[obj] = true end
    end
    refreshPlayerList()
end)

-- 主循环
task.spawn(function()
    while ScreenGui and ScreenGui.Parent do
        local now = tick()

        StatusLabel.Text = "Status: " .. ((Settings.Enabled or Settings.FullAuto) and "Running" or "Off")
        StatusLabel.TextColor3 = (Settings.Enabled or Settings.FullAuto) and Color3.fromRGB(80, 255, 100) or Color3.fromRGB(255, 90, 90)
        FragLabel.Text = "Fragments: " .. collectedFragments .. " / 5"
        KeyLabel.Text = "Shelter Key: " .. (hasShelterKey and "YES" or "No")

        if (Settings.Enabled or Settings.FullAuto) and (now - lastScan) >= SCAN_INTERVAL then
            lastScan = now
            cleanCache(knownFragments)
            cleanCache(knownShelterKeys)

            for obj in pairs(knownFragments) do
                if obj and obj.Parent then
                    pickup(obj, "Fragment")
                    task.wait(0.07)
                end
            end

            if Settings.AutoShelterKey or Settings.FullAuto then
                for obj in pairs(knownShelterKeys) do
                    if obj and obj.Parent then
                        pickup(obj, "Key")
                        task.wait(0.07)
                    end
                end
            end
        end

        if (Settings.AutoGoShelter or Settings.FullAuto) and hasShelterKey and (now - lastShelterCheck) >= SHELTER_CHECK_INTERVAL then
            lastShelterCheck = now
            goToShelter()
        end

        task.wait(0.25)
    end
end)

local dragging, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)
TitleBar.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local d = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
