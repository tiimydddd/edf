-- ==================== 加强模式检测 ====================
local Signals = RPS:FindFirstChild("Assets") and RPS.Assets:FindFirstChild("Signals")
local EventSignal = Signals and Signals:FindFirstChild("Event")

local currentMode = "Unknown"
local currentObjective = ""
local isIntermission = false
local tartarusActive = false

-- 直接监听服务器发来的 UI 事件（最准）
if EventSignal then
    EventSignal.OnClientEvent:Connect(function(data)
        if typeof(data) ~= "table" or not data.reason then return end

        local reason = data.reason

        -- 模式相关
        if reason == "Cutscene" and data.F and data.F.Name then
            currentMode = data.F.Name
            if data.F.Name == "Tartarus" then
                tartarusActive = true
                tart = true
                alrt()
                showBig("Tartarus 已出现！", Color3.fromRGB(255, 100, 80), 6)
                N("Tartarus 模式开启", true)
            end
        end

        if reason == "TartarusClock" or reason == "TartarusSummon" then
            tartarusActive = true
            tart = true
            if data.B then
                N("Tartarus: " .. tostring(data.B), true)
            end
        end

        if reason == "Intermission" then
            isIntermission = data.A == true
            if isIntermission then
                currentMode = "Intermission"
            end
        end

        if reason == "DoomsDayForecast" then
            currentMode = "Doomsday"
            N("Doomsday Forecast 出现！", true)
        end

        if reason == "ChadSummon" then
            N("Chad 已召唤！", true)
        end

        if reason == "TwistyTalk" and data.A then
            N("Twisty: " .. tostring(data.A))
        end

        if reason == "Notify" and data.B == "Loot" then
            N((data.A and data.A.Name or "?") .. " 找到了 " .. tostring(data.C))
        end
    end)
end

-- 备用：继续扫 GUI（防止 Event 漏掉）
local function getObj()
    local g = LP:FindFirstChild("PlayerGui")
    if not g then return nil end
    local ok, t = pcall(function()
        local n = g:FindFirstChild("Notifications")
        if not n then return nil end
        local sg = n:FindFirstChild("SelectedGamemode")
        if not sg then return nil end
        local o = sg:FindFirstChild("Objective")
        if not o then return nil end
        if o:IsA("TextLabel") or o:IsA("TextButton") then return o.Text end
        local l = o:FindFirstChildWhichIsA("TextLabel", true)
        return l and l.Text
    end)
    return ok and t or nil
end

local function getModeName()
    local g = LP:FindFirstChild("PlayerGui")
    if not g then return nil end
    local ok, name = pcall(function()
        local n = g:FindFirstChild("Notifications")
        if not n then return nil end
        local sg = n:FindFirstChild("SelectedGamemode")
        if not sg then return nil end
        -- 有的版本名字在 Gamemode 图片旁边的文字，或 Objective 上方
        local obj = sg:FindFirstChild("Objective")
        if obj and (obj:IsA("TextLabel") or obj:IsA("TextButton")) then
            return obj.Text
        end
        return nil
    end)
    return ok and name or nil
end

local function checkEnter()
    local ot = getObj()
    if ot and #ot >= 3 then
        local lo = string.lower(ot)
        currentObjective = ot

        if not lo:find("intermission") and not lo:find("voting") and not lo:find("vote") then
            markEnter()
        end

        -- 从 Objective 文字里再补一次模式名
        if lo:find("tartarus") then
            if not tart then
                tart = true
                tartarusActive = true
                alrt()
                showBig("Tartarus 已出现", Color3.fromRGB(255, 120, 80), 5)
                N("Tartarus!", true)
            end
            currentMode = "Tartarus"
        elseif lo:find("doomsday") or lo:find("doom") then
            currentMode = "Doomsday"
        elseif lo:find("chad") then
            currentMode = "Chad"
        elseif lo:find("twisty") then
            currentMode = "Twisty"
        elseif lo:find("weeping") then
            currentMode = "Weeping Angel"
        end
    end
end
