-- ==================== Functions ====================
local function Notify(text)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = "TAU Auto Loot",
            Text = text,
            Duration = 3.5
        })
    end)
    print("[TAU] " .. text)
end

local function getHRP()
    local char = LocalPlayer.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function isTarget(obj)
    local name = obj.Name:lower()
    for _, n in ipairs(FragmentNames) do
        if name:find(n:lower()) then return "Fragment" end
    end
    if not Settings.OnlyFragments then
        for _, n in ipairs(GlassesNames) do
            if name:find(n:lower()) then return "Glasses" end
        end
    end
    if Settings.AutoShelterKey then
        for _, n in ipairs(ShelterKeyNames) do
            if name:find(n:lower()) then return "ShelterKey" end
        end
    end
    return nil
end

local function isExtinguisher(obj)
    local name = obj.Name:lower()
    for _, n in ipairs(ExtinguisherNames) do
        if name:find(n:lower()) then return true end
    end
    return false
end

local function countPhysicalFragments()
    local count = 0
    for _, obj in ipairs(workspace:GetDescendants()) do
        local name = obj.Name:lower()
        for _, n in ipairs(FragmentNames) do
            if name:find(n:lower()) then
                count = count + 1
                break
            end
        end
    end
    return count
end

local function playSpawnSound()
    if not Settings.SpawnSound then return end
    local s = Instance.new("Sound")
    s.SoundId = "rbxassetid://6026984224" -- simple alert sound
    s.Volume = 1.5
    s.Parent = SoundService
    s:Play()
    task.delay(2, function() s:Destroy() end)
end

local function tryPickup(obj, itemType)
    local hrp = getHRP()
    if not hrp then return end
    local part = obj:IsA("BasePart") and obj or (obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")))
    if not part then return end

    hrp.CFrame = part.CFrame + Vector3.new(0, 2.8, 0)
    local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
    if prompt then pcall(fireproximityprompt, prompt) end
    local touch = part:FindFirstChildOfClass("TouchInterest")
    if touch then
        pcall(firetouchinterest, hrp, part, 0)
        task.wait(0.04)
        pcall(firetouchinterest, hrp, part, 1)
    end
    local click = obj:FindFirstChildOfClass("ClickDetector") or part:FindFirstChildOfClass("ClickDetector")
    if click then pcall(fireclickdetector, click) end

    if itemType == "Fragment" then
        foundAnyFragment = true
        collectedFragments = collectedFragments + 1
        Notify("Fragment collected! My total: " .. collectedFragments .. "/5")
        playSpawnSound()
    elseif itemType == "Glasses" then
        glassesFound = true
        Notify("Mysterious Glasses found!")
        playSpawnSound()
    elseif itemType == "ShelterKey" then
        Notify("Shelter Key picked up")
    end
    lastFoundTime = tick()
end

local function tryPickExtinguisher()
    local found = false
    for _, obj in ipairs(workspace:GetDescendants()) do
        if isExtinguisher(obj) then
            found = true
            tryPickup(obj, "Fire Extinguisher")
            task.wait(0.15)
        end
    end
    Notify(found and "Fire Extinguisher found" or "No Fire Extinguisher on map")
end

local function clearESP()
    if espFolder then
        espFolder:Destroy()
        espFolder = nil
    end
end

local function updateESP()
    clearESP()
    if not (Settings.ESPFragments or Settings.ESPGlasses) then return end

    espFolder = Instance.new("Folder")
    espFolder.Name = "TAU_ESP"
    espFolder.Parent = CoreGui

    for _, obj in ipairs(workspace:GetDescendants()) do
        local name = obj.Name:lower()
        local isFrag, isGlass = false, false
        for _, n in ipairs(FragmentNames) do
            if name:find(n:lower()) then isFrag = true break end
        end
        for _, n in ipairs(GlassesNames) do
            if name:find(n:lower()) then isGlass = true break end
        end

        if (isFrag and Settings.ESPFragments) or (isGlass and Settings.ESPGlasses) then
            local part = obj:IsA("BasePart") and obj or (obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")))
            if part then
                local highlight = Instance.new("Highlight")
                highlight.Adornee = obj:IsA("Model") and obj or part
                highlight.FillColor = isFrag and Color3.fromRGB(255, 40, 40) or Color3.fromRGB(80, 180, 255)
                highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                highlight.FillTransparency = 0.4
                highlight.Parent = espFolder
            end
        end
    end
end

local function makePartsFly()
    local hrp = getHRP()
    if not hrp then return end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and not obj.Anchored and not obj:IsDescendantOf(LocalPlayer.Character) then
            if (obj.Position - hrp.Position).Magnitude <= FlyRange then
                local old = obj:FindFirstChild("FlyForce")
                if old then old:Destroy() end
                local bv = Instance.new("BodyVelocity")
                bv.Name = "FlyForce"
                bv.MaxForce = Vector3.new(0, math.huge, 0)
                bv.Velocity = Vector3.new(0, UpForce, 0)
                bv.Parent = obj
                task.delay(2.5, function() if bv and bv.Parent then bv:Destroy() end end)
            end
        end
    end
end

local function toggleFly()
    flyEnabled = not flyEnabled
    if flyEnabled then
        FlyBtn.Text = "Fly Parts: ON"
        FlyBtn.BackgroundColor3 = Color3.fromRGB(40, 140, 70)
        Notify("Fly Parts Enabled")
        flyConnection = RunService.Heartbeat:Connect(makePartsFly)
    else
        FlyBtn.Text = "Fly Parts: OFF"
        FlyBtn.BackgroundColor3 = Color3.fromRGB(55, 55, 70)
        Notify("Fly Parts Disabled")
        if flyConnection then flyConnection:Disconnect() flyConnection = nil end
    end
end

local function doAntiAFK()
    if not Settings.AntiAFK then return end
    if tick() - lastAFKAction < 40 then return end
    lastAFKAction = tick()
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.Jump = true
    end
end

local function getMapName()
    -- Try common places for map name
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if playerGui then
        for _, gui in ipairs(playerGui:GetDescendants()) do
            if gui:IsA("TextLabel") or gui:IsA("TextButton") then
                local t = gui.Text or ""
                if #t > 3 and #t < 30 and not t:lower():find("round") and not t:lower():find("mode") then
                    -- very rough, just show something if available
                end
            end
        end
    end
    return workspace.Name ~= "Workspace" and workspace.Name or "Unknown"
end

-- 5th Fragment + System Overload detection
local function checkAlerts()
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return end
    for _, gui in ipairs(playerGui:GetDescendants()) do
        if gui:IsA("TextLabel") or gui:IsA("TextButton") then
            local text = string.lower(gui.Text or "")

            -- 5th fragment
            if (text:find("5th") or text:find("fifth") or text:find("all five")) and text:find("fragment") then
                serverFragmentProgress = 5
                tartarusUnlocked = true
            elseif text:find("tartarus has been summoned") or text:find("tartarus has been awakened") then
                serverFragmentProgress = 5
                tartarusUnlocked = true
            end

            -- System Overload
            if Settings.DetectOverload then
                if text:find("system overload") or text:find("系统过载") or text:find("overload") then
                    if not systemOverloadActive then
                        systemOverloadActive = true
                        Notify("⚠ System Overload detected!")
                    end
                end
            end
        end
    end
end

local function updateModeDetection()
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return end
    local bestLen, bestMode, bestIsNext = 0, nil, false
    for _, gui in ipairs(playerGui:GetDescendants()) do
        if gui:IsA("TextLabel") or gui:IsA("TextButton") then
            local text = string.lower(gui.Text or "")
            for keyword, data in pairs(strictModes) do
                if text:find(keyword) and #keyword > bestLen then
                    bestLen = #keyword
                    bestMode = data.mode
                    bestIsNext = data.next
                end
            end
        end
    end
    if bestMode then
        currentDetectedMode = bestMode
        isNextMode = bestIsNext
        lastModeUpdate = tick()
    elseif tick() - lastModeUpdate > MODE_TIMEOUT then
        currentDetectedMode = "Unknown"
        isNextMode = false
    end
end

-- ==================== Main Loop ====================
task.spawn(function()
    while true do
        local physical = countPhysicalFragments()
        checkAlerts()
        updateModeDetection()
        doAntiAFK()

        if FragmentCountLabel then
            if serverFragmentProgress >= 5 or tartarusUnlocked then
                FragmentCountLabel.Text = "Server Progress: 5/5 (Tartarus Unlocked)"
                FragmentCountLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
            else
                FragmentCountLabel.Text = "Map Fragments: " .. physical .. "  |  Server: ?"
                FragmentCountLabel.TextColor3 = physical > 0 and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(180, 185, 200)
            end
        end
        if CollectedLabel then
            CollectedLabel.Text = "My Collected: " .. collectedFragments .. " / 5"
            CollectedLabel.TextColor3 = collectedFragments >= 5 and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(180, 185, 200)
        end
        if TartarusLabel then
            if tartarusUnlocked or serverFragmentProgress >= 5 or collectedFragments >= 5 then
                TartarusLabel.Text = "Tartarus: UNLOCKED!"
                TartarusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
            else
                TartarusLabel.Text = "Tartarus: Locked (My " .. collectedFragments .. "/5)"
                TartarusLabel.TextColor3 = Color3.fromRGB(255, 120, 120)
            end
        end
        if GlassesLabel then
            GlassesLabel.Text = glassesFound and "Glasses: FOUND ✓" or "Glasses: Not Found"
            GlassesLabel.TextColor3 = glassesFound and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(180, 185, 200)
        end
        if ModeLabel then
            if isNextMode then
                ModeLabel.Text = "Next Mode: " .. currentDetectedMode
                ModeLabel.TextColor3 = Color3.fromRGB(100, 200, 255)
            else
                ModeLabel.Text = "Current Mode: " .. currentDetectedMode
                ModeLabel.TextColor3 = currentDetectedMode == "Unknown" and Color3.fromRGB(180, 185, 200) or Color3.fromRGB(255, 200, 80)
            end
        end
        if MapLabel and Settings.ShowMapName then
            MapLabel.Text = "Map: " .. getMapName()
        end
        if RoundLabel then
            RoundLabel.Text = "Rounds tracked: " .. roundCount
        end
        if OverloadLabel then
            OverloadLabel.Text = systemOverloadActive and "System Overload: YES ⚠" or "System Overload: No"
            OverloadLabel.TextColor3 = systemOverloadActive and Color3.fromRGB(255, 80, 80) or Color3.fromRGB(180, 185, 200)
        end

        if Settings.Enabled then
            local hrp = getHRP()
            if hrp then
                for _, obj in ipairs(workspace:GetDescendants()) do
                    if not Settings.Enabled then break end
                    local itemType = isTarget(obj)
                    if itemType then
                        tryPickup(obj, itemType)
                        task.wait(0.18)
                    end
                end
            end
        end

        if Settings.ESPFragments or Settings.ESPGlasses then
            updateESP()
        else
            clearESP()
        end

        task.wait(Settings.ScanSpeed)
    end
end)

-- Dragging
local dragging, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
