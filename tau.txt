    local n = obj.Name:lower()
    for _, k in ipairs(keywords) do
        if n:find(k:lower()) then return true end
    end

local function cleanCache(cache)
    for obj in pairs(cache) do
        if not obj or not obj.Parent then
            cache[obj] = nil
        end
    end
end

local function countFrags()
    cleanCache(knownFragments)
    local c = 0
    for _ in pairs(knownFragments) do c = c + 1 end
    return c
end

local function shouldPickup(typ)
    if Settings.Mode == 1 then
        return typ == "Fragment"
    elseif Settings.Mode == 2 then
        return typ == "Glasses"
    elseif Settings.Mode == 3 then
        return typ == "Fragment" or typ == "Glasses"
    elseif Settings.Mode == 4 then
        return typ == "Fragment" or typ == "Glasses" or typ == "Key"
    elseif Settings.Mode == 5 then
        return typ == "Fragment" or typ == "Glasses" or typ == "Bat"
    end
    return false
end

-- ★ 已移除自动对拉锁
local function pickup(obj, typ)
    if not shouldPickup(typ) then return end

    local hrp = getHRP()
    if not hrp then return end
    local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
    if not part then return end

    local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
    if prompt then pcall(fireproximityprompt, prompt) end

    local touch = part:FindFirstChildOfClass("TouchInterest")
    if touch then
        pcall(firetouchinterest, hrp, part, 0)
        task.wait(0.03)
        pcall(firetouchinterest, hrp, part, 1)
    end

    local click = obj:FindFirstChildOfClass("ClickDetector") or part:FindFirstChildOfClass("ClickDetector")
    if click then pcall(fireclickdetector, click) end

    if typ == "Fragment" then
        collectedFragments = collectedFragments + 1
        Notify("Fragment " .. collectedFragments .. "/5")
    elseif typ == "Glasses" then
        glassesFound = true
        Notify("Glasses found!")
    elseif typ == "Key" then
        keyFound = true
        Notify("Key found!")
    elseif typ == "Bat" then
        batFound = true
        Notify("Bat found!")
    end
end

local function goToShelter()
    local hrp = getHRP()
    if not hrp then return end
    cleanCache(knownShelters)

    for obj in pairs(knownShelters) do
        if obj and obj.Parent then
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                hrp.CFrame = part.CFrame + Vector3.new(0, 5, 0)
                local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
                if prompt then pcall(fireproximityprompt, prompt) end
                Notify("Trying Shelter door...")
                return
            end
        end
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if isShelter(obj) then
            knownShelters[obj] = true
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                hrp.CFrame = part.CFrame + Vector3.new(0, 5, 0)
                local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
                if prompt then pcall(fireproximityprompt, prompt) end
                Notify("Found Shelter")
                return
            end
        end
    end
end

local function teleportToKeyword(keywords, label)
    local hrp = getHRP()
    if not hrp then
        Notify("没有角色")
        return
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if matchKeywords(obj, keywords) then
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                hrp.CFrame = part.CFrame + Vector3.new(0, 5, 0)
                Notify("已传送到 " .. label)
                return
            end
        end
    end
    Notify("没找到 " .. label)
end

local function teleportToPlayer(player)
    local hrp = getHRP()
    if not hrp then return end
    local targetChar = player.Character
    if not targetChar then
        Notify(player.Name .. " 没有角色")
        return

    local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
    if targetHRP then
        hrp.CFrame = targetHRP.CFrame + Vector3.new(0, 3, 0)
        Notify("已传送到 " .. player.Name)
    end
end

local function checkFirstRoundNoFrag()
    if firstRoundChecked or not Settings.RejoinNoFrag then return end
    if tick() - joinTime < FIRST_ROUND_WAIT then return end

    firstRoundChecked = true
    local frags = countFrags()

    if frags <= 0 then
        Notify("第1局没有血月碎片，正在重进...")
        task.wait(1)
        TeleportService:Teleport(PLACE_ID, LocalPlayer)
    else
        Notify("第1局检测到碎片: " .. frags)
    end
end

-- ==================== A. 检测当前/下一个模式 ====================
local function detectModes()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end

    local foundCurrent = nil
    local foundNext = nil

    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextLabel") or g:IsA("TextButton") or g:IsA("TextBox") then
            local t = (g.Text or ""):lower()
            if t == "" then continue end

            for _, pair in ipairs(GamemodeKeywords) do
                if t:find(pair[1]) then
                    if t:find("next") or t:find("upcoming") or t:find("incoming") or t:find("下一") then
                        foundNext = pair[2]
                    else
                        foundCurrent = pair[2]
                    end
                end
            end
        end
    end

    if foundCurrent then currentMode = foundCurrent end
    if foundNext then nextMode = foundNext end
end

-- ==================== B. 自动投票地图 ====================
local function autoVoteMap()
    if not Settings.AutoVoteMap then return end
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end

    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextButton") or g:IsA("ImageButton") then
            local text = (g.Text or ""):lower()
            local name = g.Name:lower()

            -- 优先投指定地图
            for _, mapName in ipairs(PreferredMaps) do
                if text:find(mapName:lower()) or name:find(mapName:lower()) then
                    pcall(function()
                        firesignal(g.MouseButton1Click)
                        g:Activate()
                    end)
                    Notify("已自动投票地图: " .. mapName)
                    return
                end
            end
        end
    end

    -- 没找到优先地图就投第一个看起来像投票按钮的
    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextButton") then
            local text = (g.Text or ""):lower()
            if text:find("vote") or text:find("map") or #text > 3 then
                pcall(function()
                    firesignal(g.MouseButton1Click)
                end)
                return
            end
        end
    end
end

-- ==================== C. 额外灾难提醒 ====================
local function checkExtraDisaster()
    if not Settings.ExtraDisasterAlert then return end
    if tick() - lastExtraNotify < 8 then return end

    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end

    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextLabel") or g:IsA("TextButton") then
            local t = (g.Text or ""):lower()
            for _, pair in ipairs(ExtraDisasterKeywords) do
                if t:find(pair[1]) then
                    lastExtraNotify = tick()
                    Notify("⚠ 额外灾难: " .. pair[2])
                    return
                end
            end
        end
    end
end

local function checkChadDice()
    if not Settings.ChadDiceAlert then return end
    if tick() - lastChadDiceNotify < 6 then return end

    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end

    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextLabel") or g:IsA("TextButton") or g:IsA("ImageLabel") then
            local t = ((g.Text or "") .. " " .. g.Name):lower()
            if t:find("dice") or t:find("roll") or t:find("chad") and (t:find("queue") or t:find("骰子") or t:find("点数")) then
                lastChadDiceNotify = tick()
                Notify("🎲 Chad 骰子出现！准备排队稀有模式")
                return
            end
        end
    end
end

workspace.DescendantAdded:Connect(function(obj)
    task.defer(function()
        if isFragment(obj) then
            knownFragments[obj] = true
            if Settings.Enabled or Settings.FullAuto then
                pickup(obj, "Fragment")
            end
        elseif isGlasses(obj) then
            knownGlasses[obj] = true
            if Settings.Enabled or Settings.FullAuto then
                pickup(obj, "Glasses")
            end
        elseif isKey(obj) then
            knownKeys[obj] = true
            if Settings.Enabled or Settings.FullAuto then
                pickup(obj, "Key")
            end
        elseif isBat(obj) then
            knownBats[obj] = true
            if Settings.Enabled or Settings.FullAuto then
                pickup(obj, "Bat")
            end
        elseif isShelter(obj) then
            knownShelters[obj] = true
        end
    end)
end)

workspace.DescendantRemoving:Connect(function(obj)
    knownFragments[obj] = nil
    knownGlasses[obj] = nil
    knownKeys[obj] = nil
    knownBats[obj] = nil
    knownShelters[obj] = nil
end)

local dragging, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)
TitleBar.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local d = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
