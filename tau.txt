local function playSound(id, vol)
    pcall(function()
        local s = Instance.new("Sound")
        s.SoundId = id
        s.Volume = vol or 0.5
        s.Parent = SoundService
        s:Play()
        s.Ended:Connect(function() s:Destroy() end)
        task.delay(3, function() if s and s.Parent then s:Destroy() end end)
    end)
end

end

local function nameInList(objName, list)
    local n = (objName or ""):lower()
    for _, v in ipairs(list) do
        if n == v:lower() or n:find(v:lower(), 1, true) then return true end
    end
    return false
end

local function isFragment(obj)
    if not obj or not obj.Parent then return false end
    if not nameInList(obj.Name, FragmentNames) then return false end
    if obj.Parent and nameInList(obj.Parent.Name, FragmentNames) then return false end
    return true
end

local function isGlasses(obj)
    if not obj or not obj.Parent then return false end
    if not nameInList(obj.Name, GlassesNames) then return false end
    if obj.Parent and nameInList(obj.Parent.Name, GlassesNames) then return false end
    return true
end

local function isShelter(obj)
    return nameInList(obj.Name, ShelterNames)
end

end

local function getHumanoid()

local function getBestPart(obj)
    if obj:IsA("BasePart") then return obj end
    local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt and prompt.Parent and prompt.Parent:IsA("BasePart") then return prompt.Parent end
    return obj:FindFirstChildWhichIsA("BasePart", true)
end

local function cleanCache(cache)
    for obj in pairs(cache) do
        if not obj or not obj.Parent then cache[obj] = nil end
    end
end

local function countFrags()
    cleanCache(knownFragments)
    local c = 0
    for _ in pairs(knownFragments) do c = c + 1 end
    return c
end

local function scanMapLoot()
    local counts = {Fragment=0, Glasses=0, Key=0, Ext=0, Bat=0}
    local seen = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") or obj:IsA("Tool") or obj:IsA("BasePart") or obj:IsA("Folder") then
            local id = tostring(obj)
            if not seen[id] then
                if isFragment(obj) then
                    counts.Fragment = counts.Fragment + 1
                    knownFragments[obj] = true
                    seen[id] = true
                elseif isGlasses(obj) then
                    counts.Glasses = counts.Glasses + 1
                    knownGlasses[obj] = true
                    seen[id] = true
                elseif nameInList(obj.Name, KeyNames) then
                    if not (obj.Parent and nameInList(obj.Parent.Name, KeyNames)) then
                        counts.Key = counts.Key + 1
                        knownExtra[obj] = "Key"
                        seen[id] = true
                    end
                elseif nameInList(obj.Name, ExtNames) then
                    if not (obj.Parent and nameInList(obj.Parent.Name, ExtNames)) then
                        counts.Ext = counts.Ext + 1
                        knownExtra[obj] = "Ext"
                        seen[id] = true
                    end
                elseif nameInList(obj.Name, BatNames) then
                    if not (obj.Parent and nameInList(obj.Parent.Name, BatNames)) then
                        counts.Bat = counts.Bat + 1
                        knownExtra[obj] = "Bat"
                        seen[id] = true
                    end
                end
            end
        end
    end
    mapLootCounts = counts
    return counts
end

local function tryFirePrompt(prompt)
    if not prompt then return end
    pcall(function()
        if fireproximityprompt then
            fireproximityprompt(prompt)
            task.wait(0.03)
            fireproximityprompt(prompt, 1)
        end
    end)
end

local function pickup(obj, typ)
    if not obj or not obj.Parent then return end
    local key = tostring(obj)
    if recentlyTried[key] and tick() - recentlyTried[key] < TRIED_COOLDOWN then return end
    recentlyTried[key] = tick()
    local hrp = getHRP()
    if not hrp then return end
    local part = getBestPart(obj)
    if not part then return end
    local offsets = {
        Vector3.new(0, 2, 0), Vector3.new(0, 1.6, 1.5), Vector3.new(0, 1.6, -1.5),
        Vector3.new(1.5, 1.6, 0), Vector3.new(-1.5, 1.6, 0),
    }
    for _, offset in ipairs(offsets) do
        if not obj.Parent then break end
        hrp.CFrame = part.CFrame + offset
        task.wait(0.03)
        tryFirePrompt(obj:FindFirstChildWhichIsA("ProximityPrompt", true))
        local touch = part:FindFirstChildOfClass("TouchInterest")
        if touch then
            pcall(function()
                firetouchinterest(hrp, part, 0)
                task.wait(0.02)
                firetouchinterest(hrp, part, 1)
            end)
        end
        local click = obj:FindFirstChildWhichIsA("ClickDetector", true)
        if click then pcall(fireclickdetector, click) end
        task.wait(0.03)
    end
    task.delay(0.6, function()
        if not obj or not obj.Parent then
            if typ == "Fragment" then
                collectedFragments = math.min(5, collectedFragments + 1)
                sessionFrags = sessionFrags + 1
                Notify("碎片 " .. collectedFragments .. "/5")
            elseif typ == "Glasses" then
                glassesFound = true
                sessionGlasses = sessionGlasses + 1
                Notify("眼镜已找到！")
            end
        end
    end)
end

local function getObjectiveText()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return nil end
    local ok, text = pcall(function()
        local n = pg:FindFirstChild("Notifications")
        if not n then return nil end
        local sg = n:FindFirstChild("SelectedGamemode")
        if not sg then return nil end
        local obj = sg:FindFirstChild("Objective")
        if not obj then return nil end
        if obj:IsA("TextLabel") or obj:IsA("TextButton") then return obj.Text end
        local label = obj:FindFirstChildWhichIsA("TextLabel", true)
        return label and label.Text or nil
    end)
    if ok then return text end
    return nil
end

local function matchMode(raw)
    if not raw or #raw < 3 then return nil end
    local t = string.lower(raw)
    if t:find("unlikely the world will end") then return nil end
    local best, bestLen = nil, 0
    for _, rule in ipairs(CURRENT_RULES) do
        if t:find(rule.pat, 1, true) and #rule.pat > bestLen then
            bestLen = #rule.pat
            best = rule.mode
        end
    end
    return best
end

local function checkAlerts()
    local objText = getObjectiveText()
    local bestCurrent = matchMode(objText)
    local foundNext = nil
    local sawDoomsday, sawTartarus = false, false

    if objText and #objText >= 5 then
        local lower = string.lower(objText)
        if not lower:find("intermission") and not lower:find("voting") and not lower:find("vote") then
            markEnteredMap()
        end
        if lower:find("doomsday is occurring") or lower:find("survive the end of the world") then sawDoomsday = true end
        if lower:find("tartarus is occurring") or lower:find("tartarus has been summoned") then sawTartarus = true end
    end

    if not bestCurrent then
        local pg = LocalPlayer:FindFirstChild("PlayerGui")
        local notifications = pg and pg:FindFirstChild("Notifications")
        if notifications then
            for _, g in ipairs(notifications:GetDescendants()) do
                if g:IsA("TextLabel") or g:IsA("TextButton") then
                    local raw = g.Text or ""
                    if #raw >= 5 and #raw <= 220 then
                        local t = string.lower(raw)
                        if not t:find("unlikely the world will end") then
                            local n1, n2 = t:find("this round's gamemode is")
                            if n1 then
                                local after = raw:sub(n2 + 1):gsub("^[%s:]+", ""):gsub("%s+$", "")
                                if #after > 1 and #after < 40 then foundNext = after end
                            end
                            if not bestCurrent then
                                local m = matchMode(raw)
                                if m then bestCurrent = m end
                            end
                            if t:find("doomsday is occurring") then sawDoomsday = true end
                            if t:find("tartarus is occurring") or t:find("tartarus has been summoned") then sawTartarus = true end
                        end
                    end
                end
            end
        end
    end

    if foundNext and foundNext ~= lastNextMode then
        nextMode = foundNext
        lastNextMode = foundNext
        showBig("下一模式: " .. foundNext, Color3.fromRGB(120, 200, 255), 5)
        Notify("下一模式: " .. foundNext)
    elseif foundNext then
        nextMode = foundNext
    end

    if sawTartarus and not tartarusUnlocked then
        tartarusUnlocked = true
        flashDanger()
        showBig("⚠ TARTARUS！", Color3.fromRGB(255, 60, 60), 6)
        Notify("Tartarus 已出现！", true)
    end
    if sawDoomsday and not doomsdayDetected then
        doomsdayDetected = true
        flashDanger()
        showBig("⚠ DOOMSDAY！", Color3.fromRGB(255, 80, 40), 6)
        Notify("Doomsday！", true)
    end

    if bestCurrent then
        if bestCurrent ~= currentMode then
            if bestCurrent == "Doomsday" or bestCurrent == "Tartarus" or bestCurrent == "Deadnado" then
                flashDanger()
                showBig("⚠ 危险: " .. bestCurrent, Color3.fromRGB(255, 70, 70), 5)
            end
        end
        currentMode = bestCurrent
        lastModeTime = tick()
        if bestCurrent == "Doomsday" then doomsdayDetected = true end
        if bestCurrent == "Tartarus" then tartarusUnlocked = true end
        markEnteredMap()
    elseif tick() - lastModeTime > MODE_TIMEOUT then
        currentMode = "Unknown"
    end
end

    end
end

local function doTornadoPull()
    local hrp = getHRP()
    if not hrp then return end
    local myPos = hrp.Position
    local myChar = LocalPlayer.Character
    local parts
    local ok = pcall(function()
        parts = workspace:GetPartBoundsInRadius(myPos, TORNADO_RANGE)
    end)
    if not ok or not parts then return end
    for _, obj in ipairs(parts) do
        if obj:IsA("BasePart") and not obj.Anchored and obj.Parent then
            local skip = myChar and obj:IsDescendantOf(myChar)
            if not skip then
                local dist = (obj.Position - myPos).Magnitude
                if dist >= SAFE_DISTANCE then
                    local direction = (myPos - obj.Position).Unit
                    local sf = 1 - (dist / TORNADO_RANGE)
                    if dist < 20 then sf = sf * 0.3 end
                    local force = direction * (TORNADO_STRENGTH * sf) + Vector3.new(0, TORNADO_UP_FORCE * sf * 0.5, 0)
                    if isFragment(obj) or (obj.Parent and isFragment(obj.Parent)) then force = force * 1.45 end
                    pcall(function() obj.AssemblyLinearVelocity = force end)
                end
            end
        end
    end
end

local function doFragmentFly()
    local hrp = getHRP()
    if not hrp then return end
    local myPos = hrp.Position
    for obj in pairs(knownFragments) do
        if obj and obj.Parent then
            local part = getBestPart(obj)

                    local bv = Instance.new("BodyVelocity")

                    bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
                    bv.Velocity = Vector3.new((math.random()-0.5)*36, 35, (math.random()-0.5)*36)
                    bv.Parent = part
                    task.delay(1.2, function() if bv and bv.Parent then bv:Destroy() end end)
                end
            end
        end
    end
end

    local ok, err = pcall(function()
        if typeof(firesignal) ~= "function" then error("不支持 firesignal") end
        local event = ReplicatedStorage:FindFirstChild("Assets")
            and ReplicatedStorage.Assets:FindFirstChild("Signals")
            and ReplicatedStorage.Assets.Signals:FindFirstChild("Spectate")
        if not event then error("找不到 Spectate") end
        if isSpectating then
            pcall(function() firesignal(event.OnClientEvent, {A=false, reason="ExitSpectate"}) end)
            pcall(function() firesignal(event.OnClientEvent, {A=false}) end)
            isSpectating = false
            SpectateBtn.Text = "Open Spectate"
            Notify("已退出观战")
        else
            firesignal(event.OnClientEvent, {A=true, reason="EnterSpectate"})
            isSpectating = true
            SpectateBtn.Text = "Exit Spectate"
            Notify("已进入观战")
        end
    end)
    if not ok then
        isSpectating = false
        SpectateBtn.Text = "Open Spectate"
        Notify("观战提示: " .. tostring(err))
    end
end

local function tryChad()
    playClick()
    local ok, err = pcall(function()
        local chad = ReplicatedStorage:WaitForChild("Assets", 3)
            :WaitForChild("Signals", 3)
            :WaitForChild("Gamemodes", 3)
            :WaitForChild("Chad", 3)
        if chad:IsA("RemoteEvent") then
            chad:FireServer()
        elseif chad:IsA("Folder") then
            for _, c in ipairs(chad:GetDescendants()) do
                if c:IsA("RemoteEvent") then pcall(function() c:FireServer() end) end
            end
        end
    end)
    if ok then Notify("Chad signal 已发送") else Notify("Chad: " .. tostring(err)) end
end

task.spawn(function()
    while ScreenGui and ScreenGui.Parent do
        local ok, err = pcall(function()
            local now = tick()
            local c = mapLootCounts
            local frags = c.Fragment or countFrags()

            if Settings.FullAuto then
                StatusLabel.Text = "状态: Full Auto"
                StatusLabel.TextColor3 = Color3.fromRGB(100, 220, 140)
            elseif Settings.Enabled then
                StatusLabel.Text = "状态: 仅自动捡取"
                StatusLabel.TextColor3 = Color3.fromRGB(100, 200, 255)
            else
                StatusLabel.Text = "状态: 关闭"
                StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
            end

            FragProgressLabel.Text = string.format("血月碎片: 地图%d | 我%d/5", frags, collectedFragments)
            if glassesFound or (c.Glasses or 0) > 0 then
                GlassesProgressLabel.Text = "神秘眼镜: 已找到"
                GlassesProgressLabel.TextColor3 = Color3.fromRGB(100, 255, 120)
                ChadReadyLabel.Text = "Chad条件: 已满足"
            else
                GlassesProgressLabel.Text = "神秘眼镜: 未找到"
                GlassesProgressLabel.TextColor3 = Color3.fromRGB(255, 160, 100)
                ChadReadyLabel.Text = "Chad条件: 未满足"
            end
            TartarusLabel.Text = tartarusUnlocked and "Tartarus: 已出现" or "Tartarus: 未解锁"
            ModeLabel.Text = "当前模式: " .. currentMode
            NextModeLabel.Text = "下一模式: " .. nextMode
            DoomsdayLabel.Text = (doomsdayDetected or currentMode == "Doomsday") and "Doomsday: 是！" or "Doomsday: 否"
            RejoinStatusLabel.Text = "重进提示: " .. rejoinStatus
            local mins = math.floor((tick() - sessionStart) / 60)
            SessionLabel.Text = string.format("会话: 碎片%d 眼镜%d 换服%d 警报%d 时长%d分", sessionFrags, sessionGlasses, sessionHops, sessionDanger, mins)
            MapLootLabel.Text = string.format("地图: 钥匙%d 灭火器%d 球棒%d", c.Key or 0, c.Ext or 0, c.Bat or 0)

            if now - lastHeavyScan >= HEAVY_SCAN_INTERVAL then
                lastHeavyScan = now
                checkAlerts()
                scanMapLoot()
                updateDistList()
            end
            if now - lastEspUpdate >= ESP_INTERVAL then
                lastEspUpdate = now
                refreshESP()
            end

            if (Settings.Enabled or Settings.FullAuto) and (now - lastScan) >= SCAN_INTERVAL then
                lastScan = now
                cleanCache(knownFragments)
                cleanCache(knownGlasses)
                cleanCache(knownExtra)
                local hrp = getHRP()
                local myPos = hrp and hrp.Position or Vector3.zero
                local fragList = {}
                for obj in pairs(knownFragments) do
                    if obj and obj.Parent then
                        local part = getBestPart(obj)
                        if part then table.insert(fragList, {obj=obj, dist=(part.Position-myPos).Magnitude}) end
                    end
                end
                table.sort(fragList, function(a, b) return a.dist < b.dist end)
                for _, data in ipairs(fragList) do
                    pickup(data.obj, "Fragment")
                    task.wait(0.1)
                end
                if not Settings.OnlyFragments then
                    for obj in pairs(knownGlasses) do
                        if obj and obj.Parent then
                            pickup(obj, "Glasses")
                            task.wait(0.1)
                        end
                    end
                end
                for obj, k in pairs(knownExtra) do
                    if obj and obj.Parent then
                        local want = (k == "Key" and Settings.LootKey)
                            or (k == "Ext" and Settings.LootExt)
                            or (k == "Bat" and Settings.LootBat)
                            or (Settings.LootOther and k ~= "Key" and k ~= "Ext" and k ~= "Bat")
                        if want then
                            pickup(obj, k)
                            task.wait(0.1)
                        end
                    end
                end
            end

            if Settings.FullAuto and Settings.FullAutoShelter and (now - lastShelterTP) >= SHELTER_TP_INTERVAL then
                lastShelterTP = now
                forceToShelter(true)
            end
            if Settings.TornadoPull and (now - lastTornadoTick) >= TORNADO_INTERVAL then
                lastTornadoTick = now
                doTornadoPull()
            end
            if Settings.FragmentFly and (now - lastFlyTick) >= FLY_INTERVAL then
                lastFlyTick = now
                doFragmentFly()
            end
            checkFirstRoundNoFrag()
        end)
        if not ok then
            warn("[tau] loop error:", err)
        end
        task.wait(UI_REFRESH)
    end
end)

    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then

        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
