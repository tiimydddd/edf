local function inL(name, list)
    local n = string.lower(tostring(name or ""))
    for _, v in ipairs(list) do
        if n == string.lower(v) or string.find(n, string.lower(v), 1, true) then return true end
    end
    return false
end

local function isF(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, FragN) then return false end
    if o.Parent and inL(o.Parent.Name, FragN) then return false end
    return true
end

local function isG(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, GlassN) then return false end
    if o.Parent and inL(o.Parent.Name, GlassN) then return false end
    return true
end

local function best(o)
    if o:IsA("BasePart") then return o end
    local p = o:FindFirstChildWhichIsA("ProximityPrompt", true)
    if p and p.Parent and p.Parent:IsA("BasePart") then return p.Parent end
    return o:FindFirstChildWhichIsA("BasePart", true)
end

-- 抓取大廳 Tartarus 預測UI文字的核心邏輯
local function updateTartarusPrediction()
    pcall(function()
        local pg = LP:FindFirstChild("PlayerGui")
        if not pg then return end
        
        -- 搜尋 PlayerGui 中是否有 Tartarus 相關的文字元件
        local foundText = nil
        for _, gui in ipairs(pg:GetChildren()) do
            if gui:IsA("ScreenGui") and gui.Enabled then
                for _, descendant in ipairs(gui:GetDescendants()) do
                    if descendant:IsA("TextLabel") and descendant.Visible then
                        local txt = descendant.Text
                        if string.find(txt:lower(), "tartarus") or string.find(txt, "預測") or string.find(txt, "Prediction") then
                            foundText = txt
                            break
                        end
                    end
                end
            end
            if foundText then break end
        end
        
        if foundText then
            TartarusL.Text = "Tartarus: " .. foundText
        else
            TartarusL.Text = "Tartarus: 未找到大廳數據"
        end
    end)
end

-- 循環更新
task.spawn(function()
    while Gui and Gui.Parent do
        updateTartarusPrediction()
        task.wait(1) -- 每秒更新一次大廳數值
    end
end)
