-- ==================== Functions ====================
local function Notify(msg)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = "tau only apple",
            Text = tostring(msg),
            Duration = 2.5
        })
    end)
end

local function getHRP()
    local c = LocalPlayer.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function isFragment(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(FragmentNames) do
        if n:find(v:lower()) then return true end
    end
    return false
end

local function isGlasses(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(GlassesNames) do
        if n:find(v:lower()) then return true end
    end
    return false
end

local function isShelter(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(ShelterNames) do
        if n:find(v:lower()) then return true end
    end
    return false
end

local function isControlRoom(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(ControlRoomNames) do
        if n:find(v:lower()) then return true end
    end
    return false
end

local function matchKeywords(obj, keywords)
    local n = obj.Name:lower()
    for _, k in ipairs(keywords) do
        if n:find(k:lower()) then return true end
    end
    return false
end

local function cleanCache(cache)
    for obj in pairs(cache) do
        if not obj or not obj.Parent then
            cache[obj] = nil
        end
    end
end

local function countFrags()
    cleanCache(knownFragments)
    local c = 0
    for _ in pairs(knownFragments) do c = c + 1 end
    return c
end

local function pickup(obj, typ)
    local hrp = getHRP()
    if not hrp then return end
    local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
    if not part then return end

    hrp.CFrame = part.CFrame + Vector3.new(0, 3, 0)

    local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
    if prompt then pcall(fireproximityprompt, prompt) end

    local touch = part:FindFirstChildOfClass("TouchInterest")
    if touch then
        pcall(firetouchinterest, hrp, part, 0)
        task.wait(0.03)
        pcall(firetouchinterest, hrp, part, 1)
    end

    local click = obj:FindFirstChildOfClass("ClickDetector") or part:FindFirstChildOfClass("ClickDetector")
    if click then pcall(fireclickdetector, click) end

    if typ == "Fragment" then
        collectedFragments = collectedFragments + 1
        Notify("Fragment " .. collectedFragments .. "/5")
    elseif typ == "Glasses" then
        glassesFound = true
        Notify("Glasses found!")
    end
end

local function goToShelter()
    local hrp = getHRP()
    if not hrp then return end
    cleanCache(knownShelters)

    for obj in pairs(knownShelters) do
        if obj and obj.Parent then
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                hrp.CFrame = part.CFrame + Vector3.new(0, 5, 0)
                local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
                if prompt then pcall(fireproximityprompt, prompt) end
                Notify("Trying Shelter door...")
                return
            end
        end
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if isShelter(obj) then
            knownShelters[obj] = true
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                hrp.CFrame = part.CFrame + Vector3.new(0, 5, 0)
                local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
                if prompt then pcall(fireproximityprompt, prompt) end
                Notify("Found Shelter")
                return
            end
        end
    end
end

-- ★ 强制 TP 优先 ControlRoom
local function forceTPShelter()
    local hrp = getHRP()
    if not hrp then
        Notify("没有角色")
        return
    end

    local controlCandidates = {}
    local shelterCandidates = {}

    for _, obj in ipairs(workspace:GetDescendants()) do
        if isControlRoom(obj) then
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                table.insert(controlCandidates, part)
            end
        elseif isShelter(obj) then
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                table.insert(shelterCandidates, part)
            end
        end
    end

    local targets = #controlCandidates > 0 and controlCandidates or shelterCandidates

    if #targets == 0 then
        Notify("没找到 ControlRoom / 避难所")
        return
    end

    local offsets = {
        Vector3.new(0, 3, 0),
        Vector3.new(0, 2, 4),
        Vector3.new(0, 2, -4),
        Vector3.new(4, 2, 0),
        Vector3.new(-4, 2, 0),
        Vector3.new(0, 4, 0),
        Vector3.new(0, 1, 6),
        Vector3.new(0, 1, -6),
        Vector3.new(3, 2, 3),
        Vector3.new(-3, 2, -3),
    }

    Notify("正在强制 TP 到 ControlRoom...")

    for _, part in ipairs(targets) do
        for _, offset in ipairs(offsets) do
            hrp.CFrame = part.CFrame + offset
            task.wait(0.06)

            -- 尝试触发附近的观看/旁观 Prompt
            local parent = part.Parent
            if parent then
                for _, child in ipairs(parent:GetDescendants()) do
                    if child:IsA("ProximityPrompt") then
                        local text = (child.ActionText or ""):lower() .. (child.ObjectText or ""):lower()
                        if text:find("观看") or text:find("旁观") or text:find("interact") or text:find("camera") then
                            pcall(fireproximityprompt, child)
                        end
                    end
                end
            end
        end
    end

    Notify("ControlRoom TP 已尝试完成")
end

local function teleportToKeyword(keywords, label)
    local hrp = getHRP()
    if not hrp then
        Notify("没有角色")
        return
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if matchKeywords(obj, keywords) then
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                hrp.CFrame = part.CFrame + Vector3.new(0, 5, 0)
                Notify("已传送到 " .. label)
                return
            end
        end
    end
    Notify("没找到 " .. label)
end

local function teleportToPlayer(player)
    local hrp = getHRP()
    if not hrp then return end
    local targetChar = player.Character
    if not targetChar then
        Notify(player.Name .. " 没有角色")
        return
    end
    local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
    if targetHRP then
        hrp.CFrame = targetHRP.CFrame + Vector3.new(0, 3, 0)
        Notify("已传送到 " .. player.Name)
    end
end

            knownControlRooms[obj] = true

    task.wait(1.2)
    for _, obj in ipairs(workspace:GetDescendants()) do
        if isFragment(obj) then knownFragments[obj] = true end
        if isGlasses(obj) then knownGlasses[obj] = true end

task.spawn(function()
    while ScreenGui and ScreenGui.Parent do
        local now = tick()
        local frags = countFrags()

        StatusLabel.Text = "Status: " .. ((Settings.Enabled or Settings.FullAuto) and "Running" or "Off")
        StatusLabel.TextColor3 = (Settings.Enabled or Settings.FullAuto) and Color3.fromRGB(100, 200, 255) or Color3.fromRGB(255, 100, 100)
        FragLabel.Text = "Map Fragments: " .. frags
        CollectLabel.Text = "My Collected: " .. collectedFragments .. " / 5"
        GlassesLabel.Text = "Glasses: " .. (glassesFound and "FOUND" or "Not Found")

        if (Settings.Enabled or Settings.FullAuto) and (now - lastScan) >= SCAN_INTERVAL then
            lastScan = now
            cleanCache(knownFragments)
            cleanCache(knownGlasses)

            for obj in pairs(knownFragments) do
                if obj and obj.Parent then
                    pickup(obj, "Fragment")
                    task.wait(0.07)
                end
            end

            if not Settings.OnlyFragments then
                for obj in pairs(knownGlasses) do
                    if obj and obj.Parent then
                        pickup(obj, "Glasses")
                        task.wait(0.07)
                    end
                end
            end
        end

        if (Settings.AutoGoShelter or Settings.FullAuto) and (now - lastShelterCheck) >= SHELTER_CHECK_INTERVAL then
            lastShelterCheck = now
            goToShelter()
        end

        task.wait(0.25)
    end
end)

        dragStart = input.Position
        startPos = Main.Position
    end
end)
TitleBar.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local d = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
