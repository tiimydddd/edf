local function scan()
	local c={Fragment=0,Glasses=0,Key=0,Ext=0,Bat=0};local seen={}
	for _,o in ipairs(workspace:GetDescendants()) do
		if o:IsA("Model") or o:IsA("Tool") or o:IsA("BasePart") or o:IsA("Folder") then
			local id=tostring(o)
			if not seen[id] then
				if isF(o) then c.Fragment=c.Fragment+1;kF[o]=true;seen[id]=true
				elseif isG(o) then c.Glasses=c.Glasses+1;kG[o]=true;seen[id]=true
				elseif inL(o.Name,KeyN) and not(o.Parent and inL(o.Parent.Name,KeyN)) then c.Key=c.Key+1;kE[o]="Key";seen[id]=true
				elseif inL(o.Name,ExtN) and not(o.Parent and inL(o.Parent.Name,ExtN)) then c.Ext=c.Ext+1;kE[o]="Ext";seen[id]=true
				elseif inL(o.Name,BatN) and not(o.Parent and inL(o.Parent.Name,BatN)) then c.Bat=c.Bat+1;kE[o]="Bat";seen[id]=true
				end
			end
		end
	end
	mapC=c;return c
end

local function matchM(raw)
	if not raw or #raw<3 then return nil end
	local t=string.lower(raw);if t:find("unlikely the world will end") then return nil end
	local best,bl=nil,0
	for _,r in ipairs(RULES) do if t:find(r.p,1,true) and #r.p>bl then bl=#r.p;best=r.m end end
	return best
end

local function checkA()
	local ot=getObj();local best=matchM(ot);local fn=nil;local sd,st=false,false
	if ot and #ot>=5 then
		local lo=string.lower(ot)
		if not lo:find("intermission") and not lo:find("voting") and not lo:find("vote") then markEnter() end
		if lo:find("doomsday") or lo:find("survive the end") then sd=true end
		if lo:find("tartarus") then st=true end
	end
	local g=LP:FindFirstChild("PlayerGui")
	local n=g and g:FindFirstChild("Notifications")
	if n then
		for _,x in ipairs(n:GetDescendants()) do
			if x:IsA("TextLabel") or x:IsA("TextButton") then
				local raw=x.Text or ""
				if #raw>=5 and #raw<=220 then
					local t=string.lower(raw)
					local a,b=t:find("this round's gamemode is",1,true)
					if a then
						local after=raw:sub(b+1):gsub("^[%s:]+",""):gsub("%s+$",""):gsub("%.$","")
						if #after>1 and #after<40 then fn=after end
					end
					if not best then local m=matchM(raw);if m then best=m end end
					if t:find("doomsday") then sd=true end
					if t:find("tartarus") then st=true end
				end
			end
		end
	end
	if fn then onNext(fn) end
	if st and not tart then tart=true;flashD();showBig("⚠ TARTARUS",Color3.fromRGB(255,60,60),6);N("Tartarus!",true) end
	if sd and not doom then doom=true;flashD();showBig("⚠ DOOMSDAY",Color3.fromRGB(255,80,40),6);N("Doomsday!",true) end
	if best then
		if best~=curM then
			pushH(best)
			if best=="Doomsday" or best=="Tartarus" or best=="Deadnado" then flashD();showBig("⚠ "..best,Color3.fromRGB(255,70,70),5) end
		end
		curM=best;lastMT=tick()
		if best=="Doomsday" then doom=true end
		if best=="Tartarus" then tart=true end
		markEnter()
	elseif tick()-lastMT>16 then curM="Unknown" end
end

local drag,d0,p0
TB.InputBegan:Connect(function(i)
	if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then drag=true;d0=i.Position;p0=Main.Position end
end)
TB.InputEnded:Connect(function(i)
	if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then drag=false end
end)
UIS.InputChanged:Connect(function(i)
	if drag and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
		local d=i.Position-d0
		Main.Position=UDim2.new(p0.X.Scale,p0.X.Offset+d.X,p0.Y.Scale,p0.Y.Offset+d.Y)
	end
end)
