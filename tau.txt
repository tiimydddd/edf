local function safeTime()
    local ok, t = pcall(function() return os.date("%H:%M:%S") end)
    if ok and t then return t end
    return string.format("%.0f", tick() % 100000)
end

local function getUIParent()

    local candidates = {}
    pcall(function()
        if typeof(gethui) == "function" then
            local h = gethui()
            if h then table.insert(candidates, h) end
        end
    end)
    pcall(function()
        if CoreGui then table.insert(candidates, CoreGui) end
    end)
    pcall(function()
        local pg = LP:FindFirstChild("PlayerGui") or LP:WaitForChild("PlayerGui", 5)
        if pg then table.insert(candidates, pg) end
    end)
    return candidates
end

local function isF(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, FragN) then return false end
    -- 只取最外层碎片根，避免父子重复计数，但每个独立根都算
    if o.Parent and inL(o.Parent.Name, FragN) then return false end
    return true
end

local function isG(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, GlassN) then return false end
    if o.Parent and inL(o.Parent.Name, GlassN) then return false end
    return true
end

local function best(o)
    if o:IsA("BasePart") then return o end

end

-- 全图找碎片：有几个记几个，不漏
local function findAllFragments()
    local list = {}
    local seen = {}
    for _, o in ipairs(workspace:GetDescendants()) do
        if isF(o) then
            local root = o
            -- 同一模型只记一次
            if o:IsA("BasePart") and o.Parent and isF(o.Parent) then
                root = o.Parent
            end
            local id = tostring(root)
            if not seen[id] and root.Parent then
                seen[id] = true
                local p = best(root)
                if p then
                    table.insert(list, root)
                    kF[root] = true
                end
            end
        end
    end
    return list
end

local function classifyItem(o)
    if isF(o) then return "Fragment" end
    if isG(o) then return "Glasses" end
    if inL(o.Name, KeyN) and not (o.Parent and inL(o.Parent.Name, KeyN)) then return "Key" end
    if inL(o.Name, ExtN) and not (o.Parent and inL(o.Parent.Name, ExtN)) then return "Ext" end
    if inL(o.Name, BatN) and not (o.Parent and inL(o.Parent.Name, BatN)) then return "Bat" end
    if inL(o.Name, MedN) and not (o.Parent and inL(o.Parent.Name, MedN)) then return "Med" end
    if inL(o.Name, FoodN) and not (o.Parent and inL(o.Parent.Name, FoodN)) then return "Food" end
    if inL(o.Name, LightN) and not (o.Parent and inL(o.Parent.Name, LightN)) then return "Light" end
    if inL(o.Name, MiscN) and not (o.Parent and inL(o.Parent.Name, MiscN)) then return "Misc" end
    if o:FindFirstChildWhichIsA("ProximityPrompt", true) then
        local n = (o.Name or ""):lower()
        if n:find("loot") or n:find("item") or n:find("pickup") or n:find("collect") then
            return "Other"
        end
    end
    return nil
end

local function scan()
    local c = {Fragment = 0, Glasses = 0, Key = 0, Ext = 0, Bat = 0, Med = 0, Food = 0, Light = 0, Misc = 0, Other = 0}
    local seen = {}
    kMap = {}
    -- 重建碎片表，保证数量准确
    for o in pairs(kF) do kF[o] = nil end
    for o in pairs(kG) do kG[o] = nil end
    for o in pairs(kE) do kE[o] = nil end
    local frags = findAllFragments()
    c.Fragment = #frags
    for _, o in ipairs(frags) do
        kMap[o] = "Fragment"
        seen[tostring(o)] = true
    end
    for _, o in ipairs(workspace:GetDescendants()) do
        if o:IsA("Model") or o:IsA("Tool") or o:IsA("BasePart") or o:IsA("Folder") then
            local id = tostring(o)
            if not seen[id] then
                local typ = classifyItem(o)
                if typ and typ ~= "Fragment" then
                    c[typ] = (c[typ] or 0) + 1
                    kMap[o] = typ
                    if typ == "Glasses" then kG[o] = true
                    elseif typ == "Key" or typ == "Ext" or typ == "Bat" then kE[o] = typ
                    end
                    seen[id] = true
                end
            end
        end
    end
    mapC = c
    return c
end

local function fireP(pr)
    if not pr then return end
    pcall(function()
        if typeof(fireproximityprompt) == "function" then
            fireproximityprompt(pr)
            task.wait(0.02)
            fireproximityprompt(pr, 1)
            task.wait(0.02)
            fireproximityprompt(pr)
        end
    end)
end

local function pickup(o, typ)
    if not o or not o.Parent then return false end
    local key = tostring(o)
    -- 碎片冷却更短，方便连续捡多个
    local cd = (typ == "Fragment") and 0.55 or 1.8
    if tried[key] and tick() - tried[key] < cd then return false end
    tried[key] = tick()
    local h = HRP()
    if not h then return false end
    local part = best(o)
    if not part then return false end
    pcall(function()
        part.CanCollide = true
        if o:IsA("Model") then
            for _, bp in ipairs(o:GetDescendants()) do
                if bp:IsA("BasePart") then bp.CanCollide = true end
            end
        end
    end)
    local offs = {
        Vector3.new(0, 2.2, 0), Vector3.new(0, 1.5, 0),
        Vector3.new(0, 1.6, 1.8), Vector3.new(0, 1.6, -1.8),
        Vector3.new(1.8, 1.6, 0), Vector3.new(-1.8, 1.6, 0),
        Vector3.new(1.2, 1.4, 1.2), Vector3.new(-1.2, 1.4, -1.2),
    }
    for _, off in ipairs(offs) do
        if not o.Parent then break end
        pcall(function() h.CFrame = part.CFrame + off end)
        task.wait(0.025)
        fireP(o:FindFirstChildWhichIsA("ProximityPrompt", true))
        -- 对模型内所有 BasePart 尝试 touch
        local parts = {}
        if o:IsA("BasePart") then
            table.insert(parts, o)
        else
            for _, bp in ipairs(o:GetDescendants()) do
                if bp:IsA("BasePart") then table.insert(parts, bp) end
            end
            if #parts == 0 and part then table.insert(parts, part) end
        end
        for _, bp in ipairs(parts) do
            pcall(function()
                if typeof(firetouchinterest) == "function" then
                    firetouchinterest(h, bp, 0)
                    task.wait(0.015)
                    firetouchinterest(h, bp, 1)
                end
            end)
        end
        local cd2 = o:FindFirstChildWhichIsA("ClickDetector", true)
        if cd2 and typeof(fireclickdetector) == "function" then
            pcall(fireclickdetector, cd2)
        end
        task.wait(0.02)
    end
    task.delay(0.5, function()
        if not o or not o.Parent then
            if typ == "Fragment" then
                myF = math.min(5, myF + 1)
                sF = sF + 1
                if myF >= 5 then
                    alrt()
                    showBig("碎片个人已满 5/5！", Color3.fromRGB(100, 255, 120), 7)
                    N("个人碎片 5/5！", true)
                else
                    N("碎片 " .. myF .. "/5")
                end
            elseif typ == "Glasses" then
                gotG = true
                sG = sG + 1
                N("眼镜已找到")
            end
        end
    end)
    return true
end

local function queueIndex(id)
    for i, v in ipairs(modeQueue) do
        if v == id then return i end
    end
    return nil
end

local function queueAdd(id)
    if queueIndex(id) then return end
    table.insert(modeQueue, id) -- 加到队尾
end

local function queueRemove(id)
    local i = queueIndex(id)
    if i then table.remove(modeQueue, i) end
end

local function queueMove(id, dir)
    -- dir: -1 上移(更优先), +1 下移
    local i = queueIndex(id)
    if not i then return end
    local j = i + dir
    if j < 1 or j > #modeQueue then return end
    modeQueue[i], modeQueue[j] = modeQueue[j], modeQueue[i]
end

local function refreshModeQStatus()
    if #modeQueue == 0 then
        ModeQStatus.Text = "队列: (空)  点模式加入 | 再点移除"

    else
        local parts = {}
        for i, id in ipairs(modeQueue) do

        end
        ModeQStatus.Text = "排列: " .. table.concat(parts, " → ")
        ModeQStatus.TextColor3 = Color3.fromRGB(100, 220, 140)
    end
end

local function isModeInQueue(mode)
    if not mode then return false end
    local lo = tostring(mode):lower()
    for _, id in ipairs(modeQueue) do
        if lo == id:lower() or lo:find(id:lower(), 1, true) then
            return true
        end
    end
    return false
end

end

local function changeGameMusic(id)
    local success = false
    pcall(function()
        id = tostring(id or "")
        if id == "" or id == "0" then return end
        if not string.find(id, "rbxassetid://") then id = "rbxassetid://" .. id end
        local targets = {}
        local musicFolder = SS:FindFirstChild("Music")
        if musicFolder then
            local gamemode = musicFolder:FindFirstChild("Gamemode")
            local bonus = musicFolder:FindFirstChild("BonusDisaster")
            if gamemode and gamemode:IsA("Sound") then table.insert(targets, gamemode) end
            if bonus and bonus:IsA("Sound") then table.insert(targets, bonus) end
        end
        if #targets == 0 then return end
        for _, s in ipairs(targets) do
            if not originalIds[s] then originalIds[s] = s.SoundId end
            s.SoundId = id
        end
        N("已更换音乐")
        success = true
    end)
    return success
end

    end)
end

local function addToHistory(name, id)
    name = tostring(name or ""):gsub("^%s*(.-)%s*$", "%1")
    id = tostring(id or "")
    if name == "" then name = "未命名" end
    if id == "" then return end
    for i, item in ipairs(musicHistory) do
        if item.id == id or item.id == "rbxassetid://" .. id then
            table.remove(musicHistory, i)
            break
        end
    end
    table.insert(musicHistory, 1, {name = name, id = id})
    while #musicHistory > MAX_HISTORY do table.remove(musicHistory) end
    refreshMusicList()
end

local function enableNoInjury()
    if injuryConn then injuryConn:Disconnect() end
    injuryConn = RS.Heartbeat:Connect(function()
        if not C.NoInjury then return end
        local char = LP.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hum or not hrp then return end
        pcall(function()
            local state = hum:GetState()
            if state == Enum.HumanoidStateType.FallingDown
            or state == Enum.HumanoidStateType.Ragdoll
            or state == Enum.HumanoidStateType.Physics
            or state == Enum.HumanoidStateType.Freefall then
                if hrp.AssemblyLinearVelocity.Y < -100 then
                    hrp.AssemblyLinearVelocity = Vector3.new(
                        hrp.AssemblyLinearVelocity.X * 0.3,
                        math.max(hrp.AssemblyLinearVelocity.Y, -80),
                        hrp.AssemblyLinearVelocity.Z * 0.3
                    )
                end
                hum:ChangeState(Enum.HumanoidStateType.GettingUp)
                hum:ChangeState(Enum.HumanoidStateType.Running)
            end
            local bones = char:FindFirstChild("Bones")
            if bones then
                for _, bone in ipairs(bones:GetChildren()) do
                    if bone:IsA("BoolValue") and bone.Value == true then
                        bone.Value = false
                    end
                end
            end
            local gui = LP:FindFirstChild("PlayerGui")
            if gui then
                local menus = gui:FindFirstChild("Menus")
                if menus then
                    local broken = menus:FindFirstChild("BrokenBones")
                    if broken and broken:FindFirstChild("Body") then
                        for _, part in ipairs(broken.Body:GetChildren()) do
                            if part:FindFirstChild("Broken") then
                                part.Broken.Value = false
                                part.BackgroundColor3 = Color3.fromRGB(85, 255, 127)
                            end
                        end
                    end
                end
            end
            if hum.WalkSpeed < 16 and not C.SpdOn then hum.WalkSpeed = 16 end
            hum.PlatformStand = false
            hum.Sit = false
        end)
    end)
end

local function doTorn()
    local h = HRP()
    if not h then return end
    local pos = h.Position
    local ch = LP.Character
    local parts
    local ok = pcall(function() parts = workspace:GetPartBoundsInRadius(pos, TRANGE) end)
    if not ok or not parts then return end
    for _, o in ipairs(parts) do
        if o:IsA("BasePart") and not o.Anchored and o.Parent then
            if not (ch and o:IsDescendantOf(ch)) then
                local d = (o.Position - pos).Magnitude
                if d >= 9 then
                    local dir = (pos - o.Position).Unit
                    local sf = 1 - (d / TRANGE)
                    if d < 20 then sf = sf * 0.3 end
                    local f = dir * (110 * sf) + Vector3.new(0, 22 * sf * 0.5, 0)
                    if isF(o) or (o.Parent and isF(o.Parent)) then
                        f = f * 1.45
                        pcall(function() o.CanCollide = true end)
                    end
                    pcall(function() o.AssemblyLinearVelocity = f end)
                end
            end
        end
    end
end

local function doFly()
    local h = HRP()
    if not h then return end
    local pos = h.Position
    for o in pairs(kF) do
        if o and o.Parent then
            local p = best(o)
            if p and not p.Anchored and (p.Position - pos).Magnitude <= 60 then
                -- 飞碎片时打开 CanCollide，方便碰到/捡取
                pcall(function()
                    p.CanCollide = true
                    if o:IsA("Model") then
                        for _, part in ipairs(o:GetDescendants()) do
                            if part:IsA("BasePart") then
                                part.CanCollide = true
                            end
                        end
                    end
                end)
                local old = p:FindFirstChild("FragFlyForce")
                if old then old:Destroy() end
                local bv = Instance.new("BodyVelocity")
                bv.Name = "FragFlyForce"
                bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
                bv.Velocity = Vector3.new((math.random() - 0.5) * 36, 35, (math.random() - 0.5) * 36)
                bv.Parent = p
                task.delay(1.2, function() if bv and bv.Parent then bv:Destroy() end end)
            end
        end
    end
end

local SAVE_FILE = "tau_only_apple_save.json"
local HS = game:GetService("HttpService")

local function buildSaveData()
    local q = {}
    for _, id in ipairs(modeQueue) do
        table.insert(q, id)
    end
    return {
        C = {
            En = C.En, OnlyF = C.OnlyF, FA = C.FA, FAShel = C.FAShel, AFK = C.AFK,
            Torn = C.Torn, God = C.God, Fly = C.Fly, Rej = C.Rej, ESP = C.ESP,
            SpdOn = C.SpdOn, Spd = C.Spd, Key = C.Key, Ext = C.Ext, Bat = C.Bat,
            NoInjury = C.NoInjury, BoatFly = C.BoatFly, Noclip = C.Noclip,
            ModeQ = C.ModeQ, ModeHop = C.ModeHop,
        },
        TRANGE = TRANGE,
        modeQueue = q, -- 有序：1 = 最高优先
        musicHistory = musicHistory,
    }
end

local function saveConfig()
    local ok, err = pcall(function()
        local data = buildSaveData()
        local json = HS:JSONEncode(data)
        if typeof(writefile) == "function" then
            writefile(SAVE_FILE, json)
        else
            error("无 writefile（执行器不支持）")
        end
    end)
    if ok then
        if SaveStatusL then
            SaveStatusL.Text = "储存: 已保存 " .. safeTime()
            SaveStatusL.TextColor3 = Color3.fromRGB(100, 220, 140)
        end
        N("设置已保存")
        return true
    else
        if SaveStatusL then
            SaveStatusL.Text = "储存失败: " .. tostring(err)
            SaveStatusL.TextColor3 = Color3.fromRGB(255, 120, 100)
        end
        N("保存失败: " .. tostring(err))
        return false
    end
end

local function loadConfig(silent)
    local ok, err = pcall(function()
        if typeof(readfile) ~= "function" then
            error("无 readfile（执行器不支持）")
        end
        if typeof(isfile) == "function" and not isfile(SAVE_FILE) then
            error("无存档文件")
        end
        local raw = readfile(SAVE_FILE)
        if not raw or #raw < 2 then error("文件空") end
        local data = HS:JSONDecode(raw)
        if typeof(data) ~= "table" then error("格式错误") end
        if typeof(data.C) == "table" then
            for k, v in pairs(data.C) do
                if C[k] ~= nil then C[k] = v end
            end
        end
        if typeof(data.TRANGE) == "number" then
            TRANGE = math.clamp(data.TRANGE, 20, 200)
        end
        modeQueue = {}
        if typeof(data.modeQueue) == "table" then
            for _, id in ipairs(data.modeQueue) do
                if typeof(id) == "string" and not queueIndex(id) then
                    table.insert(modeQueue, id)
                end
            end
        end
        if typeof(data.musicHistory) == "table" then
            musicHistory = {}
            for _, item in ipairs(data.musicHistory) do
                if typeof(item) == "table" and item.id then
                    table.insert(musicHistory, {
                        name = tostring(item.name or "未命名"),
                        id = tostring(item.id)
                    })
                end
            end
        end
    end)
    if ok then
        if SaveStatusL then
            SaveStatusL.Text = "储存: 已读取 " .. safeTime()
            SaveStatusL.TextColor3 = Color3.fromRGB(100, 200, 255)
        end
        if not silent then N("已读取储存") end
        return true
    else
        if SaveStatusL then
            SaveStatusL.Text = "读取失败/无存档: " .. tostring(err)
            SaveStatusL.TextColor3 = Color3.fromRGB(255, 160, 100)
        end
        if not silent then N("读取失败: " .. tostring(err)) end
        return false
    end
end

                pcall(updModeVIP)
            end
            pcall(checkModeQueue)

            if now - tHeavy >= 2.5 then
                tHeavy = now
                scan()
                updDist()
                updMapItems()
            end
            if now - tEsp >= 0.5 then
                tEsp = now
                refESP()
            end
            if (C.En or C.FA) and now - tScan >= 0.35 then
                tScan = now
                -- 每次先全图重扫，有几个碎片就记几个
                local allFrags = findAllFragments()
                local h = HRP()
                local pos = h and h.Position or Vector3.zero
                local fl = {}
                for _, o in ipairs(allFrags) do
                    if o and o.Parent then
                        local p = best(o)
                        if p then
                            table.insert(fl, {o = o, d = (p.Position - pos).Magnitude})
                        end
                    end
                end
                table.sort(fl, function(a, b) return a.d < b.d end)
                -- 第一遍：按距离全捡，一个都不漏
                for _, d in ipairs(fl) do
                    if d.o and d.o.Parent then
                        pickup(d.o, "Fragment")
                        task.wait(0.06)
                    end
                end
                -- 第二遍：还在的再补捡一次（防漏）
                task.wait(0.08)
                local remain = findAllFragments()
                if #remain > 0 then
                    for _, o in ipairs(remain) do

                            -- 清冷却强制再捡
                            tried[tostring(o)] = 0
                            pickup(o, "Fragment")
                            task.wait(0.06)
                        end
                    end
                end
                if not C.OnlyF then
                    clean(kG)
                    for o in pairs(kG) do
                        if o and o.Parent then
                            pickup(o, "Glasses")
                            task.wait(0.06)
                        end
                    end
                end
                clean(kE)
                for o, k in pairs(kE) do
                    if o and o.Parent then
                        local w = (k == "Key" and C.Key) or (k == "Ext" and C.Ext) or (k == "Bat" and C.Bat)
                        if w then
                            pickup(o, k)
                            task.wait(0.06)
                        end
                    end
                end
            end
            if C.FA and C.FAShel and now - tShel >= 28 then
                tShel = now
                toShel(true)
            end
            if C.Torn and now - tTorn >= 0.35 then
                tTorn = now
                doTorn()
            end
            if C.Fly and now - tFly >= 1 then
                tFly = now
                doFly()
            end
            if C.BoatFly and now - tBoat >= 0.8 then
                tBoat = now
                doBoatFly()
            end
            if C.Noclip then
                applyNoclip()
            end
            checkFirst()
        end)
        task.wait(0.12)
    end
end)
