local function inL(name, list)
    local n = (name or ""):lower()
    for _, v in ipairs(list) do
        if n == v:lower() or n:find(v:lower(), 1, true) then return true end
    end
    return false
end

local function isF(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, FragN) then return false end
    if o.Parent and inL(o.Parent.Name, FragN) then return false end
    return true
end

local function isDebrisPart(p)
    if not p or not p:IsA("BasePart") then return false end
    if p.Anchored then return false end
    if p.AssemblyMass < 0.5 then return false end
    local char = LP.Character
    if char and p:IsDescendantOf(char) then return false end
    if isF(p) or (p.Parent and isF(p.Parent)) then return false end
    if isG(p) or (p.Parent and isG(p.Parent)) then return false end
    return true
end

local function scan()
    local c = {Fragment = 0, Glasses = 0, Key = 0, Ext = 0, Bat = 0}
    local seen = {}
    for _, o in ipairs(workspace:GetDescendants()) do
        if o:IsA("Model") or o:IsA("Tool") or o:IsA("BasePart") or o:IsA("Folder") then
            local id = tostring(o)
            if not seen[id] then
                if isF(o) then
                    c.Fragment = c.Fragment + 1
                    kF[o] = true
                    seen[id] = true
                elseif isG(o) then
                    c.Glasses = c.Glasses + 1
                    kG[o] = true
                    seen[id] = true
                elseif inL(o.Name, KeyN) and not (o.Parent and inL(o.Parent.Name, KeyN)) then
                    c.Key = c.Key + 1
                    kE[o] = "Key"
                    seen[id] = true
                elseif inL(o.Name, ExtN) and not (o.Parent and inL(o.Parent.Name, ExtN)) then
                    c.Ext = c.Ext + 1
                    kE[o] = "Ext"
                    seen[id] = true
                elseif inL(o.Name, BatN) and not (o.Parent and inL(o.Parent.Name, BatN)) then
                    c.Bat = c.Bat + 1
                    kE[o] = "Bat"
                    seen[id] = true
                end
            end
        end
    end
    mapC = c
    return c
end

local function pickup(o, typ)
    if not o or not o.Parent then return end
    local key = tostring(o)
    if tried[key] and tick() - tried[key] < 1.8 then return end
    tried[key] = tick()
    local h = HRP()
    if not h then return end
    local part = best(o)
    if not part then return end
    local offs = {
        Vector3.new(0, 2, 0), Vector3.new(0, 1.6, 1.5), Vector3.new(1.5, 1.6, 0)
    }
    for _, off in ipairs(offs) do
        if not o.Parent then break end
        h.CFrame = part.CFrame + off
        task.wait(0.02)
        fireP(o:FindFirstChildWhichIsA("ProximityPrompt", true))
        local ti = part:FindFirstChildOfClass("TouchInterest")
        if ti then
            pcall(function()
                firetouchinterest(h, part, 0)
                task.wait(0.015)
                firetouchinterest(h, part, 1)
            end)
        end
        local cd = o:FindFirstChildWhichIsA("ClickDetector", true)
        if cd then pcall(fireclickdetector, cd) end
        task.wait(0.02)
    end
    task.delay(0.5, function()
        if not o or not o.Parent then
            if typ == "Fragment" then
                myF = math.min(5, myF + 1)
                sF = sF + 1
                if myF >= 5 then
                    alrt()
                    showBig("Fragments 5/5", Color3.fromRGB(100, 255, 120), 6)
                    N("Fragments complete (5/5)", true)
                else
                    N("Fragment " .. myF .. "/5")
                end
            elseif typ == "Glasses" then
                gotG = true
                sG = sG + 1
                N("Item found")
            end
        end
    end)
end

-- ★ E: 持续黑洞（跟着你）
local function doBlackHoleContinuous()
    local hrp = HRP()
    if not hrp then return end
    local center = hrp.Position + Vector3.new(0, 2, 0)
    local parts
    pcall(function()
        parts = workspace:GetPartBoundsInRadius(center, BH_RANGE)
    end)
    if not parts then return end
    local n = 0
    for _, p in ipairs(parts) do
        if n >= BH_MAX then break end
        if isDebrisPart(p) then
            pcall(function()
                p.Anchored = false
                local offset = p.Position - center
                local dist = math.max(offset.Magnitude, 2)
                local inward = -offset.Unit
                local tangent = Vector3.new(-offset.Z, 0, offset.X)
                if tangent.Magnitude > 0.1 then
                    tangent = tangent.Unit
                else
                    tangent = Vector3.new(1, 0, 0)
                end
                local pull = math.clamp(100 / dist, 10, 60)
                local spin = math.clamp(45 / math.sqrt(dist), 8, 40)
                local up = (dist < 20) and 28 or 8
                p.AssemblyLinearVelocity = inward * pull + tangent * spin + Vector3.new(0, up, 0)
                p.AssemblyAngularVelocity = Vector3.new(
                    (math.random() - 0.5) * 10,
                    (math.random() - 0.5) * 14,
                    (math.random() - 0.5) * 10
                )
            end)
            n = n + 1
        end
    end
end

-- Debris Fly：冲高再掉
local function doFly()
    local list = {}
    for _, p in ipairs(workspace:GetDescendants()) do
        if p:IsA("BasePart") and isDebrisPart(p) then
            table.insert(list, p)
        end
    end
    local total = #list
    if total == 0 then return end
    if debrisCursor > total then debrisCursor = 1 end
    local n, i = 0, debrisCursor
    while n < FLY_MAX_PARTS and n < total do
        local p = list[i]
        if p and p.Parent then
            pcall(function()
                p.Anchored = false
                local up = 140 + math.random() * 80
                local sx = (math.random() - 0.5) * 40
                local sz = (math.random() - 0.5) * 40
                p.AssemblyLinearVelocity = Vector3.new(sx, up, sz)
                local old = p:FindFirstChild("DebrisFly")
                if old then old:Destroy() end
                local bv = Instance.new("BodyVelocity")
                bv.Name = "DebrisFly"
                bv.MaxForce = Vector3.new(4e5, 4e5, 4e5)
                bv.Velocity = Vector3.new(sx * 0.5, up, sz * 0.5)
                bv.Parent = p
                task.delay(0.45, function()
                    if bv and bv.Parent then bv:Destroy() end
                end)
            end)
            n = n + 1
        end
        i = i + 1
        if i > total then i = 1 end
        if i == debrisCursor and n > 0 then break end
    end
    debrisCursor = i
end

                or name:find("帆船") or name:find("游艇") then
                    boatCache[model] = true
                end
            end
        end
    end

local function doBoatFly()
    local hrp = HRP()
    if not hrp then return end
    if tick() - tBoatScan >= 5 then
        tBoatScan = tick()
        refreshBoatCache()
    end
    for model in pairs(boatCache) do

                            bv.MaxForce = Vector3.new(2e5, 5e5, 2e5)

        end
    end

local function offGod()
    if gC then gC:Disconnect() gC = nil end
    if hC then hC:Disconnect() hC = nil end
    if fC then fC:Disconnect() fC = nil end
end

local function onGod()
    offGod()
    gC = RS.Heartbeat:Connect(function() if C.God then forceHP() end end)
    local ch = LP.Character
    if ch then
        local h = ch:FindFirstChildOfClass("Humanoid")
        if h then
            hC = h.HealthChanged:Connect(function() if C.God then forceHP() end end)
        end
    end
    fC = RS.Heartbeat:Connect(function()
        if not C.God then return end
        local hr = HRP()
        if hr and hr.AssemblyLinearVelocity.Magnitude > 80 then
            hr.AssemblyLinearVelocity = Vector3.zero
            hr.AssemblyAngularVelocity = Vector3.zero
        end
    end)
end

local function applySpd()
    local h = Hum()
    if h then h.WalkSpeed = C.SpdOn and C.Spd or 16 end
end

workspace.DescendantAdded:Connect(function(o)
    task.defer(function()
        if isF(o) then
            kF[o] = true
            if C.En or C.FA then task.wait(0.12) pickup(o, "Fragment") end
        elseif isG(o) then
            kG[o] = true
            if (C.En or C.FA) and not C.OnlyF then task.wait(0.12) pickup(o, "Glasses") end
        elseif inL(o.Name, KeyN) then kE[o] = "Key"
        elseif inL(o.Name, ExtN) then kE[o] = "Ext"
        elseif inL(o.Name, BatN) then kE[o] = "Bat"
        end
    end)
end)

local function startSpec()
    click()
    local ok, err = pcall(function()
        if typeof(firesignal) ~= "function" then error("firesignal missing") end
        local e = RPS:FindFirstChild("Assets") and RPS.Assets:FindFirstChild("Signals") and RPS.Assets.Signals:FindFirstChild("Spectate")
        if not e then error("Spectate missing") end
        if isSpec then
            pcall(function() firesignal(e.OnClientEvent, {A = false, reason = "ExitSpectate"}) end)
            pcall(function() firesignal(e.OnClientEvent, {A = false}) end)
            isSpec = false
            SpecB.Text = "Spectate"
            N("Spectate off")
        else
            firesignal(e.OnClientEvent, {A = true, reason = "EnterSpectate"})
            isSpec = true
            SpecB.Text = "Exit Spectate"
            N("Spectate on")
        end
    end)
    if not ok then
        isSpec = false
        SpecB.Text = "Spectate"
        N("Spectate: " .. tostring(err))
    end
end

task.spawn(function()
    while Gui and Gui.Parent do
        pcall(function()
            local now = tick()
            local c = mapC
            local fr = c.Fragment or cntF()
            if C.FA then
                StatusL.Text = "Status: Full Auto"
                StatusL.TextColor3 = Color3.fromRGB(100, 220, 140)
            elseif C.En then
                StatusL.Text = "Status: Auto Loot"
                StatusL.TextColor3 = Color3.fromRGB(100, 200, 255)
            else
                StatusL.Text = "Status: Off"
                StatusL.TextColor3 = Color3.fromRGB(255, 100, 100)
            end
            FragL.Text = string.format("Fragments: map %d | me %d/5", fr, myF)
            if myF >= 5 then
                FragFullL.Text = "Personal: 5/5 done"
                FragFullL.TextColor3 = Color3.fromRGB(100, 255, 120)
            else
                FragFullL.Text = "Personal: " .. myF .. "/5"
                FragFullL.TextColor3 = Color3.fromRGB(180, 180, 195)
            end
            if gotG or (c.Glasses or 0) > 0 then
                GlassL.Text = "Glasses: found"
                GlassL.TextColor3 = Color3.fromRGB(100, 255, 120)
            else
                GlassL.Text = "Glasses: not found"
                GlassL.TextColor3 = Color3.fromRGB(255, 160, 100)
            end
            ModeL.Text = "Mode: " .. currentMode
            TartL.Text = (tart or tartarusActive) and "Tartarus: active" or "Tartarus: locked"
            RejL.Text = "Hint: " .. rejSt
            SessL.Text = string.format("Session: F%d G%d Hop%d", sF, sG, sH)
            MapL.Text = string.format("Map: Key%d Ext%d Bat%d", c.Key or 0, c.Ext or 0, c.Bat or 0)

            if now - tHeavy >= 5 then
                tHeavy = now
                scan()
                updDist()
            end
            if now - tEsp >= 1.5 then
                tEsp = now
                refESP()
            end
            if (C.En or C.FA) and now - tScan >= 0.55 then
                tScan = now
                clean(kF)
                clean(kG)
                clean(kE)
                local h = HRP()
                local pos = h and h.Position or Vector3.zero
                local fl = {}
                for o in pairs(kF) do
                    if o and o.Parent then
                        local p = best(o)
                        if p then table.insert(fl, {o = o, d = (p.Position - pos).Magnitude}) end
                    end
                end
                table.sort(fl, function(a, b) return a.d < b.d end)
                local picked = 0
                for _, d in ipairs(fl) do
                    if picked >= MAX_PICK_PER_TICK then break end
                    pickup(d.o, "Fragment")
                    picked = picked + 1
                    task.wait(0.08)
                end
                if not C.OnlyF then
                    for o in pairs(kG) do
                        if picked >= MAX_PICK_PER_TICK + 1 then break end
                        if o and o.Parent then
                            pickup(o, "Glasses")
                            picked = picked + 1
                            task.wait(0.08)
                        end
                    end
                end
                for o, k in pairs(kE) do
                    if picked >= MAX_PICK_PER_TICK + 2 then break end
                    if o and o.Parent then
                        local w = (k == "Key" and C.Key) or (k == "Ext" and C.Ext) or (k == "Bat" and C.Bat)
                        if w then
                            pickup(o, k)
                            picked = picked + 1
                            task.wait(0.08)
                        end
                    end
                end
            end
            if C.FA and C.FAShel and now - tShel >= 28 then
                tShel = now
                toShel(true)
            end
            if C.BlackHole and now - tBH >= BH_TICK then
                tBH = now
                doBlackHoleContinuous()
            end
            if C.Fly and now - tFly >= FLY_INTERVAL then
                tFly = now
                doFly()
            end
            if C.BoatFly and now - tBoat >= 1.2 then
                tBoat = now
                doBoatFly()
            end
            checkFirst()
        end)
        task.wait(0.28)
    end
end)

local drag, d0, p0
TB.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        drag = true
        d0 = i.Position
        p0 = Main.Position
    end
end)
TB.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        drag = false
    end
end)
UIS.InputChanged:Connect(function(i)
    if drag and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - d0
        Main.Position = UDim2.new(p0.X.Scale, p0.X.Offset + d.X, p0.Y.Scale, p0.Y.Offset + d.Y)
    end
end)
