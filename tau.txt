-- tau UI 最小测试 - 若这个都没窗口，是执行器问题不是主脚本问题
print("[tau-test] start")

if not game:IsLoaded() then game.Loaded:Wait() end
local LP = game:GetService("Players").LocalPlayer
if not LP then
    LP = game:GetService("Players").PlayerAdded:Wait()
end
print("[tau-test] player", LP and LP.Name)

local Gui = Instance.new("ScreenGui")
Gui.Name = "tau_test"
Gui.ResetOnSpawn = false
Gui.DisplayOrder = 999999
Gui.IgnoreGuiInset = true

pcall(function()
    if typeof(gethui) == "function" then
        Gui.Parent = gethui()
    end
end)
if not Gui.Parent then
    pcall(function() Gui.Parent = game:GetService("CoreGui") end)
end
if not Gui.Parent then
    pcall(function() Gui.Parent = LP:WaitForChild("PlayerGui", 5) end)
end

print("[tau-test] parent", tostring(Gui.Parent))

local f = Instance.new("Frame")
f.Size = UDim2.new(0, 300, 0, 120)
f.Position = UDim2.new(0.5, -150, 0.5, -60)
f.BackgroundColor3 = Color3.fromRGB(20, 120, 60)
f.Parent = Gui
Instance.new("UICorner", f).CornerRadius = UDim.new(0, 10)

local t = Instance.new("TextLabel")
t.Size = UDim2.new(1, 0, 1, 0)
t.BackgroundTransparency = 1
t.Text = "UI TEST OK\n若看到此窗=执行器可显示GUI"
t.TextColor3 = Color3.new(1, 1, 1)
t.Font = Enum.Font.GothamBold
t.TextSize = 16
t.Parent = f

pcall(function()
    game:GetService("StarterGui"):SetCore("SendNotification", {
        Title = "tau-test",
        Text = "UI test ran, parent=" .. tostring(Gui.Parent),
        Duration = 5
    })
end)

print("[tau-test] done")
