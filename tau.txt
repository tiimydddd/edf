-- ==================== Functions ====================
local function Notify(msg)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = "TAU",
            Text = tostring(msg),
            Duration = 3
        })
    end)
    print("[TAU]", msg)
end

local function getHRP()
    local c = LocalPlayer.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function isTarget(obj)
    local n = obj.Name:lower()
    for _, v in ipairs(FragmentNames) do
        if n:find(v:lower()) then return "Fragment" end
    end
    if not Settings.OnlyFragments then
        for _, v in ipairs(GlassesNames) do
            if n:find(v:lower()) then return "Glasses" end
        end
    end
    if Settings.AutoShelterKey then
        for _, v in ipairs(ShelterKeyNames) do
            if n:find(v:lower()) then return "Key" end
        end
    end
    return nil
end

local function countFrags()
    local c = 0
    for _, o in ipairs(workspace:GetDescendants()) do
        local n = o.Name:lower()
        for _, v in ipairs(FragmentNames) do
            if n:find(v:lower()) then c = c + 1 break end
        end
    end
    return c
end

local function playSound()
    if not Settings.SpawnSound then return end
    local s = Instance.new("Sound")
    s.SoundId = "rbxassetid://6026984224"
    s.Volume = 1.2
    s.Parent = SoundService
    s:Play()
    task.delay(1.5, function() s:Destroy() end)
end

local function pickup(obj, typ)
    local hrp = getHRP()
    if not hrp then return end
    local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
    if not part then return end

    hrp.CFrame = part.CFrame + Vector3.new(0, 3, 0)

    local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or part:FindFirstChildOfClass("ProximityPrompt")
    if prompt then pcall(fireproximityprompt, prompt) end

    local touch = part:FindFirstChildOfClass("TouchInterest")
    if touch then
        pcall(firetouchinterest, hrp, part, 0)
        task.wait(0.03)
        pcall(firetouchinterest, hrp, part, 1)
    end

    local click = obj:FindFirstChildOfClass("ClickDetector") or part:FindFirstChildOfClass("ClickDetector")
    if click then pcall(fireclickdetector, click) end

    if typ == "Fragment" then
        collectedFragments = collectedFragments + 1
        Notify("Fragment! " .. collectedFragments .. "/5")
        playSound()
    elseif typ == "Glasses" then
        glassesFound = true
        Notify("Glasses found!")
        playSound()
    end
end

local function clearESP()
    if espFolder then
        espFolder:Destroy()
        espFolder = nil
    end
end

local function updateESP()
    clearESP()
    if not (Settings.ESPFragments or Settings.ESPGlasses) then return end
    espFolder = Instance.new("Folder")
    espFolder.Name = "TAU_ESP"
    espFolder.Parent = CoreGui

    for _, obj in ipairs(workspace:GetDescendants()) do
        local n = obj.Name:lower()
        local frag, glass = false, false
        for _, v in ipairs(FragmentNames) do if n:find(v:lower()) then frag = true break end end
        for _, v in ipairs(GlassesNames) do if n:find(v:lower()) then glass = true break end end

        if (frag and Settings.ESPFragments) or (glass and Settings.ESPGlasses) then
            local h = Instance.new("Highlight")
            h.Adornee = obj
            h.FillColor = frag and Color3.fromRGB(255, 40, 40) or Color3.fromRGB(60, 160, 255)
            h.OutlineColor = Color3.fromRGB(255, 255, 255)
            h.FillTransparency = 0.5
            h.Parent = espFolder
        end
    end
end

local function flyParts()
    local hrp = getHRP()
    if not hrp then return end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and not obj.Anchored and not obj:IsDescendantOf(LocalPlayer.Character) then
            if (obj.Position - hrp.Position).Magnitude <= FlyRange then
                local old = obj:FindFirstChild("FlyForce")
                if old then old:Destroy() end
                local bv = Instance.new("BodyVelocity")
                bv.Name = "FlyForce"
                bv.MaxForce = Vector3.new(0, 1e9, 0)
                bv.Velocity = Vector3.new(0, UpForce, 0)
                bv.Parent = obj
                task.delay(2, function() if bv then bv:Destroy() end end)
            end
        end
    end
end

local function toggleFly()
    flyEnabled = not flyEnabled
    if flyEnabled then
        FlyBtn.Text = "Fly Parts: ON"
        FlyBtn.BackgroundColor3 = Color3.fromRGB(30, 130, 60)
        flyConnection = RunService.Heartbeat:Connect(flyParts)
        Notify("Fly ON")
    else
        FlyBtn.Text = "Fly Parts: OFF"
        FlyBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 65)
        if flyConnection then flyConnection:Disconnect() flyConnection = nil end
        Notify("Fly OFF")
    end
end

local function checkText()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end
    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextLabel") or g:IsA("TextButton") then
            local t = (g.Text or ""):lower()
            if t:find("system overload") or t:find("系统过载") then
                if not systemOverload then
                    systemOverload = true
                    Notify("System Overload!")
                end
            end
            if (t:find("5th") or t:find("fifth")) and t:find("fragment") then
                Notify("5th Fragment collected!")
            end
            if t:find("this round's gamemode is") then
                isNext = true
            end
            if t:find("double trouble") then currentMode = "Double Trouble" lastModeTime = tick()
            elseif t:find("dust devil") then currentMode = "Dust Devil" lastModeTime = tick()
            elseif t:find("infernado") then currentMode = "Infernado" lastModeTime = tick()
            elseif t:find("tartarus") then currentMode = "Tartarus" lastModeTime = tick()
            elseif t:find("doomsday") then currentMode = "Doomsday" lastModeTime = tick()
            elseif t:find("normal survival") then currentMode = "Normal Survival" lastModeTime = tick()
            elseif t:find("4-nado") or t:find("four tornadoes") then currentMode = "4-Nado" lastModeTime = tick()
            end
        end
    end
    if tick() - lastModeTime > 20 then
        currentMode = "Unknown"
        isNext = false
    end
end

-- Main Loop
task.spawn(function()
    while ScreenGui and ScreenGui.Parent do
        local frags = countFrags()
        checkText()

        StatusLabel.Text = Settings.Enabled and "Status: Running" or "Status: Off"
        StatusLabel.TextColor3 = Settings.Enabled and Color3.fromRGB(80, 255, 100) or Color3.fromRGB(255, 90, 90)
        FragLabel.Text = "Map Fragments: " .. frags
        CollectLabel.Text = "My Collected: " .. collectedFragments .. "/5"
        TartarusLabel.Text = collectedFragments >= 5 and "Tartarus: UNLOCKED" or "Tartarus: Locked"
        TartarusLabel.TextColor3 = collectedFragments >= 5 and Color3.fromRGB(255, 80, 80) or Color3.fromRGB(190, 190, 200)
        ModeLabel.Text = (isNext and "Next: " or "Mode: ") .. currentMode
        OverloadLabel.Text = systemOverload and "Overload: YES" or "Overload: No"
        OverloadLabel.TextColor3 = systemOverload and Color3.fromRGB(255, 80, 80) or Color3.fromRGB(190, 190, 200)
        GlassesLabel.Text = glassesFound and "Glasses: FOUND" or "Glasses: No"

        if Settings.Enabled then
            for _, obj in ipairs(workspace:GetDescendants()) do
                if not Settings.Enabled then break end
                local t = isTarget(obj)
                if t then
                    pickup(obj, t)
                    task.wait(0.15)
                end
            end
        end

        if Settings.ESPFragments or Settings.ESPGlasses then
            updateESP()
        else
            clearESP()
        end

        if Settings.AntiAFK and tick() - lastAFK > 50 then
            lastAFK = tick()
            local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if hum then hum.Jump = true end
        end

        task.wait(Settings.ScanSpeed)
    end
end)

    task.wait(0.1)
    local t = isTarget(obj)
    if t then
        Notify("New " .. t)
        playSound()
        pickup(obj, t)
    end
end)

-- Drag
local dragging, startPos, dragStart
Title.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)
Title.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local d = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.F then toggleFly() end
end)
