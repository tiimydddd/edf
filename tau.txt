local function playSound(soundId, volume)
    pcall(function()
        local s = Instance.new("Sound")
        s.SoundId = soundId
        s.Volume = volume or 0.55
        s.Parent = SoundService
        s:Play()
        s.Ended:Connect(function() s:Destroy() end)
        task.delay(3, function() if s and s.Parent then s:Destroy() end end)
    end)
end

local function getSettingsFile()
    local name = (Settings.SaveName or "default"):gsub("[^%w_%-]", "")
    if name == "" then name = "default" end
    return "tau_apple_" .. name .. ".txt"
end

local function saveSettings(force)
    if not writefile then return end
    if not force and not Settings.AutoSave then return end
    local data = table.concat({
        tostring(Settings.Enabled), tostring(Settings.OnlyFragments), tostring(Settings.FullAuto),
        tostring(Settings.FullAutoShelter), tostring(Settings.AntiAFK), tostring(Settings.TornadoPull),
        tostring(Settings.GodMode), tostring(Settings.FragmentFly), tostring(Settings.RejoinNoFrag),
        tostring(Settings.AutoSave), tostring(TORNADO_RANGE), tostring(Settings.SaveName or "default"),
        tostring(Settings.ESP), tostring(Settings.WalkSpeedOn), tostring(Settings.WalkSpeed),
        tostring(Settings.LootKey), tostring(Settings.LootExt), tostring(Settings.LootBat), tostring(Settings.LootOther),
    }, "\n")
    pcall(writefile, getSettingsFile(), data)
end

local function loadSettings()
    if not (isfile and readfile) then return end
    local file = getSettingsFile()
    if not isfile(file) then return end
    local ok, content = pcall(readfile, file)
    if not ok or not content then return end
    local lines = {}
    for line in content:gmatch("[^\r\n]+") do table.insert(lines, line) end
    if #lines < 9 then return end
    Settings.Enabled = lines[1] == "true"
    Settings.OnlyFragments = lines[2] == "true"
    Settings.FullAuto = lines[3] == "true"
    Settings.FullAutoShelter = lines[4] == "true"
    Settings.AntiAFK = lines[5] == "true"
    Settings.TornadoPull = lines[6] == "true"
    Settings.GodMode = lines[7] == "true"
    Settings.FragmentFly = lines[8] == "true"
    Settings.RejoinNoFrag = lines[9] == "true"
    if lines[10] then Settings.AutoSave = lines[10] == "true" end
    if lines[11] then
        local r = tonumber(lines[11])
        if r then TORNADO_RANGE = math.clamp(r, MIN_RANGE, MAX_RANGE) end
    end
    if lines[12] then Settings.SaveName = lines[12] end
    if lines[13] then Settings.ESP = lines[13] == "true" end
    if lines[14] then Settings.WalkSpeedOn = lines[14] == "true" end
    if lines[15] then
        local w = tonumber(lines[15])
        if w then Settings.WalkSpeed = math.clamp(w, 16, 80) end
    end
    if lines[16] then Settings.LootKey = lines[16] == "true" end
    if lines[17] then Settings.LootExt = lines[17] == "true" end
    if lines[18] then Settings.LootBat = lines[18] == "true" end
    if lines[19] then Settings.LootOther = lines[19] == "true" end
end

local function nameInList(objName, list)
    local n = (objName or ""):lower()
    for _, v in ipairs(list) do
        if n == v:lower() or n:find(v:lower(), 1, true) then return true end
    end
    return false
end

local function isFragment(obj)
    if not obj or not obj.Parent then return false end
    if not nameInList(obj.Name, FragmentNames) then return false end
    if obj.Parent and nameInList(obj.Parent.Name, FragmentNames) then return false end
    return true
end

local function isGlasses(obj)
    if not obj or not obj.Parent then return false end
    if not nameInList(obj.Name, GlassesNames) then return false end
    if obj.Parent and nameInList(obj.Parent.Name, GlassesNames) then return false end
    return true
end

local function isShelter(obj)
    return nameInList(obj.Name, ShelterNames)
end

local function lootKeyOf(obj)
    for _, t in ipairs(LOOT_TYPES) do
        if nameInList(obj.Name, t.names) then
            if obj.Parent and nameInList(obj.Parent.Name, t.names) then return nil end
            return t.key
        end
    end
    return nil
end

end

local function scanMapLoot()
    local counts = {}
    for _, t in ipairs(LOOT_TYPES) do counts[t.key] = 0 end
    local seen = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if not (obj:IsA("Model") or obj:IsA("Tool") or obj:IsA("BasePart") or obj:IsA("Folder")) then
            continue
        end
        local id = tostring(obj)
        if seen[id] then continue end
        for _, t in ipairs(LOOT_TYPES) do
            if nameInList(obj.Name, t.names) then
                if obj.Parent and nameInList(obj.Parent.Name, t.names) then break end
                counts[t.key] = counts[t.key] + 1
                seen[id] = true
                if t.key == "Fragment" then knownFragments[obj] = true
                elseif t.key == "Glasses" then knownGlasses[obj] = true
                else knownExtra[obj] = t.key end
                break
            end
        end
    end
    mapLootCounts = counts
    return counts
end

local function getBestPart(obj)
    if obj:IsA("BasePart") then return obj end
    local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt and prompt.Parent:IsA("BasePart") then return prompt.Parent end
    local click = obj:FindFirstChildWhichIsA("ClickDetector", true)
    if click and click.Parent:IsA("BasePart") then return click.Parent end
    local prefer = {"Handle", "Head", "Main", "Part", "MeshPart", "Union", "Primary"}
    for _, name in ipairs(prefer) do
        local p = obj:FindFirstChild(name, true)
        if p and p:IsA("BasePart") then return p end
    end
    return obj:FindFirstChildWhichIsA("BasePart", true)
end

local function tryFirePrompt(prompt)
    if not prompt then return end
    pcall(function()
        if fireproximityprompt then
            fireproximityprompt(prompt)
            task.wait(0.03)
            fireproximityprompt(prompt, 1)
        end
    end)
end

local recentlyTried = {}
local function pickup(obj, typ)
    if not obj or not obj.Parent then return end
    local key = tostring(obj)
    if recentlyTried[key] and tick() - recentlyTried[key] < TRIED_COOLDOWN then return end
    recentlyTried[key] = tick()
    local hrp = getHRP()
    if not hrp then return end
    local part = getBestPart(obj)
    if not part then return end
    local offsets = {
        Vector3.new(0, 2.0, 0), Vector3.new(0, 1.6, 1.5), Vector3.new(0, 1.6, -1.5),
        Vector3.new(1.5, 1.6, 0), Vector3.new(-1.5, 1.6, 0),
    }
    for _, offset in ipairs(offsets) do
        if not obj.Parent then break end
        hrp.CFrame = part.CFrame + offset
        task.wait(0.03)
        tryFirePrompt(obj:FindFirstChildWhichIsA("ProximityPrompt", true))
        local touch = part:FindFirstChildOfClass("TouchInterest")
        if touch then
            pcall(function()
                firetouchinterest(hrp, part, 0)
                task.wait(0.02)
                firetouchinterest(hrp, part, 1)
            end)
        end
        local click = obj:FindFirstChildWhichIsA("ClickDetector", true)
        if click then pcall(fireclickdetector, click) end
        task.wait(0.03)
    end
    task.delay(0.6, function()
        if not obj or not obj.Parent then
            if typ == "Fragment" then
                collectedFragments = math.min(5, collectedFragments + 1)
                sessionFrags = sessionFrags + 1
                Notify("碎片 " .. collectedFragments .. "/5（个人） 会话+" .. sessionFrags)
            elseif typ == "Glasses" then
                glassesFound = true
                sessionGlasses = sessionGlasses + 1
                Notify("眼镜已找到！Chad条件满足")
            end
        end
    end)
end

local function getObjectiveText()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return nil end
    local ok, text = pcall(function()
        local n = pg:FindFirstChild("Notifications")
        if not n then return nil end
        local sg = n:FindFirstChild("SelectedGamemode")
        if not sg then return nil end
        local obj = sg:FindFirstChild("Objective")
        if not obj then return nil end
        if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
            return obj.Text
        end
        local label = obj:FindFirstChildWhichIsA("TextLabel", true)
            or obj:FindFirstChildWhichIsA("TextButton", true)
        return label and label.Text or nil
    end)
    if ok then return text end
    return nil
end

local function matchModeFromText(raw)
    if not raw or #raw < 3 then return nil end
    local t = string.lower(raw)
    if t:find("unlikely the world will end") or t:find("chances are slim") then return nil end
    local best, bestLen = nil, 0
    for _, rule in ipairs(CURRENT_RULES) do
        if t:find(rule.pat, 1, true) and #rule.pat > bestLen then
            bestLen = #rule.pat
            best = rule.mode
        end
    end
    return best
end

local function checkAlerts()
    local objText = getObjectiveText()
    local bestCurrent = matchModeFromText(objText)
    local foundNext = nil
    local sawDoomsday = false
    local sawTartarusMode = false

    if objText and #objText >= 5 then
        local lower = string.lower(objText)
        if not lower:find("intermission") and not lower:find("voting") and not lower:find("vote") then
            markEnteredMap()
        end
        if lower:find("doomsday is occurring")
            or lower:find("survive the end of the world")
            or lower:find("may mother nature have mercy") then
            sawDoomsday = true
        end
        if lower:find("tartarus is occurring") or lower:find("tartarus has been summoned") then
            sawTartarusMode = true
        end
    end

    if not bestCurrent then
        local pg = LocalPlayer:FindFirstChild("PlayerGui")
        local notifications = pg and pg:FindFirstChild("Notifications")
        if notifications then
            for _, g in ipairs(notifications:GetDescendants()) do
                if g:IsA("TextLabel") or g:IsA("TextButton") then
                    local raw = g.Text or ""
                    if #raw < 5 or #raw > 220 then continue end
                    local t = string.lower(raw)
                    if t:find("unlikely the world will end") or t:find("chances are slim") then continue end
                    local n1, n2 = t:find("this round's gamemode is")
                    if n1 then
                        local after = raw:sub(n2 + 1):gsub("^[%s:]+", ""):gsub("%s+$", "")
                        if #after > 1 and #after < 40 then foundNext = after end
                    end
                    if not bestCurrent then
                        local m = matchModeFromText(raw)
                        if m then bestCurrent = m end
                    end
                    if t:find("doomsday is occurring") or t:find("survive the end of the world") then
                        sawDoomsday = true
                    end
                    if t:find("tartarus is occurring") or t:find("tartarus has been summoned") then
                        sawTartarusMode = true
                    end
                end
            end
        end
    end

    if foundNext and foundNext ~= lastNextMode then
        nextMode = foundNext
        lastNextMode = foundNext
        showBigAlert("下一模式: " .. foundNext, Color3.fromRGB(120, 200, 255), 5)
        Notify("下一模式: " .. foundNext)
    elseif foundNext then
        nextMode = foundNext
    end

    if sawTartarusMode and not tartarusUnlocked then
        tartarusUnlocked = true
        flashDanger()
        showBigAlert("⚠ TARTARUS 已出现！", Color3.fromRGB(255, 60, 60), 6)
        Notify("Tartarus 模式已出现！", true)
    end
    if sawDoomsday and not doomsdayDetected then
        doomsdayDetected = true
        flashDanger()
        showBigAlert("⚠ DOOMSDAY！", Color3.fromRGB(255, 80, 40), 6)
        Notify("检测到 Doomsday！", true)
    end

    if bestCurrent then
        if bestCurrent ~= currentMode then
            if bestCurrent == "Doomsday" or bestCurrent == "Tartarus" or bestCurrent == "Deadnado" then
                flashDanger()
                showBigAlert("⚠ 危险模式: " .. bestCurrent, Color3.fromRGB(255, 70, 70), 5)
            end
        end
        currentMode = bestCurrent
        lastModeTime = tick()
        if bestCurrent == "Doomsday" then doomsdayDetected = true end
        if bestCurrent == "Tartarus" and not tartarusUnlocked then
            tartarusUnlocked = true
            Notify("Tartarus 模式已出现！", true)
        end
        markEnteredMap()
    elseif tick() - lastModeTime > MODE_TIMEOUT then
        currentMode = "Unknown"
    end
end

local function checkFirstRoundNoFrag()
    if not Settings.RejoinNoFrag then
        rejoinStatus = "已关闭"
        return
    end
    if firstRoundChecked or isAlerting then return end
    if not enteredMap then
        rejoinStatus = "等待进入地图..."
        return
    end
    local left = math.ceil(LOOT_SPAWN_WAIT - (tick() - mapEnterTime))
    if left > 0 then
        rejoinStatus = "进图后检测中: " .. left .. "秒"
        return
    end
    firstRoundChecked = true
    scanMapLoot()
    local frags = mapLootCounts.Fragment or 0
    if frags <= 0 then frags = countFrags() end
    if frags <= 0 then
        rejoinStatus = "无碎片 → 仅提示"
        task.spawn(alertNoFragOnly)
    else
        rejoinStatus = "有碎片(" .. frags .. ") 留在此服"
        Notify("进图后检测到碎片: " .. frags .. "，留在此服")
    end
end

local function fireSignalObject(obj)
    if not obj then return end
    if obj:IsA("RemoteEvent") then
        pcall(function() obj:FireServer() end)
        pcall(function() obj:FireServer(true) end)
        pcall(function() obj:FireServer(LocalPlayer) end)
    elseif obj:IsA("BindableEvent") then
        pcall(function() obj:Fire() end)
    elseif obj:IsA("RemoteFunction") then
        pcall(function() obj:InvokeServer() end)
    elseif obj:IsA("Folder") or obj:IsA("Model") then
        for _, c in ipairs(obj:GetDescendants()) do
            if c:IsA("RemoteEvent") or c:IsA("BindableEvent") or c:IsA("RemoteFunction") then
                fireSignalObject(c)
            end
        end
    end
end

local function startSpectate()
    playClick()
    local ok, err = pcall(function()
        if typeof(firesignal) ~= "function" then
            error("当前执行器不支持 firesignal")
        end
        local assets = ReplicatedStorage:FindFirstChild("Assets")
        if not assets then error("找不到 Assets") end
        local signals = assets:FindFirstChild("Signals")
        if not signals then error("找不到 Signals") end
        local event = signals:FindFirstChild("Spectate")
        if not event then error("找不到 Spectate") end
        if isSpectating then
            pcall(function() firesignal(event.OnClientEvent, { A = false, reason = "ExitSpectate" }) end)
            pcall(function() firesignal(event.OnClientEvent, { A = false, reason = "LeaveSpectate" }) end)
            pcall(function() firesignal(event.OnClientEvent, { A = false }) end)
            isSpectating = false
            SpectateBtn.Text = "Open Spectate"
            Notify("已退出观战（再点可再进）")
        else
            firesignal(event.OnClientEvent, { A = true, reason = "EnterSpectate" })
            isSpectating = true
            SpectateBtn.Text = "Exit Spectate"
            Notify("已进入观战（再点可退出）")
        end
    end)
    if not ok then
        isSpectating = false
        SpectateBtn.Text = "Open Spectate"
        Notify("观战提示: " .. tostring(err) .. "（可手动点游戏观战）")
    end
end

local function forceFullHealth()
    local hum = getHumanoid()
    if not hum then return end
    pcall(function()
        if hum.Health < hum.MaxHealth then hum.Health = hum.MaxHealth end
        if hum:GetState() == Enum.HumanoidStateType.Dead then
            hum:ChangeState(Enum.HumanoidStateType.Running)
            hum.Health = hum.MaxHealth
        end
    end)
end

local function disableGodMode()
    if godConnection then godConnection:Disconnect() godConnection = nil end
    if healthConnection then healthConnection:Disconnect() healthConnection = nil end
    if antiFlingConnection then antiFlingConnection:Disconnect() antiFlingConnection = nil end
end

local function enableGodMode()
    disableGodMode()

    end)

            healthConnection = hum.HealthChanged:Connect(function()
                if Settings.GodMode then forceFullHealth() end

            hrp.AssemblyLinearVelocity = Vector3.zero

    end)
end

local function doTornadoPull()
    local hrp = getHRP()
    if not hrp then return end
    local myPos = hrp.Position
    local myChar = LocalPlayer.Character
    local parts
    local ok = pcall(function()
        parts = workspace:GetPartBoundsInRadius(myPos, TORNADO_RANGE)
    end)
    if not ok or not parts then return end
    for _, obj in ipairs(parts) do
        if obj:IsA("BasePart") and not obj.Anchored and obj.Parent then
            if myChar and obj:IsDescendantOf(myChar) then continue end
            local dist = (obj.Position - myPos).Magnitude
            if dist < SAFE_DISTANCE then continue end
            local direction = (myPos - obj.Position).Unit
            local sf = 1 - (dist / TORNADO_RANGE)
            if dist < 20 then sf = sf * 0.3 end
            local force = direction * (TORNADO_STRENGTH * sf)
                + Vector3.new(0, TORNADO_UP_FORCE * sf * 0.5, 0)
            if isFragment(obj) or (obj.Parent and isFragment(obj.Parent)) then
                force = force * 1.45
            end
            pcall(function() obj.AssemblyLinearVelocity = force end)
        end
    end
end

local function doFragmentFly()
    if not Settings.FragmentFly then return end
    local hrp = getHRP()
    if not hrp then return end
    local myPos = hrp.Position
    for obj in pairs(knownFragments) do
        if obj and obj.Parent then
            local part = getBestPart(obj)
            if part and not part.Anchored then
                local dist = (part.Position - myPos).Magnitude
                if dist <= FRAG_FLY_RANGE then
                    local old = part:FindFirstChild("FragFlyForce")
                    if old then old:Destroy() end
                    local bv = Instance.new("BodyVelocity")
                    bv.Name = "FragFlyForce"
                    bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
                    bv.Velocity = Vector3.new(
                        (math.random() - 0.5) * 2 * FRAG_FLY_SIDE,
                        FRAG_FLY_UP,
                        (math.random() - 0.5) * 2 * FRAG_FLY_SIDE
                    )
                    bv.Parent = part
                    task.delay(1.2, function() if bv and bv.Parent then bv:Destroy() end end)
                end
            end
        end
    end
end

local dragging, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)
TitleBar.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local d = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
