local function getSettingsFile()
    local name = (Settings.SaveName or "default"):gsub("[^%w_%-]", "")
    if name == "" then name = "default" end
    return "tau_apple_" .. name .. ".txt"
end

local function saveSettings(force)
    if not writefile then return end
    if not force and not Settings.AutoSave then return end
    local data = table.concat({
        tostring(Settings.Enabled),
        tostring(Settings.OnlyFragments),
        tostring(Settings.FullAuto),
        tostring(Settings.AntiAFK),
        tostring(Settings.TornadoPull),
        tostring(Settings.GodMode),
        tostring(Settings.FragmentFly),
        tostring(Settings.AutoSave),
        tostring(TORNADO_RANGE),
        tostring(Settings.SaveName or "default"),
    }, "\n")
    pcall(writefile, getSettingsFile(), data)
end

local function loadSettings()
    if not (isfile and readfile) then return end
    local file = getSettingsFile()
    if not isfile(file) then return end
    local ok, content = pcall(readfile, file)
    if not ok or not content then return end
    local lines = {}
    for line in content:gmatch("[^\r\n]+") do table.insert(lines, line) end
    if #lines < 8 then return end
    Settings.Enabled = lines[1] == "true"
    Settings.OnlyFragments = lines[2] == "true"
    Settings.FullAuto = lines[3] == "true"
    Settings.AntiAFK = lines[4] == "true"
    Settings.TornadoPull = lines[5] == "true"
    Settings.GodMode = lines[6] == "true"
    Settings.FragmentFly = lines[7] == "true"
    Settings.AutoSave = lines[8] == "true"
    if lines[9] then
        local r = tonumber(lines[9])
        if r then TORNADO_RANGE = math.clamp(r, MIN_RANGE, MAX_RANGE) end
    end
    if lines[10] then Settings.SaveName = lines[10] end
end

local function getHRP()
    local c = LocalPlayer.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function isFragment(obj)
    if not obj or not obj.Parent then return false end
    local n = (obj.Name or ""):lower()
    for _, v in ipairs(FragmentNames) do
        if n == v:lower() or n:find(v:lower(), 1, true) then
            if obj.Parent and isFragment(obj.Parent) then return false end
            return true
        end
    end
    return false
end

local function isGlasses(obj)
    if not obj or not obj.Parent then return false end
    local n = (obj.Name or ""):lower()
    for _, v in ipairs(GlassesNames) do
        if n:find(v:lower(), 1, true) then
            if obj.Parent and isGlasses(obj.Parent) then return false end
            return true
        end
    end
    return false
end

local function isShelter(obj)
    local n = (obj.Name or ""):lower()
    for _, v in ipairs(ShelterNames) do
        if n:find(v:lower(), 1, true) then return true end
    end
    return false
end

local function cleanCache(cache)
    for obj in pairs(cache) do
        if not obj or not obj.Parent then cache[obj] = nil end
    end
end

local function countFrags()
    cleanCache(knownFragments)
    local c = 0
    for _ in pairs(knownFragments) do c = c + 1 end
    return c
end

local function getBestPart(obj)
    if obj:IsA("BasePart") then return obj end
    local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt and prompt.Parent:IsA("BasePart") then return prompt.Parent end
    local click = obj:FindFirstChildWhichIsA("ClickDetector", true)
    if click and click.Parent:IsA("BasePart") then return click.Parent end
    local prefer = {"Handle", "Head", "Main", "Part", "MeshPart", "Union", "Primary"}
    for _, name in ipairs(prefer) do
        local p = obj:FindFirstChild(name, true)
        if p and p:IsA("BasePart") then return p end
    end
    return obj:FindFirstChildWhichIsA("BasePart", true)
end

local function tryFirePrompt(prompt)
    if not prompt then return end
    pcall(function()
        if fireproximityprompt then
            fireproximityprompt(prompt)
            task.wait(0.03)
            fireproximityprompt(prompt, 1)
        end
    end)
end

local recentlyTried = {}
local function pickup(obj, typ)
    if not obj or not obj.Parent then return end
    local key = tostring(obj)
    if recentlyTried[key] and tick() - recentlyTried[key] < TRIED_COOLDOWN then return end
    recentlyTried[key] = tick()
    local hrp = getHRP()
    if not hrp then return end
    local part = getBestPart(obj)
    if not part then return end
    local offsets = {
        Vector3.new(0, 2.0, 0), Vector3.new(0, 1.6, 1.5), Vector3.new(0, 1.6, -1.5),
        Vector3.new(1.5, 1.6, 0), Vector3.new(-1.5, 1.6, 0), Vector3.new(0, 2.8, 0),
    }
    for _, offset in ipairs(offsets) do
        if not obj.Parent then break end
        hrp.CFrame = part.CFrame + offset
        task.wait(0.028)
        tryFirePrompt(obj:FindFirstChildWhichIsA("ProximityPrompt", true))
        local touch = part:FindFirstChildOfClass("TouchInterest")
        if touch then
            pcall(function()
                firetouchinterest(hrp, part, 0)
                task.wait(0.02)
                firetouchinterest(hrp, part, 1)
            end)
        end
        local click = obj:FindFirstChildWhichIsA("ClickDetector", true)
        if click then pcall(fireclickdetector, click) end
        task.wait(0.03)
    end
    task.delay(0.6, function()
        if not obj or not obj.Parent then
            if typ == "Fragment" then
                collectedFragments = collectedFragments + 1
                Notify("Fragment " .. collectedFragments .. "/5")
            elseif typ == "Glasses" then
                glassesFound = true
                Notify("Glasses found!")
            end
        end
    end)
end

local function forceToShelter(silent)
    local hrp = getHRP()
    if not hrp then if not silent then Notify("No character") end return end
    local targetPart
    for _, obj in ipairs(workspace:GetDescendants()) do
        if isShelter(obj) then
            local part = getBestPart(obj)
            if part then targetPart = part break end
        end
    end
    if not targetPart then if not silent then Notify("Shelter not found") end return end
    hrp.CFrame = targetPart.CFrame + Vector3.new(0, 5, 0)
    if not silent then Notify("Teleported to Shelter") end
end

local function teleportToPlayer(player)
    local hrp = getHRP()
    if not hrp then return end
    local targetChar = player.Character
    if not targetChar then Notify(player.Name .. " has no character") return end
    local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
    if targetHRP then
        hrp.CFrame = targetHRP.CFrame + Vector3.new(0, 3, 0)
        Notify("Teleported to " .. player.Name)
    end
end

local function checkAlerts()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end

    local bestMode, bestLen = nil, 0
    local sawDoomsday, sawTartarus = false, false

    for _, g in ipairs(pg:GetDescendants()) do
        if g:IsA("TextLabel") or g:IsA("TextButton") then
            local raw = g.Text or ""
            if #raw < 4 or #raw > 180 then continue end
            local t = string.lower(raw)

            for _, rule in ipairs(MODE_RULES) do
                if t:find(rule.pat, 1, true) then
                    local len = #rule.pat
                    if len > bestLen then
                        bestLen = len
                        bestMode = rule.mode
                    end
                    if rule.mode == "Doomsday" then sawDoomsday = true end
                    if rule.mode == "Tartarus" then sawTartarus = true end
                end
            end

            -- 额外血月进度
            if (t:find("5th", 1, true) or t:find("fifth", 1, true) or t:find("all five", 1, true))
                and t:find("fragment", 1, true) then
                sawTartarus = true
            end
        end
    end

    if sawTartarus and not tartarusUnlocked then
        tartarusUnlocked = true
        Notify("Blood Moon / Tartarus unlocked (this server)")
    end
    if sawDoomsday and not doomsdayDetected then
        doomsdayDetected = true
        Notify("Doomsday on this server!")
    end

    if bestMode then
        currentMode = bestMode
        lastModeTime = tick()
        if bestMode == "Doomsday" then doomsdayDetected = true end
        if bestMode == "Tartarus" then tartarusUnlocked = true end
    elseif tick() - lastModeTime > MODE_TIMEOUT then
        currentMode = "Unknown"
    end
end

local godConnection, healthConnection, antiFlingConnection
local function getHumanoid()
    local char = LocalPlayer.Character
    return char and char:FindFirstChildOfClass("Humanoid")
end
local function forceFullHealth()
    local hum = getHumanoid()
    if not hum then return end
    pcall(function()
        if hum.Health < hum.MaxHealth then hum.Health = hum.MaxHealth end
        if hum:GetState() == Enum.HumanoidStateType.Dead then
            hum:ChangeState(Enum.HumanoidStateType.Running)
            hum.Health = hum.MaxHealth
        end
    end)
end
local function disableGodMode()
    if godConnection then godConnection:Disconnect() godConnection = nil end
    if healthConnection then healthConnection:Disconnect() healthConnection = nil end
    if antiFlingConnection then antiFlingConnection:Disconnect() antiFlingConnection = nil end
end
local function enableGodMode()
    disableGodMode()
    godConnection = RunService.Heartbeat:Connect(function()
        if Settings.GodMode then forceFullHealth() end
    end)
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            healthConnection = hum.HealthChanged:Connect(function()
                if Settings.GodMode then forceFullHealth() end
            end)
        end
    end
    antiFlingConnection = RunService.Heartbeat:Connect(function()
        if not Settings.GodMode then return end
        local hrp = getHRP()
        if hrp and hrp.AssemblyLinearVelocity.Magnitude > 80 then
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    end)
end

end)

local function doTornadoPull()
    local hrp = getHRP()
    if not hrp then return end
    local myPos = hrp.Position
    local myChar = LocalPlayer.Character
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and not obj.Anchored and obj.Parent then
            if myChar and obj:IsDescendantOf(myChar) then continue end
            local dist = (obj.Position - myPos).Magnitude
            if dist < SAFE_DISTANCE or dist >= TORNADO_RANGE then continue end
            local direction = (myPos - obj.Position).Unit
            local sf = 1 - (dist / TORNADO_RANGE)
            if dist < 20 then sf = sf * 0.3 end
            local force = direction * (TORNADO_STRENGTH * sf) + Vector3.new(0, TORNADO_UP_FORCE * sf * 0.5, 0)
            if isFragment(obj) or (obj.Parent and isFragment(obj.Parent)) then force = force * 1.45 end
            pcall(function() obj.AssemblyLinearVelocity = force end)
        end
    end
end

local function doFragmentFly()
    if not Settings.FragmentFly then return end
    local hrp = getHRP()
    if not hrp then return end
    local myPos = hrp.Position
    for obj in pairs(knownFragments) do
        if obj and obj.Parent then
            local part = getBestPart(obj)
            if part and not part.Anchored then
                local dist = (part.Position - myPos).Magnitude
                if dist <= FRAG_FLY_RANGE then
                    local old = part:FindFirstChild("FragFlyForce")
                    if old then old:Destroy() end
                    local bv = Instance.new("BodyVelocity")
                    bv.Name = "FragFlyForce"
                    bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
                    bv.Velocity = Vector3.new(
                        (math.random() - 0.5) * 2 * FRAG_FLY_SIDE,
                        FRAG_FLY_UP,
                        (math.random() - 0.5) * 2 * FRAG_FLY_SIDE
                    )
                    bv.Parent = part
                    task.delay(1.2, function() if bv and bv.Parent then bv:Destroy() end end)
                end
            end
        end
    end
end

        local now = tick()

        checkAlerts()

        StatusLabel.Text = "Status: " .. ((Settings.Enabled or Settings.FullAuto) and "Running" or "Off")
        StatusLabel.TextColor3 = (Settings.Enabled or Settings.FullAuto)
            and Color3.fromRGB(100, 220, 140) or Color3.fromRGB(255, 100, 100)
        FragLabel.Text = "Map Fragments: " .. frags
        CollectLabel.Text = "My Collected: " .. collectedFragments .. " / 5"
        GlassesLabel.Text = "Glasses: " .. (glassesFound and "FOUND" or "Not Found")

        ModeLabel.Text = "Mode: " .. currentMode
        ModeLabel.TextColor3 = currentMode == "Unknown"
            and Color3.fromRGB(180, 180, 195) or Color3.fromRGB(255, 200, 100)

        if tartarusUnlocked then
            BloodMoonLabel.Text = "Blood Moon: UNLOCKED (server)"
            BloodMoonLabel.TextColor3 = Color3.fromRGB(255, 90, 90)
        elseif collectedFragments >= 5 then
            BloodMoonLabel.Text = "Blood Moon: My 5/5"
            BloodMoonLabel.TextColor3 = Color3.fromRGB(255, 180, 80)
        else
            BloodMoonLabel.Text = "Blood Moon: Map " .. frags .. " | My " .. collectedFragments .. "/5"
            BloodMoonLabel.TextColor3 = Color3.fromRGB(180, 180, 195)
        end

        if doomsdayDetected or currentMode == "Doomsday" then
            DoomsdayLabel.Text = "Doomsday: YES"
            DoomsdayLabel.TextColor3 = Color3.fromRGB(255, 90, 90)
        else
            DoomsdayLabel.Text = "Doomsday: No"
            DoomsdayLabel.TextColor3 = Color3.fromRGB(180, 180, 195)
        end

        if (Settings.Enabled or Settings.FullAuto) and (now - lastScan) >= SCAN_INTERVAL then
            lastScan = now
            cleanCache(knownFragments)
            cleanCache(knownGlasses)
            local fragList, hrp = {}, getHRP()
            local myPos = hrp and hrp.Position or Vector3.zero
            for obj in pairs(knownFragments) do
                if obj and obj.Parent then
                    local part = getBestPart(obj)
                    if part then table.insert(fragList, {obj = obj, dist = (part.Position - myPos).Magnitude}) end
                end
            end
            table.sort(fragList, function(a, b) return a.dist < b.dist end)
            for _, data in ipairs(fragList) do pickup(data.obj, "Fragment") task.wait(0.07) end
            if not Settings.OnlyFragments then
                for obj in pairs(knownGlasses) do
                    if obj and obj.Parent then pickup(obj, "Glasses") task.wait(0.07) end
                end
            end
        end
        if Settings.FullAuto and (now - lastShelterTP) >= SHELTER_TP_INTERVAL then
            lastShelterTP = now
            forceToShelter(true)
        end
        if Settings.TornadoPull then doTornadoPull() end
        if Settings.FragmentFly then doFragmentFly() end
        task.wait(0.15)
    end
end)
