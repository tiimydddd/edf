local function inList(name,list)
	local n=(name or ""):lower()
	for _,v in ipairs(list) do if n==v:lower() or n:find(v:lower(),1,true) then return true end end
	return false
end
local function isFrag(o)
	if not o or not o.Parent then return false end
	if not inList(o.Name,FragN) then return false end
	if o.Parent and inList(o.Parent.Name,FragN) then return false end
	return true
end

end

local function scan()
	local c={Fragment=0,Glasses=0,Key=0,Ext=0,Bat=0}; local seen={}
	for _,o in ipairs(workspace:GetDescendants()) do
		if o:IsA("Model") or o:IsA("Tool") or o:IsA("BasePart") or o:IsA("Folder") then
			local id=tostring(o)
			if not seen[id] then
				if isFrag(o) then c.Fragment=c.Fragment+1; kFrag[o]=true; seen[id]=true
				elseif isGlass(o) then c.Glasses=c.Glasses+1; kGlass[o]=true; seen[id]=true
				elseif inList(o.Name,KeyN) and not (o.Parent and inList(o.Parent.Name,KeyN)) then c.Key=c.Key+1; kExtra[o]="Key"; seen[id]=true
				elseif inList(o.Name,ExtN) and not (o.Parent and inList(o.Parent.Name,ExtN)) then c.Ext=c.Ext+1; kExtra[o]="Ext"; seen[id]=true
				elseif inList(o.Name,BatN) and not (o.Parent and inList(o.Parent.Name,BatN)) then c.Bat=c.Bat+1; kExtra[o]="Bat"; seen[id]=true
				end
			end
		end
	end

local function pickup(o,typ)
	if not o or not o.Parent then return end
	local key=tostring(o)
	if tried[key] and tick()-tried[key]<TRY_CD then return end
	tried[key]=tick()
	local h=HRP(); if not h then return end
	local part=bestPart(o); if not part then return end
	local offs={Vector3.new(0,2,0),Vector3.new(0,1.6,1.5),Vector3.new(0,1.6,-1.5),Vector3.new(1.5,1.6,0),Vector3.new(-1.5,1.6,0)}
	for _,off in ipairs(offs) do
		if not o.Parent then break end
		h.CFrame=part.CFrame+off; task.wait(0.03)
		fireP(o:FindFirstChildWhichIsA("ProximityPrompt",true))
		local ti=part:FindFirstChildOfClass("TouchInterest")
		if ti then pcall(function() firetouchinterest(h,part,0); task.wait(0.02); firetouchinterest(h,part,1) end) end
		local cd=o:FindFirstChildWhichIsA("ClickDetector",true)
		if cd then pcall(fireclickdetector,cd) end
		task.wait(0.03)
	end
	task.delay(0.6,function()
		if not o or not o.Parent then
			if typ=="Fragment" then myFrags=math.min(5,myFrags+1); sessF=sessF+1; Notify("碎片 "..myFrags.."/5")
			elseif typ=="Glasses" then gotGlass=true; sessG=sessG+1; Notify("眼镜已找到") end
		end
	end)
end

local function onNextMode(name)
	if not name or name=="" or name==lastNext then return end
	lastNext=name
	nextMode=name

local function checkA()
	local ot=getObj(); local best=matchM(ot); local fn=nil; local sd,st=false,false
	if ot and #ot>=5 then
		local lo=string.lower(ot)
		if not lo:find("intermission") and not lo:find("voting") and not lo:find("vote") then markEnter() end
		if lo:find("doomsday") or lo:find("survive the end") then sd=true end
		if lo:find("tartarus") then st=true end
	end
	-- 扫 Notifications（含 SelectedGamemode 全树）抓「下一模式」
	local gui=LocalPlayer:FindFirstChild("PlayerGui")
	local n=gui and gui:FindFirstChild("Notifications")
	if n then
		for _,g in ipairs(n:GetDescendants()) do
			if g:IsA("TextLabel") or g:IsA("TextButton") then
				local raw=g.Text or ""
				if #raw>=5 and #raw<=220 then
					local t=string.lower(raw)
					-- 多种公布文案
					local patterns={
						"this round's gamemode is",
						"the gamemode is",
						"gamemode:",
						"next gamemode",
						"selected gamemode",
					}
					for _,pat in ipairs(patterns) do
						local a,b=t:find(pat,1,true)
						if a then
							local after=raw:sub(b+1):gsub("^[%s:]+",""):gsub("%s+$","")
							-- 去掉多余句号
							after=after:gsub("%.$",""):gsub("%s+$","")
							if #after>1 and #after<40 then fn=after break end
						end
					end
					if not best then local m=matchM(raw); if m then best=m end end
					if t:find("doomsday") then sd=true end
					if t:find("tartarus") then st=true end
				end
			end
		end
	end
	if fn then onNextMode(fn) end
	if st and not tartOK then tartOK=true; flashD(); showBig("⚠ TARTARUS",Color3.fromRGB(255,60,60),6); Notify("Tartarus!",true) end
	if sd and not doomOK then doomOK=true; flashD(); showBig("⚠ DOOMSDAY",Color3.fromRGB(255,80,40),6); Notify("Doomsday!",true) end
	if best then
		if best~=curMode and (best=="Doomsday" or best=="Tartarus" or best=="Deadnado") then flashD(); showBig("⚠ "..best,Color3.fromRGB(255,70,70),5) end
		curMode=best; lastModeT=tick()
		if best=="Doomsday" then doomOK=true end
		if best=="Tartarus" then tartOK=true end
		markEnter()
	elseif tick()-lastModeT>16 then curMode="Unknown" end
end

	end
end)
