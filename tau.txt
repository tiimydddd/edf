local function inL(name, list)
    local n = string.lower(tostring(name or ""))
    for _, v in ipairs(list) do
        local lv = string.lower(v)
        if n == lv or string.find(n, lv, 1, true) then return true end
    end
    return false
end

local function nameLooksFrag(name)
    local n = string.lower(tostring(name or ""))
    if inL(name, FragN) then return true end
    if (string.find(n, "fragment", 1, true) or string.find(n, "碎片", 1, true))
        and (string.find(n, "blood", 1, true) or string.find(n, "moon", 1, true) or string.find(n, "血月", 1, true)) then
        return true
    end
    return false
end

local function isF(o)
    if not o or not o.Parent then return false end
    if not nameLooksFrag(o.Name) then return false end
    if o.Parent and o.Parent ~= workspace and nameLooksFrag(o.Parent.Name) then return false end
    return true
end

local function isG(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, GlassN) then return false end
    if o.Parent and inL(o.Parent.Name, GlassN) then return false end
    return true
end

local function HRP()
    local c = LP.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function best(o)
    if not o then return nil end
    local ok, res = pcall(function()
        if o:IsA("BasePart") then return o end
        local pr = o:FindFirstChildWhichIsA("ProximityPrompt", true)
        if pr and pr.Parent and pr.Parent:IsA("BasePart") then return pr.Parent end
        return o:FindFirstChildWhichIsA("BasePart", true)
    end)
    return ok and res or nil
end

-- 存档
local function saveConfig()
    local ok, err = pcall(function()
        if typeof(writefile) ~= "function" then error("无 writefile") end
        writefile(SAVE_FILE, HS:JSONEncode({C = C}))
    end)
    if ok then
        SaveL.Text = "储存: 已保存"
        SaveL.TextColor3 = Color3.fromRGB(100, 220, 140)
        N("已保存")
    else
        SaveL.Text = "储存失败"
        SaveL.TextColor3 = Color3.fromRGB(255, 120, 100)
        N("保存失败: " .. tostring(err))
    end
end

local function loadConfig()
    local ok, err = pcall(function()
        if typeof(readfile) ~= "function" then error("无 readfile") end
        if typeof(isfile) == "function" and not isfile(SAVE_FILE) then error("无存档") end
        local data = HS:JSONDecode(readfile(SAVE_FILE))
        if typeof(data.C) == "table" then
            for k, v in pairs(data.C) do
                if C[k] ~= nil then C[k] = v end
            end
        end
    end)
    if ok then
        applyToggles()
        SaveL.Text = "储存: 已读取"
        SaveL.TextColor3 = Color3.fromRGB(100, 200, 255)
        N("已读取")
    else
        SaveL.Text = "无存档/失败"
        SaveL.TextColor3 = Color3.fromRGB(255, 160, 100)
    end
end
SaveB.MouseButton1Click:Connect(saveConfig)

local function toPlr(pl)
    if not pl or pl == LP then return end
    local c = pl.Character
    local th = c and c:FindFirstChild("HumanoidRootPart")
    local h = HRP()
    if not h then
        N("无角色")
        return
    end
    if not th then
        N(pl.Name .. " 无角色")
        return
    end
    pcall(function()
        h.CFrame = th.CFrame + Vector3.new(0, 3, 0)
    end)
    N("已传到 " .. pl.Name)
end

-- CallForHelp
local function fireCallForHelp(why)
    if not C.AutoHelp and why ~= "手动" then return end
    if tick() - lastHelpFire < 2 then return end
    lastHelpFire = tick()
    pcall(function()
        local Signals = RPS:FindFirstChild("Assets") and RPS.Assets:FindFirstChild("Signals")
        local ev = Signals and Signals:FindFirstChild("CallForHelp")
        if ev and ev:IsA("RemoteEvent") then
            pcall(function()
                ev:FireServer()
            end)
            N("CallForHelp " .. tostring(why or ""))
            if HelpL then
                HelpL.Text = "求救: 已发送 " .. tostring(why or "")
                HelpL.TextColor3 = Color3.fromRGB(100, 255, 140)
            end
        else
            N("未找到 CallForHelp")
        end
    end)
end

local function tryClickHelpGui()
    if not C.AutoHelp then return end
    pcall(function()
        local pg = LP:FindFirstChild("PlayerGui")
        if not pg then return end
        local root = pg:FindFirstChild("Obstructions") or pg
        for _, d in ipairs(root:GetDescendants()) do
            if d:IsA("GuiButton") then
                local n = string.lower(tostring(d.Name) .. " " .. tostring(d.Text or ""))
                if string.find(n, "help", 1, true) or string.find(n, "求救", 1, true) then
                    if typeof(firesignal) == "function" then
                        if d.MouseButton1Click then pcall(firesignal, d.MouseButton1Click) end
                        if d.Activated then pcall(firesignal, d.Activated) end
                    end
                    fireCallForHelp("UI")
                    return
                end
            end
        end
    end)
end

-- 核心逻辑
local espFolder = Instance.new("Folder")
espFolder.Name = "tau_esp"
espFolder.Parent = Gui

local function clrESP()
    for _, c in ipairs(espFolder:GetChildren()) do
        c:Destroy()
    end
end

local function addESP(o, col)
    local part = best(o)
    if not part then return end
    pcall(function()
        local hl = Instance.new("Highlight")
        hl.Adornee = o:IsA("Model") and o or part
        hl.FillColor = col
        hl.OutlineColor = col
        hl.FillTransparency = 0.55
        hl.OutlineTransparency = 0.2
        hl.Parent = espFolder
    end)
end

local function countKnown()
    local fc, gc = 0, 0
    for o in pairs(kF) do
        if o and o.Parent then fc = fc + 1 else kF[o] = nil end
    end
    for o in pairs(kG) do
        if o and o.Parent then gc = gc + 1 else kG[o] = nil end
    end
    return fc, gc
end


-- 大厅 Tartarus 预测: Workspace.Lobby.TartarusForecast.Screen...
local function readLobbyFrag()
    pcall(function()
        local bestN = nil
        local lobby = workspace:FindFirstChild("Lobby")
        local forecast = lobby and (lobby:FindFirstChild("TartarusForecast") or lobby:FindFirstChild("TrartarusForecast"))
        local screen = forecast and (forecast:FindFirstChild("Screen") or forecast:FindFirstChild("屏幕"))
        local function takeText(obj)
            if not obj then return end
            if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
                local tx = tostring(obj.Text or "")
                local a, b = string.match(tx, "(%d+)%s*/%s*(%d+)")
                if a and b and tonumber(b) == NEED_FRAG then
                    bestN = tonumber(a)
                elseif bestN == nil then
                    local only = string.match(tx, "(%d+)")
                    if only then
                        local n = tonumber(only)
                        if n and n >= 0 and n <= NEED_FRAG then bestN = n end
                    end
                end
            elseif obj:IsA("StringValue") then
                local a, b = string.match(tostring(obj.Value), "(%d+)%s*/%s*(%d+)")
                if a and tonumber(b) == NEED_FRAG then bestN = tonumber(a) end
            elseif obj:IsA("IntValue") or obj:IsA("NumberValue") then
                local n = tonumber(obj.Value)
                if n and n >= 0 and n <= NEED_FRAG then bestN = n end
            end
        end
        if screen then
            local node = screen:FindFirstChild("TartarusForecast")
                or screen:FindFirstChild("TrartarusForecast")
                or screen:FindFirstChildWhichIsA("TextLabel")
            takeText(node)
            for _, c in ipairs(screen:GetDescendants()) do
                takeText(c)
            end
        end
        if bestN ~= nil then
            serverFrag = bestN
        end
    end)
    if myF > serverFrag then
        serverFrag = math.min(NEED_FRAG, myF)
    end
    if TartL then
        if serverFrag >= NEED_FRAG then
            TartL.Text = "Tartarus: 已解锁 (5/5)"
            TartL.TextColor3 = Color3.fromRGB(100, 255, 140)
        else
            TartL.Text = string.format("Tartarus: 大厅 %d/%d", serverFrag, NEED_FRAG)
            TartL.TextColor3 = Color3.fromRGB(255, 160, 100)
        end
    end
end

local function fullScan()
    for o in pairs(kF) do
        if not o or not o.Parent then kF[o] = nil end
    end
    for o in pairs(kG) do
        if not o or not o.Parent then kG[o] = nil end
    end
    pcall(function()
        for _, o in ipairs(workspace:GetDescendants()) do
            if isF(o) then
                kF[o] = true
            elseif isG(o) then
                kG[o] = true
            end
        end
    end)
    lastFullScan = tick()
    local fc, gc = countKnown()
    if fc > 0 then markHasFrag("scan") end
    if gc > 0 then sawLootThisRound = true end
    return fc, gc
end

local function scan()
    if tick() - lastFullScan >= 3 then
        return fullScan()
    end
    return countKnown()
end

local function stillHasLoot()
    local fc, gc = countKnown()
    if fc > 0 then return true end
    if not C.OnlyFrag and gc > 0 then return true end
    return false
end

local function pickup(o, typ)
    pcall(function()
        if not o or not o.Parent then return end
        local key = tostring(o)
        if tried[key] and tick() - tried[key] < 0.8 then return end
        tried[key] = tick()
        if typ == "Fragment" or typ == "Glasses" then
            markHasFrag("pickup")
        end
        local h = HRP()
        if not h then return end
        local part = best(o)
        if not part then return end
        pcall(function()
            h.CFrame = part.CFrame + Vector3.new(0, 2, 0)
        end)
        local pr = o:FindFirstChildWhichIsA("ProximityPrompt", true)
        if pr and typeof(fireproximityprompt) == "function" then
            pcall(fireproximityprompt, pr)
        end
        if typeof(firetouchinterest) == "function" then
            pcall(function()
                firetouchinterest(h, part, 0)
                firetouchinterest(h, part, 1)
            end)
        end
        task.delay(0.4, function()
            if not o or not o.Parent then
                if typ == "Fragment" then
                    myF = myF + 1
                    serverFrag = math.min(NEED_FRAG, math.max(serverFrag, myF))
                    N("碎片 +" .. tostring(myF))
                    pcall(readLobbyFrag)
                elseif typ == "Glasses" then
                    gotG = true
                    N("眼镜已捡")
                end
            end
        end)
    end)
end

    N("没找到避难所")
end

local function queueScript()
    pcall(function()
        if typeof(queue_on_teleport) == "function" then
            queue_on_teleport([[task.wait(2) print("[tau] rejoin queue")]])
        end
    end)
end

local function doTeleport()
    local pid = game.PlaceId
    pcall(function()
        TS:Teleport(pid, LP)
    end)
    task.wait(0.2)
    pcall(function()
        TS:Teleport(PLACE, LP)
    end)
end

local function rejoin(reason)
    if rejoining then return end
    if hasFragLock() then
        pendingHop = false
        N("有碎片 → 不重进")
        return
    end
    if not C.Rejoin and not pendingHop then return end
    rejoining = true
    N("重进: " .. tostring(reason or "?"))
    queueScript()
    if not kickHandled then
        task.wait(2)
        kickHandled = true
    else
        task.wait(pendingHop and 1.5 or 0.5)
    end
    doTeleport()
    task.delay(10, function()
        rejoining = false
        pendingHop = false
        kickHandled = false
    end)
end

local function hop()
    if pendingHop or rejoining then return end
    if hasFragLock() then
        N("有碎片 → 禁止换服")
        pendingHop = false
        return
    end
    pendingHop = true
    kickHandled = false
    N("空图换服中...")
    queueScript()
    task.spawn(function()
        pcall(function()
            if LP and typeof(LP.Kick) == "function" then
                LP:Kick("tau hop")
            end
        end)
        task.wait(2.5)
        kickHandled = true
        task.wait(1)
        if pendingHop then
            if hasFragLock() then
                pendingHop = false
                N("换服前发现碎片 → 中止")
                return
            end
            rejoining = false
            rejoin("AutoHop")
        end
    end)
end

local function afk()
    -- 防挂机 / 降低 277 空闲踢
    pcall(function()
        if VU and typeof(VU.CaptureController) == "function" then
            VU:CaptureController()
        end
        if VU and typeof(VU.ClickButton2) == "function" then
            VU:ClickButton2(Vector2.new())
        end
    end)
    pcall(function()
        local h = HRP()
        if h then
            h.CFrame = h.CFrame * CFrame.new(0.05, 0, 0)
            task.wait(0.03)
            h.CFrame = h.CFrame * CFrame.new(-0.05, 0, 0)
        end
    end)
    pcall(function()
        local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.Jump = true
        end
    end)
end

    end)

local tScan, tLoot, tEsp, tAfk, tHelp = 0, 0, 0, 0, 0
local lastEspKey = ""

task.spawn(function()
    pcall(fullScan)
    pcall(readLobbyFrag)
    lastLobbyRead = tick()
    task.wait(0.3)
    pcall(loadConfig)
    while Gui and Gui.Parent do
        local ok, err = pcall(function()
            local now = tick()
            if hasFragLock() then
                pendingHop = false
            end

            if now - tScan >= 2 then
                tScan = now
                local fc, gc = scan()
                if tick() - lastLobbyRead >= 300 then
                    lastLobbyRead = tick()
                    readLobbyFrag()
                end
                FragL.Text = string.format("碎片: 地图%d | 本局%d | 大厅%d/%d", fc, myF, serverFrag, NEED_FRAG)
                GlassL.Text = (gotG or gc > 0) and "眼镜: 有" or "眼镜: 无"

                if hasFragLock() then
                    emptyScanStreak = 0
                    HopL.Text = "有碎片 → 禁止重进"
                    HopL.TextColor3 = Color3.fromRGB(100, 255, 140)
                elseif stillHasLoot() or not C.Auto then
                    emptyScanStreak = 0
                    HopL.Text = stillHasLoot() and "地图有物资" or "请开 Auto"
                else
                    emptyScanStreak = emptyScanStreak + 1
                    local ready = C.AutoHop and C.Auto and (now - serverJoinTime) >= MIN_STAY
                        and emptyScanStreak >= 3 and not sawLootThisRound and not hasFragLock()
                    if ready then
                        HopL.Text = "确认空图 → 换服"
                        HopL.TextColor3 = Color3.fromRGB(100, 255, 140)
                        hop()
                    else
                        local left = MIN_STAY - (now - serverJoinTime)
                        if left > 0 then
                            HopL.Text = string.format("等待进图 %.0fs", left)
                        else
                            HopL.Text = string.format("空图确认 %d/3", emptyScanStreak)
                        end
                        HopL.TextColor3 = Color3.fromRGB(140, 200, 255)
                    end
                end
            end

            if C.Auto and now - tLoot >= 0.9 then
                tLoot = now
                local h = HRP()
                local pos = h and h.Position or Vector3.zero
                local list = {}
                for o in pairs(kF) do
                    if o and o.Parent then
                        local p = best(o)
                        if p then
                            table.insert(list, {o = o, d = (p.Position - pos).Magnitude})
                        end
                    end
                end
                table.sort(list, function(a, b)
                    return a.d < b.d
                end)
                local n = 0
                for _, d in ipairs(list) do
                    if n >= 3 then break end
                    pickup(d.o, "Fragment")
                    n = n + 1
                end
                if not C.OnlyFrag and not gotG then
                    for o in pairs(kG) do
                        if o and o.Parent then
                            pickup(o, "Glasses")
                            break
                        end
                    end
                end
            end

            if C.ESP and now - tEsp >= 2 then
                tEsp = now
                local fc, gc = countKnown()
                local key = fc .. ":" .. gc
                if key ~= lastEspKey then
                    lastEspKey = key
                    clrESP()
                    local n = 0
                    for o in pairs(kF) do
                        if o and o.Parent and n < 12 then
                            addESP(o, Color3.fromRGB(255, 70, 70))
                            n = n + 1
                        end
                    end
                    for o in pairs(kG) do
                        if o and o.Parent then
                            addESP(o, Color3.fromRGB(80, 220, 220))
                        end
                    end
                end
            elseif not C.ESP then
                lastEspKey = ""
                clrESP()
            end

            if C.AFK and now - tAfk >= 25 then
                tAfk = now
                afk()
            end

            if C.AutoHelp and now - tHelp >= 4 then
                tHelp = now
                tryClickHelpGui()
            end
        end)
        if not ok and err then
            warn("[tau] loop:", err)
        end
        task.wait(0.3)
    end
end)
