local function inL(name, list)
    local n = string.lower(tostring(name or ""))
    for _, v in ipairs(list) do
        if n == string.lower(v) or string.find(n, string.lower(v), 1, true) then return true end
    end
    return false
end

local function isF(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, FragN) then return false end
    if o.Parent and inL(o.Parent.Name, FragN) then return false end
    return true
end

local function isG(o)
    if not o or not o.Parent then return false end
    if not inL(o.Name, GlassN) then return false end
    if o.Parent and inL(o.Parent.Name, GlassN) then return false end
    return true
end

local function detectMode()
    pcall(function()
        local g = LP:FindFirstChild("PlayerGui")
        if not g then return end
        local n = g:FindFirstChild("Notifications")
        local sg = n and n:FindFirstChild("SelectedGamemode")
        local o = sg and sg:FindFirstChild("Objective")
        local txt = o and (o:IsA("TextLabel") and o.Text or nil)
        if not txt then
            local l = o and o:FindFirstChildWhichIsA("TextLabel", true)
            txt = l and l.Text
        end
        if not txt or #txt < 2 then return end
        local lo = string.lower(txt)
        if string.find(lo, "tartarus") then currentMode = "Tartarus"
        elseif string.find(lo, "doomsday") or string.find(lo, "doom") then currentMode = "Doomsday"
        elseif string.find(lo, "chad") then currentMode = "Chad"
        elseif string.find(lo, "twisty") then currentMode = "Twisty"
        elseif string.find(lo, "weeping") then currentMode = "Weeping Angel"
        elseif string.find(lo, "intermission") then currentMode = "Intermission"
        end
        ModeL.Text = "模式: " .. currentMode
    end)
    pcall(function()
        local Signals = RPS:FindFirstChild("Assets") and RPS.Assets:FindFirstChild("Signals")
        local Ev = Signals and Signals:FindFirstChild("Event")
        if Ev and not Ev:GetAttribute("tauHooked") then
            Ev:SetAttribute("tauHooked", true)
            Ev.OnClientEvent:Connect(function(data)
                if typeof(data) ~= "table" or not data.reason then return end
                if data.reason == "Cutscene" and data.F and data.F.Name then
                    currentMode = data.F.Name
                    ModeL.Text = "模式: " .. currentMode
                elseif data.reason == "TartarusClock" or data.reason == "TartarusSummon" then
                    currentMode = "Tartarus"
                    ModeL.Text = "模式: Tartarus"
                elseif data.reason == "DoomsDayForecast" then
                    currentMode = "Doomsday"
                    ModeL.Text = "模式: Doomsday"
                end
            end)
        end
    end)
end

local function changeMusic(id)
    id = tostring(id or "")
    if id == "" then N("请输入音乐ID") return false end
    if not string.find(id, "rbxassetid://") then id = "rbxassetid://" .. id end
    local ok = false
    pcall(function()
        local music = SS:FindFirstChild("Music")
        local targets = {}
        if music then
            for _, name in ipairs({"Gamemode", "BonusDisaster"}) do
                local s = music:FindFirstChild(name)
                if s and s:IsA("Sound") then table.insert(targets, s) end
            end
        end
        if #targets == 0 then
            for _, s in ipairs(SS:GetDescendants()) do
                if s:IsA("Sound") and s.IsPlaying then table.insert(targets, s) end
            end
        end
        for _, s in ipairs(targets) do
            if not originalIds[s] then originalIds[s] = s.SoundId end
            s.SoundId = id
            ok = true
        end
    end)
    if ok then N("音乐已更换") else N("未找到可换音乐") end
    return ok
end

    end)
end

local function saveConfig()
    local ok, err = pcall(function()
        if typeof(writefile) ~= "function" then error("无 writefile") end
        writefile(SAVE_FILE, HS:JSONEncode({C = C, musicHistory = musicHistory}))
    end)
    if ok then
        SaveL.Text = "储存: 已保存"
        SaveL.TextColor3 = Color3.fromRGB(100, 220, 140)
        N("已保存")
    else
        SaveL.Text = "储存失败"
        SaveL.TextColor3 = Color3.fromRGB(255, 120, 100)
        N("保存失败: " .. tostring(err))
    end
end

local function loadConfig()
    local ok, err = pcall(function()
        if typeof(readfile) ~= "function" then error("无 readfile") end
        if typeof(isfile) == "function" and not isfile(SAVE_FILE) then error("无存档") end
        local data = HS:JSONDecode(readfile(SAVE_FILE))
        if typeof(data.C) == "table" then
            for k, v in pairs(data.C) do if C[k] ~= nil then C[k] = v end end
        end
        musicHistory = typeof(data.musicHistory) == "table" and data.musicHistory or {}
    end)
    if ok then
        applyToggles()
        SaveL.Text = "储存: 已读取"
        SaveL.TextColor3 = Color3.fromRGB(100, 200, 255)
        N("已读取")
    else
        SaveL.Text = "无存档/失败"
        SaveL.TextColor3 = Color3.fromRGB(255, 160, 100)
    end
end

local function scan()
    for o in pairs(kF) do kF[o] = nil end
    for o in pairs(kG) do kG[o] = nil end
    local fc, gc = 0, 0
    for _, o in ipairs(workspace:GetDescendants()) do
        if isF(o) then kF[o] = true; fc = fc + 1
        elseif isG(o) then kG[o] = true; gc = gc + 1 end
    end
    return fc, gc
end

local function pickup(o, typ)
    if not o or not o.Parent then return end
    local key = tostring(o)
    if tried[key] and tick() - tried[key] < 0.55 then return end
    tried[key] = tick()
    local h = HRP()
    if not h then return end
    local part = best(o)
    if not part then return end
    local offs = {
        Vector3.new(0, 2, 0), Vector3.new(0, 1.5, 1.5), Vector3.new(0, 1.5, -1.5),
        Vector3.new(1.5, 1.5, 0), Vector3.new(-1.5, 1.5, 0),
    }
    for _, off in ipairs(offs) do
        if not o.Parent then break end
        pcall(function() h.CFrame = part.CFrame + off end)
        task.wait(0.03)
        local pr = o:FindFirstChildWhichIsA("ProximityPrompt", true)
        if pr and typeof(fireproximityprompt) == "function" then
            pcall(fireproximityprompt, pr)
            pcall(fireproximityprompt, pr, 1)
        end
        if typeof(firetouchinterest) == "function" then
            pcall(function()
                firetouchinterest(h, part, 0)
                task.wait(0.02)
                firetouchinterest(h, part, 1)
            end)
        end
        task.wait(0.02)
    end
    task.delay(0.5, function()
        if not o or not o.Parent then
            if typ == "Fragment" then
                myF = math.min(5, myF + 1)
                N("碎片 " .. myF .. "/5")
            elseif typ == "Glasses" then
                gotG = true
                N("眼镜已找到")
            end
        end
    end)
end


local function toPart(part, label)
    local h = HRP()
    if not h then N("无角色") return end
    if not part then N("无目标") return end
    pcall(function()
        h.CFrame = part.CFrame + Vector3.new(0, 4, 0)
    end)
    N("已传到 " .. tostring(label or "?"))
end

local function toPlr(pl)
    if not pl or pl == LP then return end
    local c = pl.Character
    local th = c and c:FindFirstChild("HumanoidRootPart")
    if th then
        toPart(th, pl.Name)
    else
        N(pl.Name .. " 无角色")
    end
end

local function itemType(o)
    if isF(o) then return "碎片", Color3.fromRGB(255, 100, 100) end
    if isG(o) then return "眼镜", Color3.fromRGB(80, 220, 220) end
    if inL(o.Name, KeyN) then return "钥匙", Color3.fromRGB(255, 200, 80) end
    if inL(o.Name, ExtN) then return "灭火器", Color3.fromRGB(255, 140, 60) end
    if inL(o.Name, BatN) then return "棒球棒", Color3.fromRGB(180, 220, 100) end
    if inL(o.Name, ShelN) then return "避难所", Color3.fromRGB(140, 180, 255) end
    return nil
end

local function refPlrList()
    for _, ch in ipairs(PlrF:GetChildren()) do
        if ch:IsA("TextButton") then ch:Destroy() end
    end
    local y = 0
    local list = {}
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= LP then table.insert(list, pl) end
    end
    table.sort(list, function(a, b) return a.Name:lower() < b.Name:lower() end)
    for _, pl in ipairs(list) do
        local b = Instance.new("TextButton")
        b.Size = UDim2.new(1, -4, 0, 22)
        b.BackgroundColor3 = Color3.fromRGB(40, 40, 52)
        b.Text = "  " .. pl.Name
        b.TextColor3 = Color3.fromRGB(220, 220, 230)
        b.Font = Enum.Font.Gotham
        b.TextSize = 11
        b.TextXAlignment = Enum.TextXAlignment.Left
        b.Parent = PlrF
        Instance.new("UICorner", b).CornerRadius = UDim.new(0, 4)
        local target = pl
        b.MouseButton1Click:Connect(function()
            toPlr(target)
        end)
        y = y + 24
    end
    PlrF.CanvasSize = UDim2.new(0, 0, 0, math.max(y, 1))
    if #list == 0 then
        N("没有其他玩家")
    end
end

local function refItemList()
    for _, ch in ipairs(ItemF:GetChildren()) do
        if ch:IsA("TextButton") then ch:Destroy() end
    end
    local h = HRP()
    local pos = h and h.Position or Vector3.zero
    local rows = {}
    local seen = {}
    for _, o in ipairs(workspace:GetDescendants()) do
        local typ, col = itemType(o)
        if typ then
            local part = best(o)
            if part and not seen[part] then
                seen[part] = true
                local d = (part.Position - pos).Magnitude
                table.insert(rows, {o = o, part = part, typ = typ, col = col, d = d, name = o.Name})
            end
        end
    end
    table.sort(rows, function(a, b) return a.d < b.d end)
    local y = 0
    for i, r in ipairs(rows) do
        if i > 40 then break end -- 防太多卡顿
        local b = Instance.new("TextButton")
        b.Size = UDim2.new(1, -4, 0, 22)
        b.BackgroundColor3 = Color3.fromRGB(40, 40, 52)
        b.Text = string.format("  [%s] %s  %.0f", r.typ, r.name, r.d)
        b.TextColor3 = r.col
        b.Font = Enum.Font.Gotham
        b.TextSize = 10
        b.TextXAlignment = Enum.TextXAlignment.Left
        b.Parent = ItemF
        Instance.new("UICorner", b).CornerRadius = UDim.new(0, 4)
        local part, label = r.part, r.typ .. " " .. r.name
        b.MouseButton1Click:Connect(function()
            toPart(part, label)
        end)
        y = y + 24
    end
    ItemF.CanvasSize = UDim2.new(0, 0, 0, math.max(y, 1))
    N("物品 " .. #rows .. " 个")
end

local tScan, tShel, tEsp, tAfk, tMode = 0, 0, 0, 0, 0
task.spawn(function()
    while Gui and Gui.Parent do
        pcall(function()
            local now = tick()
            if now - tScan >= 0.5 then
                tScan = now
                local fc, gc = scan()
                FragL.Text = string.format("碎片: 地图%d | 我%d/5", fc, myF)
                GlassL.Text = (gotG or gc > 0) and "眼镜: 有" or "眼镜: 未找到"
                if C.Auto then
                    local h = HRP()
                    local pos = h and h.Position or Vector3.zero
                    local list = {}
                    for o in pairs(kF) do
                        if o and o.Parent then
                            local p = best(o)
                            if p then table.insert(list, {o = o, d = (p.Position - pos).Magnitude}) end
                        end
                    end
                    table.sort(list, function(a, b) return a.d < b.d end)
                    for _, d in ipairs(list) do
                        pickup(d.o, "Fragment")
                        task.wait(0.07)
                    end
                    for o in pairs(kF) do
                        if o and o.Parent then
                            tried[tostring(o)] = 0
                            pickup(o, "Fragment")
                            task.wait(0.05)
                        end
                    end
                    if not C.OnlyFrag then
                        for o in pairs(kG) do
                            if o and o.Parent then
                                pickup(o, "Glasses")
                                task.wait(0.07)
                            end
                        end
                    end
                end
            end
            if C.ESP and now - tEsp >= 1 then
                tEsp = now
                clrESP()
                for o in pairs(kF) do if o and o.Parent then addESP(o, Color3.fromRGB(255, 70, 70)) end end
                for o in pairs(kG) do if o and o.Parent then addESP(o, Color3.fromRGB(80, 220, 220)) end end
            end
            if C.ShelAuto and now - tShel >= 30 then
                tShel = now
                toShel()
            end
            if C.AFK and now - tAfk >= 30 then
                tAfk = now
                afk()
            end
            if now - tMode >= 1.2 then
                tMode = now
                detectMode()
            end
        end)
        task.wait(0.15)
    end
end)
